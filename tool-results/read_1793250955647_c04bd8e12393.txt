     1→// ============================================================================
     2→// PRISMA SCHEMA - MADRASAH DINIYAH TERPADU (SIM MADIN)
     3→// ============================================================================
     4→// Filosofi: "Satu pintu, bertahap, fondasi kuat"
     5→// - Role + Permission (bukan hard-code akses per aplikasi)
     6→// - Satu Data Santri (single source of truth)
     7→// - Satu Data Wali (satu wali bisa punya banyak anak)
     8→// - Riwayat Data (audit log & soft delete)
     9→// - Tahun Ajaran & Semester (semua data akademik terikat)
    10→// - Data Keuangan Terhubung Santri
    11→// - Modular (siap V2-V14 tanpa bongkar total)
    12→// ============================================================================
    13→

    16→  provider = "prisma-client-js"

    25→// ============================================================================
    26→// 1. AUTENTIKASI, ROLE & PERMISSION
    27→// ============================================================================
    28→// Sistem hak akses berbasis Role + Permission.
    29→// Setiap User bisa punya banyak Role, setiap Role punya banyak Permission.
    30→// Contoh: Guru + Wali Kelas = 1 User dengan 2 Role yang aktif.
    31→// ============================================================================
    32→
    33→model User {
    34→  id            String   @id @default(cuid())
    35→  email         String   @unique
    36→  username      String   @unique
    37→  passwordHash  String
    38→  name          String
    39→  phone         String?
    40→  avatarUrl     String?
    41→  firebaseUid   String?  @unique // untuk integrasi Supabase Auth / Firebase Auth
    42→  isActive      Boolean  @default(true)
    43→  lastLoginAt   DateTime?
    44→  createdAt     DateTime @default(now())
    45→  updatedAt     DateTime @updatedAt
    46→  deletedAt     DateTime? // soft delete
    47→
    48→  // Relasi
    49→  userRoles     UserRole[]
    50→  auditLogs     AuditLog[]
    51→  payments      Payment[]      @relation("RecordedBy")
    52→  announcements Announcement[] @relation("CreatedBy")
    53→  santri        Santri?
    54→  guru          Guru?
    55→  wali          Wali?
    56→
    57→  @@index([isActive])
    58→  @@index([deletedAt])
    59→}
    60→
    61→model Role {
    62→  id          String   @id @default(cuid())
    63→  code        String   @unique // SUPER_ADMIN, KEPALA, BENDAHARA, GURU, WALI
    64→  name        String
    65→  description String?
    66→  sortOrder   Int      @default(0)
    67→  isSystem    Boolean  @default(false) // role sistem, tidak bisa dihapus
    68→  createdAt   DateTime @default(now())
    69→  updatedAt   DateTime @updatedAt
    70→
    71→  userRoles   UserRole[]
    72→  permissions RolePermission[]
    73→}
    74→
    75→model Permission {
    76→  id          String   @id @default(cuid())
    77→  code        String   @unique // contoh: "santri:create", "nilai:read"
    78→  name        String
    79→  module      String // "santri", "nilai", "keuangan", dll
    80→  description String?
    81→  createdAt   DateTime @default(now())
    82→
    83→  roles       RolePermission[]
    84→}
    85→
    86→model UserRole {
    87→  id        String   @id @default(cuid())
    88→  userId    String
    89→  roleId    String
    90→  isActive  Boolean  @default(true)
    91→  createdAt DateTime @default(now())
    92→
    93→  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
    94→  role      Role     @relation(fields: [roleId], references: [id])
    95→
    96→  @@unique([userId, roleId])
    97→  @@index([userId])
    98→  @@index([roleId])
    99→}
   100→
   101→model RolePermission {
   102→  id           String   @id @default(cuid())
   103→  roleId       String
   104→  permissionId String
   105→  createdAt    DateTime @default(now())
   106→
   107→  role         Role       @relation(fields: [roleId], references: [id], onDelete: Cascade)
   108→  permission   Permission @relation(fields: [permissionId], references: [id])
   109→
   110→  @@unique([roleId, permissionId])
   111→  @@index([roleId])
   112→  @@index([permissionId])
   113→}

   227→model Wali {
   228→  id          String   @id @default(cuid())

   240→  otpCode     String?  // OTP aktif saat ini
   241→  otpExpiresAt DateTime?

   243→  otpRequestCount Int  @default(0) // anti-spam: max 5/jam
   244→  otpRequestResetAt DateTime?

   246→  lockedUntil DateTime?

   248→  createdAt   DateTime @default(now())
   249→  updatedAt   DateTime @updatedAt
   250→  deletedAt   DateTime?
   251→
   252→  user        User?    @relation(fields: [userId], references: [id])

   254→  santriLinks SantriWali[] // semua relasi wali-santri
   255→  activations WaliActivation[]
   256→  phoneChangeLogs PhoneChangeLog[]
   257→
   258→  @@index([phone])
   259→  @@index([activationStatus])
   260→  @@index([deletedAt])
   261→}
