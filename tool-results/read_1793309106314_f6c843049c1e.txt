   586→export function formatRupiah(amount: number): string {
   587→  return new Intl.NumberFormat("id-ID", {
   588→    style: "currency",
   589→    currency: "IDR",
   590→    minimumFractionDigits: 0,
   591→    maximumFractionDigits: 0,
   592→  }).format(amount)
   593→}

   595→export function formatDate(date: Date | string, withTime = false): string {
   596→  const d = typeof date === "string" ? new Date(date) : date
   597→  if (withTime) {
   598→    return new Intl.DateTimeFormat("id-ID", {
   599→      day: "numeric",
   600→      month: "long",
   601→      year: "numeric",
   602→      hour: "2-digit",
   603→      minute: "2-digit",
   604→    }).format(d)
   605→  }
   606→  return new Intl.DateTimeFormat("id-ID", {
   607→    day: "numeric",
   608→    month: "long",
   609→    year: "numeric",
   610→  }).format(d)
   611→}
   612→
   613→export function formatDateShort(date: Date | string): string {
   614→  const d = typeof date === "string" ? new Date(date) : date
   615→  return new Intl.DateTimeFormat("id-ID", {
   616→    day: "2-digit",
   617→    month: "2-digit",
   618→    year: "numeric",
   619→  }).format(d)
   620→}
