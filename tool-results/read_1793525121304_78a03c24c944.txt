   112→function LoginScreen({ onLogin, onForgotPin }: { onLogin: (token: string, wali: WaliData, santriList: SantriData[]) => void; onForgotPin?: () => void }) {
   113→  const [phone, setPhone] = useState("")
   114→  const [pin, setPin] = useState("")
   115→  const [loading, setLoading] = useState(false)
   116→  const [showPin, setShowPin] = useState(false)
   117→
   118→  const handleLogin = async () => {
   119→    if (!phone || !pin) {
   120→      Alert.alert("Validasi", "Nomor WhatsApp dan PIN wajib diisi")
   121→      return
   122→    }
   123→    if (pin.length !== 6) {
   124→      Alert.alert("Validasi", "PIN harus 6 digit")
   125→      return
   126→    }
   127→
   128→    setLoading(true)
   129→    try {
   130→      const data = await api.post("/api/wali/login", { phone, pin })
   131→      if (data.success) {
   132→        waliToken = data.token
   133→        await saveWaliSession(data.token, data.wali, data.santriList)
   134→        onLogin(data.token, data.wali, data.santriList)
   135→      } else {
   136→        Alert.alert("Login Gagal", data.error || "Terjadi kesalahan")
   137→      }
   138→    } catch (e: any) {
   139→      Alert.alert("Login Gagal", e.message)
   140→    } finally {
   141→      setLoading(false)
   142→    }
   143→  }
   144→
   145→  return (
   146→    <SafeAreaView style={styles.loginContainer}>
   147→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   148→      <ScrollView contentContainerStyle={styles.loginScroll}>
   149→        <View style={styles.loginLogo}>
   150→          <Image source={require("../assets/logo-madrasah.png")} style={{ width: 120, height: 120, resizeMode: "contain", marginBottom: 8 }} />
   151→          <Text style={styles.loginTitle}>SIM MADIN</Text>
   152→          <Text style={styles.loginSubtitle}>Wali Santri</Text>
   153→        </View>
   154→
   155→        <View style={styles.loginCard}>
   156→          <Text style={styles.loginCardTitle}>Masuk ke Akun</Text>
   157→          <Text style={styles.loginCardDesc}>Masuk dengan Nomor WhatsApp & PIN</Text>
   158→
   159→          <Text style={styles.label}>Nomor WhatsApp</Text>
   160→          <View style={styles.inputContainer}>
   161→            <Ionicons name="call" size={20} color={Colors.textMuted} style={styles.inputIcon} />
   162→            <TextInput
   163→              style={styles.input}
   164→              placeholder="08xxx atau 628xxx"
   165→              value={phone}
   166→              onChangeText={setPhone}
   167→              keyboardType="phone-pad"
   168→              autoCapitalize="none"

   172→          <Text style={styles.label}>PIN (6 digit)</Text>

   212→// --- Forgot PIN Screen (3-step OTP flow) ---
   213→function ForgotPinScreen({ onBack }: { onBack: () => void }) {
   214→  const [step, setStep] = useState<"phone" | "otp" | "setPin">("phone")
   215→  const [phone, setPhone] = useState("")
   216→  const [otp, setOtp] = useState("")
   217→  const [newPin, setNewPin] = useState("")
   218→  const [confirmPin, setConfirmPin] = useState("")
   219→  const [loading, setLoading] = useState(false)
   220→  const [verifyToken, setVerifyToken] = useState("")
   221→
   222→  const handleRequestOtp = async () => {
   223→    if (!phone) { Alert.alert("Validasi", "Nomor WhatsApp wajib diisi"); return }
   224→    setLoading(true)
   225→    try {
   226→      const res = await api.post("/api/wali/request-otp", { phone, type: "FORGOT_PIN" })
   227→      Alert.alert("OTP Terkirim", res.devOtp ? `Kode OTP (DEV): ${res.devOtp}` : res.message)
   228→      setStep("otp")
   229→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   230→    finally { setLoading(false) }
   231→  }
   232→
   233→  const handleVerifyOtp = async () => {
   234→    if (!otp || otp.length !== 6) { Alert.alert("Validasi", "OTP harus 6 digit"); return }
   235→    setLoading(true)
   236→    try {
   237→      const res = await api.post("/api/wali/verify-otp", { phone, otpCode: otp })
   238→      if (res.verifyToken) {
   239→        setVerifyToken(res.verifyToken)
   240→        setStep("setPin")
   241→        Alert.alert("Terverifikasi", "Silakan buat PIN baru")
   242→      } else { Alert.alert("Gagal", "Token verifikasi tidak diterima") }
   243→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   244→    finally { setLoading(false) }
   245→  }
   246→
   247→  const handleSetPin = async () => {
   248→    if (!newPin || newPin.length !== 6) { Alert.alert("Validasi", "PIN harus 6 digit"); return }
   249→    if (newPin !== confirmPin) { Alert.alert("Validasi", "PIN tidak cocok"); return }
   250→    setLoading(true)
   251→    try {
   252→      await api.post("/api/wali/set-pin", { verifyToken, pin: newPin })
   253→      Alert.alert("Berhasil", "PIN berhasil diubah. Silakan login dengan PIN baru.", [{ text: "OK", onPress: onBack }])
   254→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   255→    finally { setLoading(false) }
   256→  }
   257→
   258→  return (
   259→    <SafeAreaView style={styles.loginContainer}>
   260→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   261→      <ScrollView contentContainerStyle={styles.loginScroll}>
   262→        <View style={styles.loginLogo}>
   263→          <Ionicons name="key" size={80} color={Colors.white} />
   264→          <Text style={styles.loginTitle}>Lupa PIN</Text>
   265→          <Text style={styles.loginSubtitle}>Reset PIN via OTP</Text>
   266→        </View>
   267→
   268→        <View style={styles.loginCard}>
   269→          {step === "phone" && (
   270→            <>
   271→              <Text style={styles.loginCardTitle}>Masukkan Nomor WhatsApp</Text>
   272→              <Text style={styles.loginCardDesc}>Kami akan kirim kode OTP ke nomor Anda</Text>
   273→              <Text style={styles.label}>Nomor WhatsApp</Text>
   274→              <View style={styles.inputContainer}>
   275→                <Ionicons name="call" size={20} color={Colors.textMuted} style={styles.inputIcon} />
   276→                <TextInput style={styles.input} placeholder="08xxx atau 628xxx" value={phone} onChangeText={setPhone} keyboardType="phone-pad" />
   277→              </View>
   278→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleRequestOtp} disabled={loading}>
   279→                {loading ? <ActivityIndicator color={Colors.white} /> : <Text style={styles.loginButtonText}>Kirim OTP</Text>}
   280→              </TouchableOpacity>
   281→            </>
   282→          )}
   283→
   284→          {step === "otp" && (
   285→            <>
   286→              <Text style={styles.loginCardTitle}>Verifikasi OTP</Text>
   287→              <Text style={styles.loginCardDesc}>Masukkan 6 digit kode yang dikirim</Text>
   288→              <TextInput
   289→                style={[styles.input, { textAlign: "center", fontSize: 24, letterSpacing: 8, fontFamily: "monospace" }]}
   290→                placeholder="000000"
   291→                value={otp}
   292→                onChangeText={(t) => setOtp(t.replace(/\D/g, "").slice(0, 6))}
   293→                keyboardType="numeric"
   294→                maxLength={6}
   295→              />
   296→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleVerifyOtp} disabled={loading}>
   297→                {loading ? <ActivityIndicator color={Colors.white} /> : <Text style={styles.loginButtonText}>Verifikasi</Text>}
   298→              </TouchableOpacity>
   299→            </>
   300→          )}
   301→
   302→          {step === "setPin" && (
   303→            <>
   304→              <Text style={styles.loginCardTitle}>Buat PIN Baru</Text>
   305→              <Text style={styles.loginCardDesc}>PIN 6 digit untuk login</Text>
   306→              <Text style={styles.label}>PIN Baru</Text>
   307→              <TextInput
   308→                style={[styles.input, { textAlign: "center", fontSize: 20, letterSpacing: 6, fontFamily: "monospace" }]}
   309→                placeholder="••••••"
   310→                value={newPin}
   311→                onChangeText={(t) => setNewPin(t.replace(/\D/g, "").slice(0, 6))}
   312→                keyboardType="numeric"
   313→                secureTextEntry
   314→                maxLength={6}
   315→              />
   316→              <Text style={styles.label}>Konfirmasi PIN</Text>
   317→              <TextInput
   318→                style={[styles.input, { textAlign: "center", fontSize: 20, letterSpacing: 6, fontFamily: "monospace" }]}
   319→                placeholder="••••••"
   320→                value={confirmPin}
   321→                onChangeText={(t) => setConfirmPin(t.replace(/\D/g, "").slice(0, 6))}
   322→                keyboardType="numeric"
   323→                secureTextEntry
   324→                maxLength={6}
   325→              />
   326→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleSetPin} disabled={loading}>
   327→                {loading ? <ActivityIndicator color={Colors.white} /> : <Text style={styles.loginButtonText}>Simpan PIN</Text>}
   328→              </TouchableOpacity>
   329→            </>
   330→          )}
   331→
   332→          <TouchableOpacity onPress={onBack} style={[styles.forgotPinBtn, { marginTop: 16 }]}>
   333→            <Text style={styles.forgotPinText}>← Kembali ke Login</Text>
   334→          </TouchableOpacity>
   335→        </View>
   336→      </ScrollView>
   337→    </SafeAreaView>
   338→  )
   339→}

   421→function SantriDetailScreen({ santri, onBack, refreshKey }: { santri: SantriData, onBack: () => void, refreshKey?: number }) {
   422→  const [detail, setDetail] = useState<SantriDetail | null>(null)
   423→  const [perkembangan, setPerkembangan] = useState<PerkembanganV2 | null>(null)
   424→  const [loading, setLoading] = useState(true)
   425→  const [error, setError] = useState<string | null>(null)
   426→  const [activeTab, setActiveTab] = useState("profil")
   427→
   428→  const loadDetail = useCallback(async () => {
   429→    try {
   430→      setLoading(true)
   431→      setError(null)
   432→      // V2: fetch both detail (V1) and perkembangan (V2) in parallel
   433→      const [detailRes, perkRes] = await Promise.all([
   434→        api.get(`/api/wali/santri/${santri.id}`),
   435→        api.get(`/api/wali/perkembangan/${santri.id}`).catch(() => null), // best-effort
   436→      ])
   437→      setDetail(detailRes.data)
   438→      setPerkembangan(perkRes?.data || null)
   439→    } catch (e: any) {
   440→      setError(e.message)
   441→    } finally {
   442→      setLoading(false)
   443→    }
   444→  }, [santri.id])
   445→
   446→  useEffect(() => { loadDetail() }, [loadDetail])
   447→
   448→  // Auto-refresh when realtime push arrives (App-level refreshKey changes)
   449→  useEffect(() => {
   450→    if (refreshKey === undefined) return
   451→    loadDetail()
   452→    // eslint-disable-next-line react-hooks/exhaustive-deps
   453→  }, [refreshKey])
   454→
   455→  // Tab groups: 6 tabs that each render multiple sections (scrollable).
   456→  // Reduces visual clutter from 12 individual tabs. Each group lists the
   457→  // section ids (matching the original activeTab checks) that it shows.
   458→  const TAB_GROUPS: Array<{ id: string; label: string; icon: string; sections: string[] }> = [
   459→    { id: "profil", label: "Profil", icon: "person", sections: ["profile", "qr"] },
   460→    { id: "akademik", label: "Akademik", icon: "school", sections: ["absensi", "nilai", "perkembangan", "tugas", "rapor"] },
   461→    { id: "ujian", label: "Ujian", icon: "file-tray-full", sections: ["ujian"] },
   462→    { id: "keuangan", label: "Keuangan", icon: "wallet", sections: ["tagihan"] },
   463→    { id: "notif", label: "Notif", icon: "notifications", sections: ["notifikasi"] },
   464→    { id: "info", label: "Info", icon: "megaphone", sections: ["pengumuman", "chat"] },
   465→  ]
   466→  const activeSections = TAB_GROUPS.find(g => g.id === activeTab)?.sections || []
   467→  const showSection = (id: string) => activeSections.includes(id)
   468→  const tabs = TAB_GROUPS
   469→
   470→  return (
   471→    <SafeAreaView style={styles.screen}>
   472→      <View style={styles.detailHeader}>
   473→        <TouchableOpacity onPress={onBack} style={styles.backButton}>
   474→          <Ionicons name="arrow-back" size={24} color={Colors.white} />
   475→        </TouchableOpacity>
   476→        <Text style={styles.detailHeaderTitle}>{santri.name}</Text>
   477→        <View style={{ width: 24 }} />
   478→      </View>
   479→
   480→      {/* Santri Info Card */}
   481→      <View style={styles.detailCard}>
   482→        <View style={styles.detailAvatar}>
   483→          <Text style={styles.detailAvatarText}>{santri.name[0]}</Text>

   805→function WaliNotifikasiSection({ token }: { token: string | null }) {
   806→  const [list, setList] = useState<any[]>([])
   807→  const [loading, setLoading] = useState(true)
   808→  const [stats, setStats] = useState<any>(null)
   809→  const [error, setError] = useState<string | null>(null)
   810→
   811→  const load = useCallback(async () => {
   812→    if (!token) return
   813→    try {
   814→      setLoading(true)
   815→      setError(null)
   816→      const res = await fetch(`${API_BASE_URL}/api/wali/notifikasi`, {
   817→        headers: { Authorization: `Bearer ${token}` },
   818→      })
   819→      const data = await res.json()
   820→      setList(data.data || [])
   821→      setStats(data.stats)
   822→    } catch (e: any) {
   823→      setError(e.message || "Terjadi kesalahan")
   824→    } finally {
   825→      setLoading(false)
   826→    }
   827→  }, [token])
   828→
   829→  useEffect(() => { load() }, [load])
   830→
   831→  const markRead = async (id: string) => {
   832→    try {
   833→      await fetch(`${API_BASE_URL}/api/wali/notifikasi/${id}`, {
   834→        method: "PUT",
   835→        headers: { Authorization: `Bearer ${token}` },

   839→      Alert.alert("Gagal", e.message)

   841→  }
   842→
   843→  if (loading) return <View style={styles.contentCard}><ActivityIndicator size="large" color={Colors.primary} /></View>

   851→        </TouchableOpacity>

   854→  }

   860→        <View style={[styles.statusBadge, { backgroundColor: Colors.dangerLight, alignSelf: "flex-start", marginBottom: 8 }]}>
   861→          <Text style={[styles.statusBadgeText, { color: Colors.danger }]}>{stats.totalUnread} belum dibaca</Text>
   862→        </View>
   863→      )}

   865→        <Text style={styles.emptyTextSmall}>Tidak ada notifikasi</Text>
   866→      ) : (
   867→        list.map((n: any, i: number) => (
   868→          <View key={i} style={[styles.listRow, !n.isRead && { backgroundColor: Colors.primaryLight + "40" }]}>
