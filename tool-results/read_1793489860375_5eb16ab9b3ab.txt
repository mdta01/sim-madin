    21→const API_BASE_URL = process.env.EXPO_PUBLIC_API_URL || "https://sim-madin.vercel.app"

    23→// For local dev: set EXPO_PUBLIC_SUPABASE_URL and EXPO_PUBLIC_SUPABASE_ANON_KEY in .env
    24→// For EAS Build: configured in eas.json → build.preview.env / build.production.env
    25→const SUPABASE_URL = process.env.EXPO_PUBLIC_SUPABASE_URL || ""
    26→const SUPABASE_ANON_KEY = process.env.EXPO_PUBLIC_SUPABASE_ANON_KEY || ""
    27→
    28→let supabase: any = null
    29→function getSupabase() {
    30→  if (!SUPABASE_URL || !SUPABASE_ANON_KEY) {
    31→    console.warn("[Supabase] env vars not set. Configure EXPO_PUBLIC_SUPABASE_URL and EXPO_PUBLIC_SUPABASE_ANON_KEY in eas.json")
    32→    return null
    33→  }
    34→  if (supabase) return supabase
    35→  supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY, {
    36→    auth: {
    37→      storage: {
    38→        getItem: async (k: string) => { try { return await SecureStore.getItemAsync(k) } catch { return null } },
    39→        setItem: async (k: string, v: string) => { try { await SecureStore.setItemAsync(k, v) } catch {} },
    40→        removeItem: async (k: string) => { try { await SecureStore.deleteItemAsync(k) } catch {} },
    41→      },
    42→      autoRefreshToken: true, persistSession: true,
    43→    },
    44→  })
    45→  return supabase
    46→}

    48→let authToken: string | null = null
    49→
    50→async function apiGet(path: string) {
    51→  const res = await fetch(`${API_BASE_URL}${path}`, {
    52→    headers: authToken ? { Authorization: `Bearer ${authToken}` } : {},
    53→  })
    54→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    55→  return res.json()
    56→}

    75→// Login
    76→function LoginScreen({ onLogin }: { onLogin: () => void }) {
    77→  const [email, setEmail] = useState("")
    78→  const [password, setPassword] = useState("")
    79→  const [loading, setLoading] = useState(false)
    80→
    81→  const handleLogin = async () => {
    82→    setLoading(true)
    83→    try {
    84→      const { data, error } = await getSupabase().auth.signInWithPassword({ email, password })
    85→      if (error) throw error
    86→      authToken = data.session?.access_token
    87→      await SecureStore.setItemAsync("authToken", authToken || "")
    88→      onLogin()
    89→    } catch (e: any) { Alert.alert("Login Gagal", e.message) }
    90→    finally { setLoading(false) }
    91→  }
    92→
    93→  return (
    94→    <SafeAreaView style={styles.loginContainer}>
    95→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
    96→      <ScrollView contentContainerStyle={styles.loginScroll}>
    97→        <View style={styles.loginLogo}>
    98→          <Ionicons name="analytics" size={80} color={Colors.primary} />
    99→          <Text style={styles.loginTitle}>SIM MADIN</Text>
   100→          <Text style={styles.loginSubtitle}>Kepala Madrasah</Text>
   101→        </View>
   102→        <View style={styles.loginCard}>
   103→          <Text style={styles.loginCardTitle}>Masuk sebagai Kepala</Text>
   104→          <Text style={styles.label}>Email</Text>
   105→          <TextInput style={styles.input} value={email} onChangeText={setEmail} keyboardType="email-address" autoCapitalize="none" />
   106→          <Text style={styles.label}>Password</Text>
   107→          <TextInput style={styles.input} value={password} onChangeText={setPassword} secureTextEntry />
   108→          <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleLogin} disabled={loading}>

   644→export default function App() {
   645→  const [isLoggedIn, setIsLoggedIn] = useState(false)
   646→  const [checking, setChecking] = useState(true)
   647→  // Realtime refresh tick — bumped whenever a subscribed table changes.
   648→  // Drives dashboard stats refresh on Santri/Guru/Payment events.
   649→  const [refreshKey, setRefreshKey] = useState(0)
   650→
   651→  useEffect(() => {
   652→    async function check() {
   653→      const token = await SecureStore.getItemAsync("authToken")
   654→      if (token) {
   655→        authToken = token
   656→        const { data } = await getSupabase().auth.getSession()
   657→        if (data.session) { authToken = data.session.access_token; setIsLoggedIn(true) }
   658→      }
   659→      setChecking(false)
   660→    }
   661→    check()
   662→  }, [])
   663→
   664→  // ============================================================================
   665→  // Supabase Realtime subscriptions (OPTIONAL — fails silently if unavailable)
   666→  // - Santri   INSERT/UPDATE : data santri berubah → refresh dashboard stats
   667→  // - Guru     INSERT/UPDATE : data guru berubah  → refresh dashboard stats
   668→  // - Payment  INSERT/UPDATE : pembayaran baru    → refresh keuangan stats
   669→  // ============================================================================
   670→  useEffect(() => {
   671→    if (!isLoggedIn) return
   672→    let cancelled = false
   673→    const unsubs: Array<() => void> = []
   674→    ;(async () => {
   675→      try {
   676→        for (const table of ["Santri", "Guru", "Payment"] as const) {
   677→          try {
   678→            const unsub = await subscribeToTable(table, () => {
   679→              setRefreshKey(k => k + 1)
   680→            })
   681→            if (cancelled) { try { unsub() } catch {} }
   682→            else { unsubs.push(unsub) }
   683→          } catch {}
   684→        }
   685→      } catch(e) { /* realtime optional */ }
   686→    })()
   687→    return () => {
   688→      cancelled = true
   689→      unsubs.forEach(u => { try { u() } catch {} })
   690→    }
   691→  }, [isLoggedIn])
   692→
   693→  if (checking) {
   694→    return <View style={{ flex: 1, justifyContent: "center", alignItems: "center", backgroundColor: Colors.primary }}><ActivityIndicator size="large" color={Colors.white} /></View>
   695→  }
   696→
   697→  return (
   698→    <NavigationContainer>
   699→      <Stack.Navigator screenOptions={{ headerShown: false }}>
   700→        {isLoggedIn ? (
   701→          <>
   702→            <Stack.Screen name="Dashboard">
   703→              {({ navigation }) => <DashboardScreen navigation={navigation} refreshKey={refreshKey} onLogout={async () => { await getSupabase().auth.signOut(); await SecureStore.deleteItemAsync("authToken"); authToken = null; setIsLoggedIn(false) }} />}
   704→            </Stack.Screen>
   705→            <Stack.Screen name="MonitorSantri">
   706→              {({ navigation }) => <MonitorSantriScreen navigation={navigation} refreshKey={refreshKey} />}
   707→            </Stack.Screen>
   708→            <Stack.Screen name="MonitorGuru">
   709→              {({ navigation }) => <MonitorGuruScreen navigation={navigation} refreshKey={refreshKey} />}
