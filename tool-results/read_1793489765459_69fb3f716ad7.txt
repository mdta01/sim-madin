    99→function LoginScreen({ onLogin, onForgotPin }: { onLogin: (token: string, wali: WaliData, santriList: SantriData[]) => void; onForgotPin?: () => void }) {
   100→  const [phone, setPhone] = useState("")
   101→  const [pin, setPin] = useState("")
   102→  const [loading, setLoading] = useState(false)
   103→  const [showPin, setShowPin] = useState(false)
   104→
   105→  const handleLogin = async () => {
   106→    if (!phone || !pin) {
   107→      Alert.alert("Validasi", "Nomor WhatsApp dan PIN wajib diisi")
   108→      return
   109→    }
   110→    if (pin.length !== 6) {
   111→      Alert.alert("Validasi", "PIN harus 6 digit")
   112→      return
   113→    }
   114→
   115→    setLoading(true)
   116→    try {
   117→      const data = await api.post("/api/wali/login", { phone, pin })
   118→      if (data.success) {
   119→        waliToken = data.token
   120→        await saveWaliSession(data.token, data.wali, data.santriList)
   121→        onLogin(data.token, data.wali, data.santriList)
   122→      } else {
   123→        Alert.alert("Login Gagal", data.error || "Terjadi kesalahan")
   124→      }
   125→    } catch (e: any) {
   126→      Alert.alert("Login Gagal", e.message)
   127→    } finally {
   128→      setLoading(false)
   129→    }
   130→  }
   131→
   132→  return (
   133→    <SafeAreaView style={styles.loginContainer}>
   134→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   135→      <ScrollView contentContainerStyle={styles.loginScroll}>
   136→        <View style={styles.loginLogo}>
   137→          <Ionicons name="people-circle" size={80} color={Colors.primary} />
   138→          <Text style={styles.loginTitle}>SIM MADIN</Text>
   139→          <Text style={styles.loginSubtitle}>Wali Santri</Text>
   140→        </View>
   141→
   142→        <View style={styles.loginCard}>
   143→          <Text style={styles.loginCardTitle}>Masuk ke Akun</Text>
   144→          <Text style={styles.loginCardDesc}>Masuk dengan Nomor WhatsApp & PIN</Text>
   145→
   146→          <Text style={styles.label}>Nomor WhatsApp</Text>
   147→          <View style={styles.inputContainer}>
   148→            <Ionicons name="call" size={20} color={Colors.textMuted} style={styles.inputIcon} />
   149→            <TextInput
   150→              style={styles.input}
   151→              placeholder="08xxx atau 628xxx"
   152→              value={phone}
   153→              onChangeText={setPhone}
   154→              keyboardType="phone-pad"
   155→              autoCapitalize="none"
   156→            />
   157→          </View>
   158→
   159→          <Text style={styles.label}>PIN (6 digit)</Text>
   160→          <View style={styles.inputContainer}>
   161→            <Ionicons name="lock-closed" size={20} color={Colors.textMuted} style={styles.inputIcon} />
   162→            <TextInput
   163→              style={[styles.input, { letterSpacing: 8, textAlign: "center" }]}
   164→              placeholder="••••••"
   165→              value={pin}
   166→              onChangeText={(t) => setPin(t.replace(/\D/g, "").slice(0, 6))}
   167→              keyboardType="numeric"
   168→              secureTextEntry={!showPin}

   173→            </TouchableOpacity>

   176→          <TouchableOpacity
   177→            style={[styles.loginButton, loading && { opacity: 0.6 }]}

   181→            {loading ? (
   182→              <ActivityIndicator color={Colors.white} />
   183→            ) : (
   184→              <Text style={styles.loginButtonText}>Masuk</Text>

   186→          </TouchableOpacity>
   187→
   188→          <Text style={styles.demoText}>Demo: 081234511001 / PIN: 258041</Text>
   189→

   191→            <Text style={styles.forgotPinText}>Lupa PIN?</Text>

   194→      </ScrollView>
   195→    </SafeAreaView>

   199→// --- Forgot PIN Screen (3-step OTP flow) ---
   200→function ForgotPinScreen({ onBack }: { onBack: () => void }) {
   201→  const [step, setStep] = useState<"phone" | "otp" | "setPin">("phone")
   202→  const [phone, setPhone] = useState("")
   203→  const [otp, setOtp] = useState("")
   204→  const [newPin, setNewPin] = useState("")
   205→  const [confirmPin, setConfirmPin] = useState("")
   206→  const [loading, setLoading] = useState(false)
   207→  const [verifyToken, setVerifyToken] = useState("")
   208→
   209→  const handleRequestOtp = async () => {
   210→    if (!phone) { Alert.alert("Validasi", "Nomor WhatsApp wajib diisi"); return }
   211→    setLoading(true)
   212→    try {
   213→      const res = await api.post("/api/wali/request-otp", { phone, type: "FORGOT_PIN" })
   214→      Alert.alert("OTP Terkirim", res.devOtp ? `Kode OTP (DEV): ${res.devOtp}` : res.message)
   215→      setStep("otp")
   216→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   217→    finally { setLoading(false) }
   218→  }
   219→
   220→  const handleVerifyOtp = async () => {
   221→    if (!otp || otp.length !== 6) { Alert.alert("Validasi", "OTP harus 6 digit"); return }
   222→    setLoading(true)
   223→    try {
   224→      const res = await api.post("/api/wali/verify-otp", { phone, otpCode: otp })
   225→      if (res.verifyToken) {
   226→        setVerifyToken(res.verifyToken)
   227→        setStep("setPin")
   228→        Alert.alert("Terverifikasi", "Silakan buat PIN baru")
   229→      } else { Alert.alert("Gagal", "Token verifikasi tidak diterima") }
   230→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   231→    finally { setLoading(false) }
   232→  }
   233→
   234→  const handleSetPin = async () => {
   235→    if (!newPin || newPin.length !== 6) { Alert.alert("Validasi", "PIN harus 6 digit"); return }
   236→    if (newPin !== confirmPin) { Alert.alert("Validasi", "PIN tidak cocok"); return }
   237→    setLoading(true)
   238→    try {
   239→      await api.post("/api/wali/set-pin", { verifyToken, pin: newPin })
   240→      Alert.alert("Berhasil", "PIN berhasil diubah. Silakan login dengan PIN baru.", [{ text: "OK", onPress: onBack }])
   241→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   242→    finally { setLoading(false) }
   243→  }
   244→
   245→  return (
   246→    <SafeAreaView style={styles.loginContainer}>
   247→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   248→      <ScrollView contentContainerStyle={styles.loginScroll}>
   249→        <View style={styles.loginLogo}>
   250→          <Ionicons name="key" size={80} color={Colors.white} />
   251→          <Text style={styles.loginTitle}>Lupa PIN</Text>
   252→          <Text style={styles.loginSubtitle}>Reset PIN via OTP</Text>
   253→        </View>
   254→
   255→        <View style={styles.loginCard}>
   256→          {step === "phone" && (
   257→            <>
   258→              <Text style={styles.loginCardTitle}>Masukkan Nomor WhatsApp</Text>
   259→              <Text style={styles.loginCardDesc}>Kami akan kirim kode OTP ke nomor Anda</Text>
   260→              <Text style={styles.label}>Nomor WhatsApp</Text>
   261→              <View style={styles.inputContainer}>
   262→                <Ionicons name="call" size={20} color={Colors.textMuted} style={styles.inputIcon} />
   263→                <TextInput style={styles.input} placeholder="08xxx atau 628xxx" value={phone} onChangeText={setPhone} keyboardType="phone-pad" />
   264→              </View>
   265→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleRequestOtp} disabled={loading}>
   266→                {loading ? <ActivityIndicator color={Colors.white} /> : <Text style={styles.loginButtonText}>Kirim OTP</Text>}
   267→              </TouchableOpacity>
   268→            </>
   269→          )}
   270→
   271→          {step === "otp" && (
   272→            <>
   273→              <Text style={styles.loginCardTitle}>Verifikasi OTP</Text>
   274→              <Text style={styles.loginCardDesc}>Masukkan 6 digit kode yang dikirim</Text>
   275→              <TextInput
   276→                style={[styles.input, { textAlign: "center", fontSize: 24, letterSpacing: 8, fontFamily: "monospace" }]}
   277→                placeholder="000000"
   278→                value={otp}
   279→                onChangeText={(t) => setOtp(t.replace(/\D/g, "").slice(0, 6))}
   280→                keyboardType="numeric"
   281→                maxLength={6}
   282→              />
   283→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleVerifyOtp} disabled={loading}>
   284→                {loading ? <ActivityIndicator color={Colors.white} /> : <Text style={styles.loginButtonText}>Verifikasi</Text>}
   285→              </TouchableOpacity>
   286→            </>
   287→          )}
   288→
   289→          {step === "setPin" && (
   290→            <>
   291→              <Text style={styles.loginCardTitle}>Buat PIN Baru</Text>
   292→              <Text style={styles.loginCardDesc}>PIN 6 digit untuk login</Text>
   293→              <Text style={styles.label}>PIN Baru</Text>
   294→              <TextInput
   295→                style={[styles.input, { textAlign: "center", fontSize: 20, letterSpacing: 6, fontFamily: "monospace" }]}
   296→                placeholder="••••••"
   297→                value={newPin}
   298→                onChangeText={(t) => setNewPin(t.replace(/\D/g, "").slice(0, 6))}
   299→                keyboardType="numeric"
   300→                secureTextEntry
   301→                maxLength={6}
   302→              />
   303→              <Text style={styles.label}>Konfirmasi PIN</Text>
   304→              <TextInput
   305→                style={[styles.input, { textAlign: "center", fontSize: 20, letterSpacing: 6, fontFamily: "monospace" }]}
   306→                placeholder="••••••"
   307→                value={confirmPin}
   308→                onChangeText={(t) => setConfirmPin(t.replace(/\D/g, "").slice(0, 6))}
   309→                keyboardType="numeric"
   310→                secureTextEntry
   311→                maxLength={6}
   312→              />
   313→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleSetPin} disabled={loading}>
   314→                {loading ? <ActivityIndicator color={Colors.white} /> : <Text style={styles.loginButtonText}>Simpan PIN</Text>}
   315→              </TouchableOpacity>
   316→            </>
   317→          )}
   318→
   319→          <TouchableOpacity onPress={onBack} style={[styles.forgotPinBtn, { marginTop: 16 }]}>
   320→            <Text style={styles.forgotPinText}>← Kembali ke Login</Text>
   321→          </TouchableOpacity>
   322→        </View>
   323→      </ScrollView>
   324→    </SafeAreaView>
   325→  )
   326→}

   407→// --- Santri Detail Screen ---
   408→function SantriDetailScreen({ santri, onBack, refreshKey }: { santri: SantriData, onBack: () => void, refreshKey?: number }) {
   409→  const [detail, setDetail] = useState<SantriDetail | null>(null)
   410→  const [perkembangan, setPerkembangan] = useState<PerkembanganV2 | null>(null)
   411→  const [loading, setLoading] = useState(true)
   412→  const [error, setError] = useState<string | null>(null)
   413→  const [activeTab, setActiveTab] = useState("profil")
   414→
   415→  const loadDetail = useCallback(async () => {
   416→    try {
   417→      setLoading(true)
   418→      setError(null)
   419→      // V2: fetch both detail (V1) and perkembangan (V2) in parallel
   420→      const [detailRes, perkRes] = await Promise.all([
   421→        api.get(`/api/wali/santri/${santri.id}`),
   422→        api.get(`/api/wali/perkembangan/${santri.id}`).catch(() => null), // best-effort
   423→      ])
   424→      setDetail(detailRes.data)
   425→      setPerkembangan(perkRes?.data || null)
   426→    } catch (e: any) {
   427→      setError(e.message)
   428→    } finally {
   429→      setLoading(false)
   430→    }
   431→  }, [santri.id])
   432→
   433→  useEffect(() => { loadDetail() }, [loadDetail])
   434→
   435→  // Auto-refresh when realtime push arrives (App-level refreshKey changes)
   436→  useEffect(() => {
   437→    if (refreshKey === undefined) return
   438→    loadDetail()
   439→    // eslint-disable-next-line react-hooks/exhaustive-deps
   440→  }, [refreshKey])
   441→
   442→  // Tab groups: 6 tabs that each render multiple sections (scrollable).
   443→  // Reduces visual clutter from 12 individual tabs. Each group lists the
   444→  // section ids (matching the original activeTab checks) that it shows.
   445→  const TAB_GROUPS: Array<{ id: string; label: string; icon: string; sections: string[] }> = [
   446→    { id: "profil", label: "Profil", icon: "person", sections: ["profile", "qr"] },
   447→    { id: "akademik", label: "Akademik", icon: "school", sections: ["absensi", "nilai", "perkembangan", "tugas", "rapor"] },
   448→    { id: "ujian", label: "Ujian", icon: "file-tray-full", sections: ["ujian"] },
   449→    { id: "keuangan", label: "Keuangan", icon: "wallet", sections: ["tagihan"] },
   450→    { id: "notif", label: "Notif", icon: "notifications", sections: ["notifikasi"] },
   451→    { id: "info", label: "Info", icon: "megaphone", sections: ["pengumuman", "chat"] },
   452→  ]
   453→  const activeSections = TAB_GROUPS.find(g => g.id === activeTab)?.sections || []
   454→  const showSection = (id: string) => activeSections.includes(id)
   455→  const tabs = TAB_GROUPS
   456→
   457→  return (
   458→    <SafeAreaView style={styles.screen}>
   459→      <View style={styles.detailHeader}>
   460→        <TouchableOpacity onPress={onBack} style={styles.backButton}>
   461→          <Ionicons name="arrow-back" size={24} color={Colors.white} />
   462→        </TouchableOpacity>
   463→        <Text style={styles.detailHeaderTitle}>{santri.name}</Text>
   464→        <View style={{ width: 24 }} />
   465→      </View>
   466→
   467→      {/* Santri Info Card */}
   468→      <View style={styles.detailCard}>
   469→        <View style={styles.detailAvatar}>
   470→          <Text style={styles.detailAvatarText}>{santri.name[0]}</Text>
   471→        </View>
   472→        <View style={styles.detailInfo}>
   473→          <Text style={styles.detailName}>{santri.name}</Text>
   474→          <Text style={styles.detailNis}>NIS: {santri.nis}</Text>
   475→          <Text style={styles.detailKelas}>{santri.kelas?.name || "Tanpa Kelas"}</Text>
   476→          {detail?.activePeriod && (
   477→            <Text style={styles.detailPeriod}>{detail.activePeriod.tahunAjaran} — Semester {detail.activePeriod.semester}</Text>
   478→          )}
   479→        </View>
   480→      </View>
   481→
   482→      {/* Quick Stats */}
   483→      {detail && (
   484→        <View style={styles.statsRow}>

   790→function WaliNotifikasiSection({ token }: { token: string | null }) {

   797→    if (!token) return

   799→      setLoading(true)

   925→function WaliChatRoom({ room, token, onBack }: { room: any, token: string | null, onBack: () => void }) {
   926→  const [messages, setMessages] = useState<any[]>([])
   927→  const [input, setInput] = useState("")
   928→  const [loading, setLoading] = useState(true)
   929→
   930→  const loadMessages = useCallback(async () => {
   931→    if (!token) return
   932→    try {
   933→      const res = await fetch(`${API_BASE_URL}/api/wali/chat/${room.id}/messages`, {
   934→        headers: { Authorization: `Bearer ${token}` },
   935→      })
   936→      const data = await res.json()
   937→      setMessages(data.data || [])
   938→    } catch (e: any) {
   939→      console.error("msg load error:", e.message)
   940→    } finally {
   941→      setLoading(false)
   942→    }
   943→  }, [token, room.id])
   944→
   945→  useEffect(() => { loadMessages() }, [loadMessages])
   946→
   947→  const send = async () => {
   948→    if (!input.trim() || !token) return
   949→    try {
   950→      await fetch(`${API_BASE_URL}/api/wali/chat/${room.id}/messages`, {
   951→        method: "POST",
   952→        headers: { "Content-Type": "application/json", Authorization: `Bearer ${token}` },
   953→        body: JSON.stringify({ message: input }),
   954→      })
   955→      setInput("")
   956→      loadMessages()
   957→    } catch (e: any) {
   958→      Alert.alert("Gagal", e.message)
   959→    }
   960→  }
   961→
   962→  if (loading) return <View style={styles.contentCard}><ActivityIndicator size="large" color={Colors.primary} /></View>
