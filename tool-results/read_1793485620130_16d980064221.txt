    99→function LoginScreen({ onLogin, onForgotPin }: { onLogin: (token: string, wali: WaliData, santriList: SantriData[]) => void; onForgotPin?: () => void }) {
   100→  const [phone, setPhone] = useState("081234511001")
   101→  const [pin, setPin] = useState("")
   102→  const [loading, setLoading] = useState(false)
   103→  const [showPin, setShowPin] = useState(false)
   104→
   105→  const handleLogin = async () => {
   106→    if (!phone || !pin) {
   107→      Alert.alert("Validasi", "Nomor WhatsApp dan PIN wajib diisi")
   108→      return
   109→    }
   110→    if (pin.length !== 6) {
   111→      Alert.alert("Validasi", "PIN harus 6 digit")
   112→      return
   113→    }
   114→
   115→    setLoading(true)
   116→    try {
   117→      const data = await api.post("/api/wali/login", { phone, pin })
   118→      if (data.success) {
   119→        waliToken = data.token
   120→        await saveWaliSession(data.token, data.wali, data.santriList)
   121→        onLogin(data.token, data.wali, data.santriList)
   122→      } else {
   123→        Alert.alert("Login Gagal", data.error || "Terjadi kesalahan")
   124→      }
   125→    } catch (e: any) {
   126→      Alert.alert("Login Gagal", e.message)
   127→    } finally {
   128→      setLoading(false)
   129→    }
   130→  }
   131→
   132→  return (
   133→    <SafeAreaView style={styles.loginContainer}>
   134→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   135→      <ScrollView contentContainerStyle={styles.loginScroll}>
   136→        <View style={styles.loginLogo}>
   137→          <Ionicons name="people-circle" size={80} color={Colors.primary} />
   138→          <Text style={styles.loginTitle}>SIM MADIN</Text>
   139→          <Text style={styles.loginSubtitle}>Wali Santri</Text>
   140→        </View>
   141→
   142→        <View style={styles.loginCard}>
   143→          <Text style={styles.loginCardTitle}>Masuk ke Akun</Text>
   144→          <Text style={styles.loginCardDesc}>Masuk dengan Nomor WhatsApp & PIN</Text>
   145→
   146→          <Text style={styles.label}>Nomor WhatsApp</Text>
   147→          <View style={styles.inputContainer}>
   148→            <Ionicons name="call" size={20} color={Colors.textMuted} style={styles.inputIcon} />
   149→            <TextInput
   150→              style={styles.input}
   151→              placeholder="08xxx atau 628xxx"
   152→              value={phone}

   164→              placeholder="••••••"

   199→// --- Forgot PIN Screen (3-step OTP flow) ---
   200→function ForgotPinScreen({ onBack }: { onBack: () => void }) {
   201→  const [step, setStep] = useState<"phone" | "otp" | "setPin">("phone")
   202→  const [phone, setPhone] = useState("")
   203→  const [otp, setOtp] = useState("")
   204→  const [newPin, setNewPin] = useState("")
   205→  const [confirmPin, setConfirmPin] = useState("")
   206→  const [loading, setLoading] = useState(false)
   207→  const [verifyToken, setVerifyToken] = useState("")
   208→
   209→  const handleRequestOtp = async () => {
   210→    if (!phone) { Alert.alert("Validasi", "Nomor WhatsApp wajib diisi"); return }
   211→    setLoading(true)
   212→    try {
   213→      const res = await api.post("/api/wali/request-otp", { phone, type: "FORGOT_PIN" })
   214→      Alert.alert("OTP Terkirim", res.devOtp ? `Kode OTP (DEV): ${res.devOtp}` : res.message)
   215→      setStep("otp")
   216→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   217→    finally { setLoading(false) }
   218→  }
   219→
   220→  const handleVerifyOtp = async () => {
   221→    if (!otp || otp.length !== 6) { Alert.alert("Validasi", "OTP harus 6 digit"); return }
   222→    setLoading(true)
   223→    try {
   224→      const res = await api.post("/api/wali/verify-otp", { phone, otpCode: otp })
   225→      if (res.verifyToken) {
   226→        setVerifyToken(res.verifyToken)
   227→        setStep("setPin")
   228→        Alert.alert("Terverifikasi", "Silakan buat PIN baru")
   229→      } else { Alert.alert("Gagal", "Token verifikasi tidak diterima") }
   230→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   231→    finally { setLoading(false) }
   232→  }
   233→
   234→  const handleSetPin = async () => {
   235→    if (!newPin || newPin.length !== 6) { Alert.alert("Validasi", "PIN harus 6 digit"); return }
   236→    if (newPin !== confirmPin) { Alert.alert("Validasi", "PIN tidak cocok"); return }
   237→    setLoading(true)
   238→    try {
   239→      await api.post("/api/wali/set-pin", { verifyToken, pin: newPin })
   240→      Alert.alert("Berhasil", "PIN berhasil diubah. Silakan login dengan PIN baru.", [{ text: "OK", onPress: onBack }])
   241→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   242→    finally { setLoading(false) }
   243→  }
   244→
   245→  return (
   246→    <SafeAreaView style={styles.loginContainer}>
   247→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   248→      <ScrollView contentContainerStyle={styles.loginScroll}>
   249→        <View style={styles.loginLogo}>
   250→          <Ionicons name="key" size={80} color={Colors.white} />
   251→          <Text style={styles.loginTitle}>Lupa PIN</Text>
   252→          <Text style={styles.loginSubtitle}>Reset PIN via OTP</Text>
   253→        </View>
   254→
   255→        <View style={styles.loginCard}>
   256→          {step === "phone" && (
   257→            <>
   258→              <Text style={styles.loginCardTitle}>Masukkan Nomor WhatsApp</Text>
   259→              <Text style={styles.loginCardDesc}>Kami akan kirim kode OTP ke nomor Anda</Text>
   260→              <Text style={styles.label}>Nomor WhatsApp</Text>
   261→              <View style={styles.inputContainer}>
   262→                <Ionicons name="call" size={20} color={Colors.textMuted} style={styles.inputIcon} />
   263→                <TextInput style={styles.input} placeholder="08xxx atau 628xxx" value={phone} onChangeText={setPhone} keyboardType="phone-pad" />
   264→              </View>
   265→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleRequestOtp} disabled={loading}>
   266→                {loading ? <ActivityIndicator color={Colors.white} /> : <Text style={styles.loginButtonText}>Kirim OTP</Text>}
   267→              </TouchableOpacity>
   268→            </>
   269→          )}
   270→
   271→          {step === "otp" && (
   272→            <>
   273→              <Text style={styles.loginCardTitle}>Verifikasi OTP</Text>
   274→              <Text style={styles.loginCardDesc}>Masukkan 6 digit kode yang dikirim</Text>
   275→              <TextInput
   276→                style={[styles.input, { textAlign: "center", fontSize: 24, letterSpacing: 8, fontFamily: "monospace" }]}
   277→                placeholder="000000"
   278→                value={otp}
   279→                onChangeText={(t) => setOtp(t.replace(/\D/g, "").slice(0, 6))}
   280→                keyboardType="numeric"
   281→                maxLength={6}
   282→              />
   283→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleVerifyOtp} disabled={loading}>
   284→                {loading ? <ActivityIndicator color={Colors.white} /> : <Text style={styles.loginButtonText}>Verifikasi</Text>}
   285→              </TouchableOpacity>
   286→            </>
   287→          )}
   288→
   289→          {step === "setPin" && (
   290→            <>
   291→              <Text style={styles.loginCardTitle}>Buat PIN Baru</Text>
   292→              <Text style={styles.loginCardDesc}>PIN 6 digit untuk login</Text>
   293→              <Text style={styles.label}>PIN Baru</Text>
   294→              <TextInput
   295→                style={[styles.input, { textAlign: "center", fontSize: 20, letterSpacing: 6, fontFamily: "monospace" }]}
   296→                placeholder="••••••"
   297→                value={newPin}
   298→                onChangeText={(t) => setNewPin(t.replace(/\D/g, "").slice(0, 6))}
   299→                keyboardType="numeric"
   300→                secureTextEntry
   301→                maxLength={6}
   302→              />
   303→              <Text style={styles.label}>Konfirmasi PIN</Text>
   304→              <TextInput
   305→                style={[styles.input, { textAlign: "center", fontSize: 20, letterSpacing: 6, fontFamily: "monospace" }]}
   306→                placeholder="••••••"
   307→                value={confirmPin}
   308→                onChangeText={(t) => setConfirmPin(t.replace(/\D/g, "").slice(0, 6))}
   309→                keyboardType="numeric"
   310→                secureTextEntry
   311→                maxLength={6}
   312→              />
   313→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleSetPin} disabled={loading}>
   314→                {loading ? <ActivityIndicator color={Colors.white} /> : <Text style={styles.loginButtonText}>Simpan PIN</Text>}
   315→              </TouchableOpacity>
   316→            </>
   317→          )}
   318→
   319→          <TouchableOpacity onPress={onBack} style={[styles.forgotPinBtn, { marginTop: 16 }]}>
   320→            <Text style={styles.forgotPinText}>← Kembali ke Login</Text>
   321→          </TouchableOpacity>
   322→        </View>
   323→      </ScrollView>
   324→    </SafeAreaView>
   325→  )
   326→}

   766→function WaliNotifikasiSection({ token }: { token: string | null }) {
   767→  const [list, setList] = useState<any[]>([])
   768→  const [loading, setLoading] = useState(true)
   769→  const [stats, setStats] = useState<any>(null)
   770→
   771→  const load = useCallback(async () => {
   772→    if (!token) return
   773→    try {
   774→      setLoading(true)
   775→      const res = await fetch(`${API_BASE_URL}/api/wali/notifikasi`, {
   776→        headers: { Authorization: `Bearer ${token}` },
   777→      })
   778→      const data = await res.json()
   779→      setList(data.data || [])
   780→      setStats(data.stats)
   781→    } catch (e: any) {
   782→      console.error("notif load error:", e.message)
   783→    } finally {
   784→      setLoading(false)
   785→    }
   786→  }, [token])
   787→
   788→  useEffect(() => { load() }, [load])
   789→
   790→  const markRead = async (id: string) => {
   791→    try {
   792→      await fetch(`${API_BASE_URL}/api/wali/notifikasi/${id}`, {
   793→        method: "PUT",
   794→        headers: { Authorization: `Bearer ${token}` },
   795→      })
   796→      load()
   797→    } catch (e: any) {
   798→      Alert.alert("Gagal", e.message)
   799→    }
   800→  }
   801→
   802→  if (loading) return <View style={styles.contentCard}><ActivityIndicator size="large" color={Colors.primary} /></View>
   803→
   804→  return (
   805→    <View style={styles.contentCard}>
   806→      <Text style={styles.contentTitle}>Notifikasi</Text>
   807→      {stats && stats.totalUnread > 0 && (
   808→        <View style={[styles.statusBadge, { backgroundColor: Colors.dangerLight, alignSelf: "flex-start", marginBottom: 8 }]}>
   809→          <Text style={[styles.statusBadgeText, { color: Colors.danger }]}>{stats.totalUnread} belum dibaca</Text>

   811→      )}
   812→      {list.length === 0 ? (
   813→        <Text style={styles.emptyTextSmall}>Tidak ada notifikasi</Text>
   814→      ) : (
   815→        list.map((n: any, i: number) => (
   816→          <View key={i} style={[styles.listRow, !n.isRead && { backgroundColor: Colors.primaryLight + "40" }]}>

   818→              <Text style={[styles.tagihanName, !n.isRead && { fontWeight: "bold" }]}>{n.title}</Text>

   888→function WaliChatRoom({ room, token, onBack }: { room: any, token: string | null, onBack: () => void }) {
   889→  const [messages, setMessages] = useState<any[]>([])
   890→  const [input, setInput] = useState("")
   891→  const [loading, setLoading] = useState(true)
   892→
   893→  const loadMessages = useCallback(async () => {
   894→    if (!token) return
   895→    try {
   896→      const res = await fetch(`${API_BASE_URL}/api/wali/chat/${room.id}/messages`, {
   897→        headers: { Authorization: `Bearer ${token}` },
   898→      })
   899→      const data = await res.json()
   900→      setMessages(data.data || [])
   901→    } catch (e: any) {
   902→      console.error("msg load error:", e.message)
   903→    } finally {
   904→      setLoading(false)
   905→    }
   906→  }, [token, room.id])
   907→
   908→  useEffect(() => { loadMessages() }, [loadMessages])
   909→
   910→  const send = async () => {
   911→    if (!input.trim() || !token) return
   912→    try {
   913→      await fetch(`${API_BASE_URL}/api/wali/chat/${room.id}/messages`, {
   914→        method: "POST",
   915→        headers: { "Content-Type": "application/json", Authorization: `Bearer ${token}` },
   916→        body: JSON.stringify({ message: input }),
   917→      })
   918→      setInput("")
   919→      loadMessages()
   920→    } catch (e: any) {
   921→      Alert.alert("Gagal", e.message)
   922→    }
   923→  }
   924→
   925→  if (loading) return <View style={styles.contentCard}><ActivityIndicator size="large" color={Colors.primary} /></View>
   926→
   927→  return (
   928→    <View style={styles.contentCard}>
   929→      <TouchableOpacity onPress={onBack} style={{ flexDirection: "row", alignItems: "center", marginBottom: 8 }}>
   930→        <Ionicons name="arrow-back" size={20} color={Colors.primary} />

   932→      </TouchableOpacity>
   933→      <Text style={styles.contentTitle}>{room.type === "WALI_WALI_KELAS" ? "Chat Wali Kelas" : "Chat Madrasah"}</Text>
   934→      <View style={{ maxHeight: 300, marginBottom: 8 }}>

   937→            <View key={i} style={{
   938→              alignSelf: m.senderType === "WALI" ? "flex-end" : "flex-start",
   939→              backgroundColor: m.senderType === "WALI" ? Colors.primaryLight : Colors.border,
   940→              padding: 8, borderRadius: 8, marginBottom: 4, maxWidth: "80%",
   941→            }}>
   942→              <Text style={{ fontSize: 13, color: m.senderType === "WALI" ? Colors.primaryDark : Colors.text }}>{m.message}</Text>

   944→            </View>
   945→          ))}
   946→        </ScrollView>
   947→      </View>
   948→      <View style={{ flexDirection: "row", gap: 8 }}>
   949→        <TextInput
   950→          style={[styles.textInput, { flex: 1 }]}
   951→          value={input}
   952→          onChangeText={setInput}
   953→          placeholder="Tulis pesan..."
   954→        />
   955→        <TouchableOpacity onPress={send} style={[styles.submitButton, { paddingHorizontal: 16 }]}>

  1527→function UjianTakeScreen({ ujian, santriId, onClose }: { ujian: any, santriId: string, onClose: () => void }) {

  1545→        setRemainingSeconds(res.data.remainingSeconds || ujian.durationMnt * 60)

  1547→        const init: Record<string, { answer: string; optionOrder?: number[] }> = {}

  1549→          init[s.ujianSoalId] = {
  1550→            answer: s.existingAnswer || "",
  1551→            optionOrder: s.optionOrder,

  1559→        setLoading(false)

  1592→      "Konfirmasi Submit",

  1652→  if (!detail) return null

  1655→  const answeredCount = Object.values(answers).filter((a) => a.answer).length
