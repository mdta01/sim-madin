    24→// For EAS Build: configured in eas.json → build.preview.env / build.production.env
    25→const SUPABASE_URL = process.env.EXPO_PUBLIC_SUPABASE_URL || ""

    28→let supabase: any = null
    29→function getSupabase() {
    30→  if (!SUPABASE_URL || !SUPABASE_ANON_KEY) {
    31→    console.warn("[Supabase] env vars not set. Configure EXPO_PUBLIC_SUPABASE_URL and EXPO_PUBLIC_SUPABASE_ANON_KEY in eas.json")
    32→    return null
    33→  }
    34→  if (supabase) return supabase
    35→  supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY, {
    36→    auth: {
    37→      storage: {
    38→        getItem: async (k: string) => { try { return await SecureStore.getItemAsync(k) } catch { return null } },
    39→        setItem: async (k: string, v: string) => { try { await SecureStore.setItemAsync(k, v) } catch {} },
    40→        removeItem: async (k: string) => { try { await SecureStore.deleteItemAsync(k) } catch {} },
    41→      },
    42→      autoRefreshToken: true, persistSession: true,
    43→    },
    44→  })
    45→  return supabase
    46→}

    48→let authToken: string | null = null
    49→
    50→async function apiGet(path: string) {
    51→  const res = await fetch(`${API_BASE_URL}${path}`, { headers: authToken ? { Authorization: `Bearer ${authToken}` } : {} })
    52→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    53→  return res.json()
    54→}
    55→
    56→async function apiPost(path: string, body: any) {
    57→  const res = await fetch(`${API_BASE_URL}${path}`, {
    58→    method: "POST",
    59→    headers: { "Content-Type": "application/json", ...(authToken ? { Authorization: `Bearer ${authToken}` } : {}) },
    60→    body: JSON.stringify(body),
    61→  })
    62→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    63→  return res.json()
    64→}
    65→
    66→async function apiPut(path: string, body: any) {
    67→  const res = await fetch(`${API_BASE_URL}${path}`, {
    68→    method: "PUT",
    69→    headers: { "Content-Type": "application/json", ...(authToken ? { Authorization: `Bearer ${authToken}` } : {}) },
    70→    body: JSON.stringify(body),
    71→  })
    72→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    73→  return res.json()
    74→}

   251→function PembayaranInputScreen({ navigation }: { navigation: any }) {
   252→  const [santriList, setSantriList] = useState<any[]>([])
   253→  const [tagihanList, setTagihanList] = useState<any[]>([])
   254→  const [selectedSantri, setSelectedSantri] = useState("")
   255→  const [selectedTagihan, setSelectedTagihan] = useState("")
   256→  const [amount, setAmount] = useState("")
   257→  const [method, setMethod] = useState("TUNAI")
   258→  const [loading, setLoading] = useState(false)
   259→  const [saving, setSaving] = useState(false)
   260→  const [searchText, setSearchText] = useState("")
   261→  const [error, setError] = useState<string | null>(null)
   262→
   263→  const loadSantri = useCallback(async () => {
   264→    try {
   265→      setLoading(true)
   266→      setError(null)
   267→      const res = await apiGet("/api/admin/santri?status=AKTIF")
   268→      setSantriList(res.data || [])
   269→    } catch (e: any) {
   270→      setError(e.message || "Terjadi kesalahan")
   271→    } finally {
   272→      setLoading(false)
   273→    }
   274→  }, [])
   275→
   276→  const loadTagihan = useCallback(async () => {
   277→    if (!selectedSantri) { setTagihanList([]); return }
   278→    try {
   279→      setError(null)
   280→      const res = await apiGet(`/api/admin/tagihan?santriId=${selectedSantri}`)
   281→      const unpaid = (res.data || []).filter((t: any) => t.status !== "LUNAS")
   282→      setTagihanList(unpaid)
   283→    } catch (e: any) {
   284→      setError(e.message || "Terjadi kesalahan")
   285→    }
   286→  }, [selectedSantri])
   287→
   288→  useEffect(() => { loadSantri() }, [loadSantri])
   289→  useEffect(() => { loadTagihan() }, [loadTagihan])
   290→
   291→  const selectTagihan = (tagihanId: string, tagihanAmount: number) => {
   292→    setSelectedTagihan(tagihanId)
   293→    setAmount(String(tagihanAmount - (tagihanList.find(t => t.id === tagihanId)?.paidAmount || 0)))
   294→  }
   295→
   296→  const handleSave = async () => {
   297→    if (!selectedSantri || !amount || Number(amount) <= 0) {
   298→      Alert.alert("Validasi", "Pilih santri dan isi nominal")
   299→      return
   300→    }
   301→    setSaving(true)
   302→    try {
   303→      await apiPost("/api/admin/pembayaran", {
   304→        santriId: selectedSantri,
   305→        tagihanId: selectedTagihan || undefined,
   306→        amount: Number(amount),
   307→        paymentMethod: method,
   308→      })
   309→      Alert.alert("Berhasil", `Pembayaran ${formatRupiah(Number(amount))} tercatat`)
   310→      navigation.goBack()
   311→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   312→    finally { setSaving(false) }
   313→  }
   314→
   315→  const filteredSantri = santriList.filter(s =>
   316→    !searchText || s.name.toLowerCase().includes(searchText.toLowerCase()) || s.nis.includes(searchText)
   317→  )
   318→
   319→  return (
   320→    <SafeAreaView style={styles.screen}>
   321→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   322→      <View style={styles.header}>
   323→        <TouchableOpacity onPress={() => navigation.goBack()} style={styles.backButton}>
   324→          <Ionicons name="arrow-back" size={24} color={Colors.white} />
   325→        </TouchableOpacity>
   326→        <Text style={styles.headerTitle}>Input Pembayaran</Text>
   327→        <View style={{ width: 24 }} />
   328→      </View>
   329→
   330→      {error ? (
   331→        <View style={{ flex: 1, justifyContent: "center", alignItems: "center", padding: 20 }}>
   332→          <Text style={{ color: Colors.danger, textAlign: "center", marginBottom: 16 }}>{error}</Text>
   333→          <TouchableOpacity onPress={() => { setError(null); loadSantri() }} style={{ backgroundColor: Colors.primary, paddingHorizontal: 24, paddingVertical: 12, borderRadius: 8 }}>
   334→            <Text style={{ color: Colors.white, fontWeight: "600" }}>Coba Lagi</Text>
   335→          </TouchableOpacity>
   336→        </View>
   337→      ) : (
   338→      <ScrollView style={styles.content}>
   339→        {/* Search Santri */}
   340→        <Text style={styles.label}>Cari Santri</Text>
   341→        <View style={styles.searchContainer}>
   342→          <Ionicons name="search" size={20} color={Colors.textMuted} style={{ marginRight: 8 }} />
   343→          <TextInput
   344→            style={styles.searchInput}
   345→            placeholder="Nama atau NIS..."
   346→            value={searchText}
   347→            onChangeText={setSearchText}
   348→          />
   349→        </View>
   350→
   351→        {/* Santri List */}
   352→        {loading ? (
   353→          <ActivityIndicator size="large" color={Colors.primary} style={{ marginTop: 20 }} />
   354→        ) : (
   355→          <View>
   356→            {filteredSantri.slice(0, 5).map(s => (
   357→              <TouchableOpacity
   358→                key={s.id}
   359→                style={[styles.santriOption, selectedSantri === s.id && styles.santriOptionActive]}
   360→                onPress={() => { setSelectedSantri(s.id); setSelectedTagihan(""); setAmount("") }}
   361→              >
   362→                <View style={{ flex: 1 }}>
   363→                  <Text style={styles.santriOptionName}>{s.name}</Text>

   371→            )}
   372→          </View>
   373→        )}
   374→
   375→        {/* Tagihan List (if santri selected) */}

   385→                <View style={{ flex: 1 }}>
   386→                  <Text style={styles.tagihanOptionName}>{t.jenisTagihan?.name}</Text>
   387→                  <Text style={styles.tagihanOptionDesc}>
   388→                    Total: {formatRupiah(t.amount)} • Sisa: {formatRupiah(t.amount - t.paidAmount)}
   389→                  </Text>
   390→                </View>
   391→                {selectedTagihan === t.id && <Ionicons name="checkmark-circle" size={24} color={Colors.primary} />}
   392→              </TouchableOpacity>
   393→            ))}
   394→          </View>
   395→        )}
   396→
   397→        {/* Amount */}
   398→        <Text style={styles.label}>Nominal Pembayaran</Text>
   399→        <View style={styles.amountContainer}>
   400→          <Text style={styles.amountPrefix}>Rp</Text>

   403→            value={amount}
   404→            onChangeText={(v) => setAmount(v.replace(/\D/g, ""))}
   405→            keyboardType="numeric"
   406→            placeholder="0"
   407→          />
   408→        </View>
   409→
   410→        {/* Method */}
   411→        <Text style={styles.label}>Metode Pembayaran</Text>
   412→        <View style={styles.methodRow}>
   413→          {PAYMENT_METHODS.map(m => (

   415→              key={m.value}
   416→              style={[styles.methodBtn, method === m.value && styles.methodBtnActive]}

   419→              <Text style={[styles.methodBtnText, method === m.value && styles.methodBtnTextActive]}>{m.label}</Text>

   421→          ))}
   422→        </View>
   423→
   424→        {/* Save */}
   425→        <TouchableOpacity
   426→          style={[styles.saveButton, saving && { opacity: 0.6 }]}
   427→          onPress={handleSave}
   428→          disabled={saving}
   429→        >
   430→          {saving ? <ActivityIndicator color={Colors.white} /> : (
   431→            <>
   432→              <Ionicons name="save" size={20} color={Colors.white} />
   433→              <Text style={styles.saveButtonText}>Catat Pembayaran</Text>
   434→            </>
   435→          )}
   436→        </TouchableOpacity>
   437→
   438→        <View style={{ height: 40 }} />
   439→      </ScrollView>
   440→      )}
   441→    </SafeAreaView>
   442→  )
