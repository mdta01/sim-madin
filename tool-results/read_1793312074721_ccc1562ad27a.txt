    42→    42→async function apiGet(path: string) {
    43→    43→  const res = await fetch(`${API_BASE_URL}${path}`, { headers: authToken ? { Authorization: `Bearer ${authToken}` } : {} })

    45→    45→  return res.json()

    48→    48→async function apiPost(path: string, body: any) {
    49→    49→  const res = await fetch(`${API_BASE_URL}${path}`, {
    50→    50→    method: "POST",
    51→    51→    headers: { "Content-Type": "application/json", ...(authToken ? { Authorization: `Bearer ${authToken}` } : {}) },
    52→    52→    body: JSON.stringify(body),
    53→    53→  })
    54→    54→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    55→    55→  return res.json()
    56→    56→}
