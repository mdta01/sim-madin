    26→// For EAS Build: configured in eas.json → build.preview.env / build.production.env
    27→const SUPABASE_URL = process.env.EXPO_PUBLIC_SUPABASE_URL || ""
    28→const SUPABASE_ANON_KEY = process.env.EXPO_PUBLIC_SUPABASE_ANON_KEY || ""

    30→let supabase: any = null
    31→function getSupabase() {
    32→  if (!SUPABASE_URL || !SUPABASE_ANON_KEY) {
    33→    console.warn("[Supabase] env vars not set. Configure EXPO_PUBLIC_SUPABASE_URL and EXPO_PUBLIC_SUPABASE_ANON_KEY in eas.json")
    34→    return null
    35→  }
    36→  if (supabase) return supabase
    37→  supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY, {
    38→    auth: {
    39→      storage: {
    40→        getItem: async (k: string) => { try { return await SecureStore.getItemAsync(k) } catch { return null } },
    41→        setItem: async (k: string, v: string) => { try { await SecureStore.setItemAsync(k, v) } catch {} },
    42→        removeItem: async (k: string) => { try { await SecureStore.deleteItemAsync(k) } catch {} },
    43→      },
    44→      autoRefreshToken: true, persistSession: true,
    45→    },
    46→  })
    47→  return supabase
    48→}

    50→let authToken: string | null = null
    51→
    52→async function apiGet(path: string) {
    53→  const res = await fetch(`${API_BASE_URL}${path}`, { headers: authToken ? { Authorization: `Bearer ${authToken}` } : {} })
    54→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    55→  return res.json()
    56→}
    57→
    58→async function apiPost(path: string, body: any) {
    59→  const res = await fetch(`${API_BASE_URL}${path}`, {
    60→    method: "POST",
    61→    headers: { "Content-Type": "application/json", ...(authToken ? { Authorization: `Bearer ${authToken}` } : {}) },
    62→    body: JSON.stringify(body),
    63→  })
    64→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    65→  return res.json()
    66→}
    67→
    68→async function apiPut(path: string, body: any) {
    69→  const res = await fetch(`${API_BASE_URL}${path}`, {
    70→    method: "PUT",
    71→    headers: { "Content-Type": "application/json", ...(authToken ? { Authorization: `Bearer ${authToken}` } : {}) },
    72→    body: JSON.stringify(body),
    73→  })
    74→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    75→  return res.json()
    76→}

    91→function LoginScreen({ onLogin }: { onLogin: () => void }) {

    97→    setLoading(true)

    99→      const { data, error } = await getSupabase().auth.signInWithPassword({ email, password })

   101→      authToken = data.session?.access_token
   102→      await SecureStore.setItemAsync("authToken", authToken || "")

   255→function PembayaranInputScreen({ navigation }: { navigation: any }) {
   256→  const [santriList, setSantriList] = useState<any[]>([])
   257→  const [tagihanList, setTagihanList] = useState<any[]>([])
   258→  const [selectedSantri, setSelectedSantri] = useState("")
   259→  const [selectedTagihan, setSelectedTagihan] = useState("")
   260→  const [amount, setAmount] = useState("")
   261→  const [method, setMethod] = useState("TUNAI")
   262→  const [loading, setLoading] = useState(false)
   263→  const [saving, setSaving] = useState(false)
   264→  const [searchText, setSearchText] = useState("")
   265→  const [error, setError] = useState<string | null>(null)
   266→
   267→  const loadSantri = useCallback(async () => {
   268→    try {
   269→      setLoading(true)
   270→      setError(null)
   271→      const res = await apiGet("/api/admin/santri?status=AKTIF")
   272→      setSantriList(res.data || [])
   273→    } catch (e: any) {
   274→      setError(e.message || "Terjadi kesalahan")
   275→    } finally {
   276→      setLoading(false)
   277→    }
   278→  }, [])
   279→
   280→  const loadTagihan = useCallback(async () => {
   281→    if (!selectedSantri) { setTagihanList([]); return }
   282→    try {
   283→      setError(null)
   284→      const res = await apiGet(`/api/admin/tagihan?santriId=${selectedSantri}`)
   285→      const unpaid = (res.data || []).filter((t: any) => t.status !== "LUNAS")
   286→      setTagihanList(unpaid)
   287→    } catch (e: any) {
   288→      setError(e.message || "Terjadi kesalahan")
   289→    }
   290→  }, [selectedSantri])
   291→
   292→  useEffect(() => { loadSantri() }, [loadSantri])
   293→  useEffect(() => { loadTagihan() }, [loadTagihan])
   294→
   295→  const selectTagihan = (tagihanId: string, tagihanAmount: number) => {
   296→    setSelectedTagihan(tagihanId)
   297→    setAmount(String(tagihanAmount - (tagihanList.find(t => t.id === tagihanId)?.paidAmount || 0)))
   298→  }

   301→    if (!selectedSantri || !amount || Number(amount) <= 0) {
   302→      Alert.alert("Validasi", "Pilih santri dan isi nominal")
   303→      return
   304→    }

   307→      await apiPost("/api/admin/pembayaran", {
   308→        santriId: selectedSantri,
   309→        tagihanId: selectedTagihan || undefined,
   310→        amount: Number(amount),
   311→        paymentMethod: method,
   312→      })
   313→      Alert.alert("Berhasil", `Pembayaran ${formatRupiah(Number(amount))} tercatat`)
   314→      navigation.goBack()
   315→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   316→    finally { setSaving(false) }
   317→  }
   318→
   319→  const filteredSantri = santriList.filter(s =>
   320→    !searchText || s.name.toLowerCase().includes(searchText.toLowerCase()) || s.nis.includes(searchText)
   321→  )
   322→
   323→  return (
   324→    <SafeAreaView style={styles.screen}>
   325→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   326→      <View style={styles.header}>
   327→        <TouchableOpacity onPress={() => navigation.goBack()} style={styles.backButton}>
   328→          <Ionicons name="arrow-back" size={24} color={Colors.white} />
   329→        </TouchableOpacity>
   330→        <Text style={styles.headerTitle}>Input Pembayaran</Text>
   331→        <View style={{ width: 24 }} />
   332→      </View>
   333→
   334→      {error ? (
   335→        <View style={{ flex: 1, justifyContent: "center", alignItems: "center", padding: 20 }}>
   336→          <Text style={{ color: Colors.danger, textAlign: "center", marginBottom: 16 }}>{error}</Text>
   337→          <TouchableOpacity onPress={() => { setError(null); loadSantri() }} style={{ backgroundColor: Colors.primary, paddingHorizontal: 24, paddingVertical: 12, borderRadius: 8 }}>
   338→            <Text style={{ color: Colors.white, fontWeight: "600" }}>Coba Lagi</Text>
   339→          </TouchableOpacity>
   340→        </View>
   341→      ) : (
   342→      <ScrollView style={styles.content}>
   343→        {/* Search Santri */}
   344→        <Text style={styles.label}>Cari Santri</Text>
   345→        <View style={styles.searchContainer}>
   346→          <Ionicons name="search" size={20} color={Colors.textMuted} style={{ marginRight: 8 }} />

   348→            style={styles.searchInput}

  1380→  const [checking, setChecking] = useState(true)
  1381→  // Realtime refresh tick — bumped whenever a subscribed table changes.
  1382→  // Passed to DashboardScreen so it can re-fetch on Payment/Tagihan events.
  1383→  const [refreshKey, setRefreshKey] = useState(0)
  1384→
  1385→  useEffect(() => {
  1386→    initNetworkMonitor()
  1387→    async function check() {
  1388→      const token = await SecureStore.getItemAsync("authToken")
  1389→      if (token) {
  1390→        authToken = token
  1391→        const { data } = await getSupabase().auth.getSession()
  1392→        if (data.session) { authToken = data.session.access_token; setIsLoggedIn(true) }
  1393→      }
  1394→      setChecking(false)
  1395→    }
  1396→    check()
  1397→  }, [])
  1398→
  1399→  // ============================================================================
  1400→  // Supabase Realtime subscriptions (OPTIONAL — fails silently if unavailable)
  1401→  // - Payment INSERT  : new pembayaran recorded → refresh dashboard
  1402→  // - Tagihan  UPDATE : status BELUM_BAYAR → LUNAS → refresh tunggakan/tagihan
  1403→  // ============================================================================
  1404→  useEffect(() => {
  1405→    if (!isLoggedIn) return
  1406→    let cancelled = false
  1407→    const unsubs: Array<() => void> = []
  1408→    ;(async () => {
  1409→      try {
  1410→        for (const table of ["Payment", "Tagihan"] as const) {
  1411→          try {
  1412→            const unsub = await subscribeToTable(table, () => {
  1413→              setRefreshKey(k => k + 1)
  1414→            })
  1415→            if (cancelled) { try { unsub() } catch {} }
  1416→            else { unsubs.push(unsub) }
  1417→          } catch {}
  1418→        }
  1419→      } catch(e) { /* realtime optional */ }
  1420→    })()
  1421→    return () => {
  1422→      cancelled = true
  1423→      unsubs.forEach(u => { try { u() } catch {} })
  1424→    }
  1425→  }, [isLoggedIn])
  1426→
  1427→  if (checking) {
  1428→    return <View style={{ flex: 1, justifyContent: "center", alignItems: "center", backgroundColor: Colors.primary }}><ActivityIndicator size="large" color={Colors.white} /></View>
  1429→  }
  1430→
  1431→  return (
  1432→    <View style={{ flex: 1 }}>
  1433→      <NetworkStatusIndicator />
  1434→      <NavigationContainer>
  1435→        <Stack.Navigator screenOptions={{ headerShown: false }}>
  1436→        {isLoggedIn ? (
  1437→          <>
  1438→            <Stack.Screen name="Dashboard">
  1439→              {({ navigation }) => (
  1440→                <DashboardScreen
  1441→                  navigation={navigation}
  1442→                  refreshKey={refreshKey}
  1443→                  onLogout={async () => { await getSupabase().auth.signOut(); await SecureStore.deleteItemAsync("authToken"); authToken = null; setIsLoggedIn(false) }}
  1444→                />
  1445→              )}
  1446→            </Stack.Screen>
  1447→            <Stack.Screen name="PembayaranInput" component={PembayaranInputScreen} />
  1448→            <Stack.Screen name="TunggakanList" component={TunggakanListScreen} />
  1449→            <Stack.Screen name="TagihanList" component={TagihanListScreen} />
  1450→            <Stack.Screen name="RiwayatList" component={RiwayatListScreen} />
  1451→            <Stack.Screen name="KasList" component={KasListScreen} />
  1452→            <Stack.Screen name="TransaksiKas" component={TransaksiKasScreen} />
  1453→            <Stack.Screen name="HonorGuru" component={HonorGuruScreen} />
  1454→            <Stack.Screen name="Laporan" component={LaporanScreen} />
  1455→          </>
  1456→        ) : (
  1457→          <Stack.Screen name="Login">{() => <LoginScreen onLogin={() => setIsLoggedIn(true)} />}</Stack.Screen>
  1458→        )}
  1459→      </Stack.Navigator>
  1460→    </NavigationContainer>
  1461→    </View>
  1462→  )
  1463→}
