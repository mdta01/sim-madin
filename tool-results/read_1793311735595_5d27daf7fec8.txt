    98→function LoginScreen({ onLogin, onForgotPin }: { onLogin: (token: string, wali: WaliData, santriList: SantriData[]) => void; onForgotPin?: () => void }) {
    99→  const [phone, setPhone] = useState("081234511001")
   100→  const [pin, setPin] = useState("")
   101→  const [loading, setLoading] = useState(false)
   102→  const [showPin, setShowPin] = useState(false)
   103→
   104→  const handleLogin = async () => {
   105→    if (!phone || !pin) {
   106→      Alert.alert("Validasi", "Nomor WhatsApp dan PIN wajib diisi")
   107→      return
   108→    }
   109→    if (pin.length !== 6) {
   110→      Alert.alert("Validasi", "PIN harus 6 digit")
   111→      return
   112→    }
   113→
   114→    setLoading(true)
   115→    try {
   116→      const data = await api.post("/api/wali/login", { phone, pin })
   117→      if (data.success) {
   118→        waliToken = data.token
   119→        await saveWaliSession(data.token, data.wali, data.santriList)
   120→        onLogin(data.token, data.wali, data.santriList)
   121→      } else {
   122→        Alert.alert("Login Gagal", data.error || "Terjadi kesalahan")
   123→      }
   124→    } catch (e: any) {
   125→      Alert.alert("Login Gagal", e.message)
   126→    } finally {
   127→      setLoading(false)
   128→    }
   129→  }
   130→
   131→  return (
   132→    <SafeAreaView style={styles.loginContainer}>
   133→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   134→      <ScrollView contentContainerStyle={styles.loginScroll}>
   135→        <View style={styles.loginLogo}>
   136→          <Ionicons name="people-circle" size={80} color={Colors.primary} />
   137→          <Text style={styles.loginTitle}>SIM MADIN</Text>
   138→          <Text style={styles.loginSubtitle}>Wali Santri</Text>
   139→        </View>
   140→
   141→        <View style={styles.loginCard}>
   142→          <Text style={styles.loginCardTitle}>Masuk ke Akun</Text>
   143→          <Text style={styles.loginCardDesc}>Masuk dengan Nomor WhatsApp & PIN</Text>
   144→
   145→          <Text style={styles.label}>Nomor WhatsApp</Text>
   146→          <View style={styles.inputContainer}>
   147→            <Ionicons name="call" size={20} color={Colors.textMuted} style={styles.inputIcon} />
   148→            <TextInput
   149→              style={styles.input}
   150→              placeholder="08xxx atau 628xxx"
   151→              value={phone}
   152→              onChangeText={setPhone}
   153→              keyboardType="phone-pad"
   154→              autoCapitalize="none"
   155→            />
   156→          </View>
   157→
   158→          <Text style={styles.label}>PIN (6 digit)</Text>
   159→          <View style={styles.inputContainer}>
   160→            <Ionicons name="lock-closed" size={20} color={Colors.textMuted} style={styles.inputIcon} />
   161→            <TextInput
   162→              style={[styles.input, { letterSpacing: 8, textAlign: "center" }]}
   163→              placeholder="••••••"
   164→              value={pin}
   165→              onChangeText={(t) => setPin(t.replace(/\D/g, "").slice(0, 6))}
   166→              keyboardType="numeric"
   167→              secureTextEntry={!showPin}
   168→              maxLength={6}

   172→            </TouchableOpacity>

   176→            style={[styles.loginButton, loading && { opacity: 0.6 }]}

   181→              <ActivityIndicator color={Colors.white} />

   183→              <Text style={styles.loginButtonText}>Masuk</Text>

   193→      </ScrollView>
   194→    </SafeAreaView>

   198→// --- Forgot PIN Screen (3-step OTP flow) ---
   199→function ForgotPinScreen({ onBack }: { onBack: () => void }) {
   200→  const [step, setStep] = useState<"phone" | "otp" | "setPin">("phone")
   201→  const [phone, setPhone] = useState("")
   202→  const [otp, setOtp] = useState("")
   203→  const [newPin, setNewPin] = useState("")
   204→  const [confirmPin, setConfirmPin] = useState("")
   205→  const [loading, setLoading] = useState(false)
   206→  const [verifyToken, setVerifyToken] = useState("")
   207→
   208→  const handleRequestOtp = async () => {
   209→    if (!phone) { Alert.alert("Validasi", "Nomor WhatsApp wajib diisi"); return }
   210→    setLoading(true)
   211→    try {
   212→      const res = await api.post("/api/wali/request-otp", { phone, type: "FORGOT_PIN" })
   213→      Alert.alert("OTP Terkirim", res.devOtp ? `Kode OTP (DEV): ${res.devOtp}` : res.message)
   214→      setStep("otp")
   215→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   216→    finally { setLoading(false) }
   217→  }
   218→
   219→  const handleVerifyOtp = async () => {
   220→    if (!otp || otp.length !== 6) { Alert.alert("Validasi", "OTP harus 6 digit"); return }
   221→    setLoading(true)
   222→    try {
   223→      const res = await api.post("/api/wali/verify-otp", { phone, otpCode: otp })
   224→      if (res.verifyToken) {
   225→        setVerifyToken(res.verifyToken)
   226→        setStep("setPin")
   227→        Alert.alert("Terverifikasi", "Silakan buat PIN baru")
   228→      } else { Alert.alert("Gagal", "Token verifikasi tidak diterima") }
   229→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   230→    finally { setLoading(false) }
   231→  }
   232→
   233→  const handleSetPin = async () => {
   234→    if (!newPin || newPin.length !== 6) { Alert.alert("Validasi", "PIN harus 6 digit"); return }
   235→    if (newPin !== confirmPin) { Alert.alert("Validasi", "PIN tidak cocok"); return }
   236→    setLoading(true)
   237→    try {
   238→      await api.post("/api/wali/set-pin", { verifyToken, pin: newPin })
   239→      Alert.alert("Berhasil", "PIN berhasil diubah. Silakan login dengan PIN baru.", [{ text: "OK", onPress: onBack }])
   240→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   241→    finally { setLoading(false) }
   242→  }
   243→
   244→  return (
   245→    <SafeAreaView style={styles.loginContainer}>
   246→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   247→      <ScrollView contentContainerStyle={styles.loginScroll}>
   248→        <View style={styles.loginLogo}>
   249→          <Ionicons name="key" size={80} color={Colors.white} />
   250→          <Text style={styles.loginTitle}>Lupa PIN</Text>
   251→          <Text style={styles.loginSubtitle}>Reset PIN via OTP</Text>
   252→        </View>
   253→
   254→        <View style={styles.loginCard}>
   255→          {step === "phone" && (
   256→            <>
   257→              <Text style={styles.loginCardTitle}>Masukkan Nomor WhatsApp</Text>
   258→              <Text style={styles.loginCardDesc}>Kami akan kirim kode OTP ke nomor Anda</Text>
   259→              <Text style={styles.label}>Nomor WhatsApp</Text>
   260→              <View style={styles.inputContainer}>
   261→                <Ionicons name="call" size={20} color={Colors.textMuted} style={styles.inputIcon} />
   262→                <TextInput style={styles.input} placeholder="08xxx atau 628xxx" value={phone} onChangeText={setPhone} keyboardType="phone-pad" />
   263→              </View>
   264→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleRequestOtp} disabled={loading}>
   265→                {loading ? <ActivityIndicator color={Colors.white} /> : <Text style={styles.loginButtonText}>Kirim OTP</Text>}
   266→              </TouchableOpacity>
   267→            </>
   268→          )}
   269→
   270→          {step === "otp" && (
   271→            <>
   272→              <Text style={styles.loginCardTitle}>Verifikasi OTP</Text>
   273→              <Text style={styles.loginCardDesc}>Masukkan 6 digit kode yang dikirim</Text>
   274→              <TextInput
   275→                style={[styles.input, { textAlign: "center", fontSize: 24, letterSpacing: 8, fontFamily: "monospace" }]}
   276→                placeholder="000000"
   277→                value={otp}
   278→                onChangeText={(t) => setOtp(t.replace(/\D/g, "").slice(0, 6))}
   279→                keyboardType="numeric"
   280→                maxLength={6}
   281→              />
   282→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleVerifyOtp} disabled={loading}>
   283→                {loading ? <ActivityIndicator color={Colors.white} /> : <Text style={styles.loginButtonText}>Verifikasi</Text>}
   284→              </TouchableOpacity>
   285→            </>
   286→          )}
   287→
   288→          {step === "setPin" && (
   289→            <>
   290→              <Text style={styles.loginCardTitle}>Buat PIN Baru</Text>
   291→              <Text style={styles.loginCardDesc}>PIN 6 digit untuk login</Text>
   292→              <Text style={styles.label}>PIN Baru</Text>
   293→              <TextInput
   294→                style={[styles.input, { textAlign: "center", fontSize: 20, letterSpacing: 6, fontFamily: "monospace" }]}
   295→                placeholder="••••••"
   296→                value={newPin}
   297→                onChangeText={(t) => setNewPin(t.replace(/\D/g, "").slice(0, 6))}
   298→                keyboardType="numeric"
   299→                secureTextEntry
   300→                maxLength={6}
   301→              />
   302→              <Text style={styles.label}>Konfirmasi PIN</Text>
   303→              <TextInput
   304→                style={[styles.input, { textAlign: "center", fontSize: 20, letterSpacing: 6, fontFamily: "monospace" }]}
   305→                placeholder="••••••"
   306→                value={confirmPin}
   307→                onChangeText={(t) => setConfirmPin(t.replace(/\D/g, "").slice(0, 6))}
   308→                keyboardType="numeric"
   309→                secureTextEntry
   310→                maxLength={6}
   311→              />
   312→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleSetPin} disabled={loading}>
   313→                {loading ? <ActivityIndicator color={Colors.white} /> : <Text style={styles.loginButtonText}>Simpan PIN</Text>}
   314→              </TouchableOpacity>
   315→            </>
   316→          )}
   317→
   318→          <TouchableOpacity onPress={onBack} style={[styles.forgotPinBtn, { marginTop: 16 }]}>
   319→            <Text style={styles.forgotPinText}>← Kembali ke Login</Text>
   320→          </TouchableOpacity>
   321→        </View>
   322→      </ScrollView>
   323→    </SafeAreaView>
   324→  )
   325→}

   744→function WaliNotifikasiSection({ token }: { token: string | null }) {

   866→function WaliChatRoom({ room, token, onBack }: { room: any, token: string | null, onBack: () => void }) {
   867→  const [messages, setMessages] = useState<any[]>([])
   868→  const [input, setInput] = useState("")
   869→  const [loading, setLoading] = useState(true)
   870→
   871→  const loadMessages = useCallback(async () => {
   872→    if (!token) return
   873→    try {
   874→      const res = await fetch(`${API_BASE_URL}/api/wali/chat/${room.id}/messages`, {
   875→        headers: { Authorization: `Bearer ${token}` },
   876→      })
   877→      const data = await res.json()
   878→      setMessages(data.data || [])
   879→    } catch (e: any) {
   880→      console.error("msg load error:", e.message)
   881→    } finally {
   882→      setLoading(false)
   883→    }
   884→  }, [token, room.id])
   885→
   886→  useEffect(() => { loadMessages() }, [loadMessages])
   887→
   888→  const send = async () => {
   889→    if (!input.trim() || !token) return
   890→    try {
   891→      await fetch(`${API_BASE_URL}/api/wali/chat/${room.id}/messages`, {
   892→        method: "POST",
   893→        headers: { "Content-Type": "application/json", Authorization: `Bearer ${token}` },
   894→        body: JSON.stringify({ message: input }),
   895→      })
   896→      setInput("")
   897→      loadMessages()
   898→    } catch (e: any) {
   899→      Alert.alert("Gagal", e.message)
   900→    }
   901→  }
   902→
   903→  if (loading) return <View style={styles.contentCard}><ActivityIndicator size="large" color={Colors.primary} /></View>
   904→
   905→  return (
   906→    <View style={styles.contentCard}>
   907→      <TouchableOpacity onPress={onBack} style={{ flexDirection: "row", alignItems: "center", marginBottom: 8 }}>
   908→        <Ionicons name="arrow-back" size={20} color={Colors.primary} />
   909→        <Text style={{ color: Colors.primary, marginLeft: 4 }}>Kembali</Text>
   910→      </TouchableOpacity>
   911→      <Text style={styles.contentTitle}>{room.type === "WALI_WALI_KELAS" ? "Chat Wali Kelas" : "Chat Madrasah"}</Text>
   912→      <View style={{ maxHeight: 300, marginBottom: 8 }}>
   913→        <ScrollView>

   915→            <View key={i} style={{
   916→              alignSelf: m.senderType === "WALI" ? "flex-end" : "flex-start",
   917→              backgroundColor: m.senderType === "WALI" ? Colors.primaryLight : Colors.border,
   918→              padding: 8, borderRadius: 8, marginBottom: 4, maxWidth: "80%",
   919→            }}>
   920→              <Text style={{ fontSize: 13, color: m.senderType === "WALI" ? Colors.primaryDark : Colors.text }}>{m.message}</Text>
   921→              <Text style={{ fontSize: 9, color: Colors.textMuted, marginTop: 2 }}>{formatDate(m.createdAt, true)}</Text>
   922→            </View>
   923→          ))}
   924→        </ScrollView>
   925→      </View>
   926→      <View style={{ flexDirection: "row", gap: 8 }}>
   927→        <TextInput
   928→          style={[styles.textInput, { flex: 1 }]}
   929→          value={input}
   930→          onChangeText={setInput}
   931→          placeholder="Tulis pesan..."
   932→        />
   933→        <TouchableOpacity onPress={send} style={[styles.submitButton, { paddingHorizontal: 16 }]}>
   934→          <Ionicons name="send" size={18} color={Colors.white} />
   935→        </TouchableOpacity>
   936→      </View>

  1505→function UjianTakeScreen({ ujian, santriId, onClose }: { ujian: any, santriId: string, onClose: () => void }) {
  1506→  const [detail, setDetail] = useState<any>(null)

  1508→  const [answers, setAnswers] = useState<Record<string, { answer: string; optionOrder?: number[] }>>({})

  1512→  const [result, setResult] = useState<any>(null)
  1513→

  1517→        setLoading(true)
  1518→        // Start or resume ujian
  1519→        await api.post(`/api/wali/ujian/${ujian.id}/start`, { santriId })
  1520→        // Get detail with shuffled soal
  1521→        const res = await api.get(`/api/wali/ujian/${ujian.id}`, { santriId })
  1522→        setDetail(res.data)
  1523→        setRemainingSeconds(res.data.remainingSeconds || ujian.durationMnt * 60)
  1524→        // Init answers from existing
  1525→        const init: Record<string, { answer: string; optionOrder?: number[] }> = {}
  1526→        res.data.soalList.forEach((s: any) => {
  1527→          init[s.ujianSoalId] = {
  1528→            answer: s.existingAnswer || "",
  1529→            optionOrder: s.optionOrder,
  1530→          }
  1531→        })
  1532→        setAnswers(init)
  1533→      } catch (e: any) {
  1534→        Alert.alert("Error", e.message)
  1535→        onClose()
  1536→      } finally {
  1537→        setLoading(false)
  1538→      }
  1539→    })()
  1540→  }, [ujian.id, santriId, ujian.durationMnt])
  1541→
  1542→  // Timer countdown
  1543→  useEffect(() => {
  1544→    if (loading || result) return
  1545→    const timer = setInterval(() => {
  1546→      setRemainingSeconds((prev) => {
  1547→        if (prev <= 1) {
  1548→          clearInterval(timer)
  1549→          handleAutoSubmit()
  1550→          return 0
  1551→        }
  1552→        return prev - 1
  1553→      })
  1554→    }, 1000)
  1555→    return () => clearInterval(timer)
  1556→  }, [loading, result])
  1557→
  1558→  const formatTime = (sec: number) => {
  1559→    const m = Math.floor(sec / 60)
  1560→    const s = sec % 60
  1561→    return `${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`
  1562→  }
  1563→
  1564→  const handleAnswer = (ujianSoalId: string, answer: string) => {
  1565→    setAnswers({ ...answers, [ujianSoalId]: { ...answers[ujianSoalId], answer } })
  1566→  }
  1567→
  1568→  const handleSubmit = () => {
  1569→    Alert.alert(
  1570→      "Konfirmasi Submit",
  1571→      "Yakin ingin mengumpulkan ujian? Anda tidak bisa mengubah jawaban setelah submit.",
  1572→      [
  1573→        { text: "Batal", style: "cancel" },
  1574→        { text: "Kumpulkan", onPress: () => doSubmit(false) },
  1575→      ]
  1576→    )
  1577→  }
  1578→
  1579→  const handleAutoSubmit = () => {
  1580→    Alert.alert("Waktu Habis", "Waktu ujian telah berakhir. Jawaban akan dikumpulkan otomatis.")
  1581→    doSubmit(true)
  1582→  }
  1583→

  1588→        ujianSoalId,
  1589→        answer: v.answer,
  1590→        optionOrder: v.optionOrder,
  1591→      }))
  1592→      const res = await api.post(`/api/wali/ujian/${ujian.id}/submit`, { santriId, answers: answerArray })
  1593→      setResult(res.data)
  1594→    } catch (e: any) {
  1595→      Alert.alert("Gagal Submit", e.message)
  1596→    } finally {
  1597→      setSubmitting(false)
  1598→    }
  1599→  }
  1600→
  1601→  if (loading) {
  1602→    return (
  1603→      <View style={styles.contentCard}>
  1604→        <ActivityIndicator size="large" color={Colors.primary} />
  1605→        <Text style={{ textAlign: "center", marginTop: 10, color: Colors.textMuted }}>Memuat soal ujian...</Text>
  1606→      </View>
  1607→    )
  1608→  }
  1609→
  1610→  if (result) {
  1611→    return (
  1612→      <View style={styles.contentCard}>
  1613→        <Text style={[styles.contentTitle, { textAlign: "center" }]}>Ujian Selesai</Text>
  1614→        <View style={{ alignItems: "center", marginTop: 20 }}>
  1615→          <Text style={{ fontSize: 48, fontWeight: "bold", color: result.data.score >= 70 ? Colors.success : Colors.danger }}>

  1620→            {result.message}
  1621→          </Text>
  1622→        </View>
  1623→        <TouchableOpacity style={[styles.submitButton, { marginTop: 24 }]} onPress={onClose}>
  1624→          <Text style={styles.submitButtonText}>Kembali ke Daftar Ujian</Text>
  1625→        </TouchableOpacity>
  1626→      </View>
  1627→    )
  1628→  }
  1629→
  1630→  if (!detail) return null
  1631→  const soalList = detail.soalList || []
  1632→  const soal = soalList[currentSoal]
  1633→  const answeredCount = Object.values(answers).filter((a) => a.answer).length

  1636→  return (

  1638→      {/* Timer + Progress */}
  1639→      <View style={{ flexDirection: "row", justifyContent: "space-between", marginBottom: 12 }}>
  1640→        <View style={[styles.statusBadge, { backgroundColor: isLowTime ? Colors.dangerLight : Colors.primaryLight }]}>
  1641→          <Ionicons name="time" size={14} color={isLowTime ? Colors.danger : Colors.primary} />
  1642→          <Text style={[styles.statusBadgeText, { color: isLowTime ? Colors.danger : Colors.primary, marginLeft: 4 }]}>
  1643→            {formatTime(remainingSeconds)}

  1645→        </View>
  1646→        <Text style={{ fontSize: 12, color: Colors.textMuted }}>
  1647→          Soal {currentSoal + 1}/{soalList.length} • Terjawab: {answeredCount}
  1648→        </Text>
  1649→      </View>
  1650→
  1651→      {/* Soal */}
  1652→      <View style={{ marginBottom: 16 }}>
  1653→        <Text style={{ fontSize: 14, fontWeight: "600", marginBottom: 8 }}>{soal.question}</Text>

  1655→          <View>

  1659→              return (

  1663→                  style={{
  1664→                    flexDirection: "row", alignItems: "center", padding: 10, marginBottom: 6,
