   190→function UjianForm({ open, onOpenChange, editing, onSuccess, mapelList, kelasList, guruList }: any) {
   191→  const [form, setForm] = useState<any>({})
   192→  const [saving, setSaving] = useState(false)
   193→  const { toast } = useToast()
   194→
   195→  useEffect(() => {
   196→    if (editing) {
   197→      setForm({
   198→        ...editing,
   199→        startTime: editing.startTime ? new Date(editing.startTime).toISOString().slice(0, 16) : "",
   200→        endTime: editing.endTime ? new Date(editing.endTime).toISOString().slice(0, 16) : "",
   201→      })
   202→    } else {
   203→      const now = new Date()
   204→      const later = new Date(now.getTime() + 2 * 3600000)
   205→      setForm({
   206→        type: "UTS",
   207→        durationMnt: 90,
   208→        maxScore: 100,
   209→        isPublished: false,
   210→        // V3: ujian selalu offline (manual input nilai)
   211→        startTime: now.toISOString().slice(0, 16),
   212→        endTime: later.toISOString().slice(0, 16),
   213→      })
   214→    }
   215→  }, [editing, open])
   216→
   217→  const handleSave = async () => {
   218→    if (!form.title || !form.type || !form.mapelId || !form.kelasId || !form.startTime || !form.endTime) {
   219→      toast({ title: "Validasi gagal", description: "Semua field wajib diisi", variant: "destructive" })
   220→      return
   221→    }
   222→    setSaving(true)
   223→    try {
   224→      if (editing) {
   225→        await apiPut(`/api/admin/ujian/${editing.id}`, form)
   226→        toast({ title: "Berhasil", description: "Ujian diperbarui" })
   227→      } else {
   228→        await apiPost("/api/admin/ujian", form)
   229→        toast({ title: "Berhasil", description: "Ujian baru dibuat" })
   230→      }
   231→      onOpenChange(false)
   232→      onSuccess()
   233→    } catch (e: any) {
   234→      toast({ title: "Gagal", description: e.message, variant: "destructive" })
   235→    } finally {
   236→      setSaving(false)
   237→    }
   238→  }
   239→
   240→  return (
   241→    <Dialog open={open} onOpenChange={onOpenChange}>
   242→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   243→        <DialogHeader><DialogTitle>{editing ? "Edit Ujian" : "Tambah Ujian"}</DialogTitle></DialogHeader>
   244→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 py-2">
   245→          <div className="space-y-1 sm:col-span-2">

   249→          <div className="space-y-1">
   250→            <Label className="text-xs">Tipe Ujian *</Label>
   251→            <Select value={form.type} onValueChange={(v) => setForm({ ...form, type: v })}>
   252→              <SelectTrigger><SelectValue /></SelectTrigger>
   253→              <SelectContent>
   254→                {UJIAN_TYPES.map((t) => <SelectItem key={t.value} value={t.value}>{t.label}</SelectItem>)}
   255→              </SelectContent>
   256→            </Select>
   257→          </div>
   258→          <div className="space-y-1">
   259→            <Label className="text-xs">Durasi (menit)</Label>
   260→            <Input type="number" value={form.durationMnt || 90} onChange={(e) => setForm({ ...form, durationMnt: Number(e.target.value) })} />
   261→          </div>
   262→          <div className="space-y-1">

   376→function UjianSoalTab({ ujianId }: { ujianId: string }) {
   377→  const [soalLinks, setSoalLinks] = useState<any[]>([])
   378→  const [bankSoalList, setBankSoalList] = useState<any[]>([])
   379→  const [loading, setLoading] = useState(true)
   380→  const [assignOpen, setAssignOpen] = useState(false)
   381→  const [selectedSoal, setSelectedSoal] = useState<string[]>([])
   382→  const { toast } = useToast()
   383→
   384→  const load = useCallback(async () => {
   385→    try {
   386→      setLoading(true)
   387→      const [linksRes, bankRes] = await Promise.all([
   388→        apiGet(`/api/admin/ujian/${ujianId}/soal`),
   389→        apiGet("/api/admin/bank-soal"),
   390→      ])
   391→      setSoalLinks(linksRes.data)
   392→      setBankSoalList(bankRes.data)
   393→    } catch (e: any) {
   394→      toast({ title: "Gagal", description: e.message, variant: "destructive" })
   395→    } finally {
   396→      setLoading(false)
   397→    }
   398→  }, [ujianId, toast])
   399→
   400→  useEffect(() => { load() }, [load])
   401→
   402→  const handleAssign = async () => {
   403→    if (selectedSoal.length === 0) {
   404→      toast({ title: "Validasi", description: "Pilih minimal 1 soal" })
   405→      return
   406→    }
   407→    try {
   408→      const res = await apiPost(`/api/admin/ujian/${ujianId}/soal`, { soalIds: selectedSoal })
   409→      toast({ title: "Berhasil", description: `${res.count} soal ditambahkan` })
   410→      setAssignOpen(false)
   411→      setSelectedSoal([])
   412→      load()
   413→    } catch (e: any) {
   414→      toast({ title: "Gagal", description: e.message, variant: "destructive" })
   415→    }
   416→  }
   417→
   418→  const handleRemove = async (linkId: string) => {
   419→    if (!confirm("Hapus soal ini dari ujian?")) return
   420→    try {
   421→      await apiDelete(`/api/admin/ujian/${ujianId}/soal?linkId=${linkId}`)
   422→      toast({ title: "Berhasil", description: "Soal dihapus dari ujian" })
   423→      load()
   424→    } catch (e: any) {
   425→      toast({ title: "Gagal", description: e.message, variant: "destructive" })
   426→    }
   427→  }
   428→
   429→  if (loading) return <LoadingState />
   430→
   431→  return (
   432→    <div>
   433→      <div className="flex justify-between items-center mb-3">
   434→        <p className="text-sm text-muted-foreground">{soalLinks.length} soal di ujian ini</p>
   435→        <Button size="sm" onClick={() => setAssignOpen(true)}><Plus className="h-4 w-4 mr-1" /> Tambah Soal</Button>
   436→      </div>
   437→      {soalLinks.length === 0 ? (
   438→        <EmptyState icon={<Library className="h-5 w-5" />} title="Belum ada soal" description="Tambahkan soal dari Bank Soal" />
   439→      ) : (
   440→        <div className="space-y-2">
   441→          {soalLinks.map((link) => (
   442→            <Card key={link.id}>
   443→              <CardContent className="p-3 flex items-start gap-3">
   444→                <Badge variant="secondary" className="text-xs">#{link.urutan}</Badge>
   445→                <div className="flex-1 min-w-0">
   446→                  <div className="flex items-center gap-2 mb-1">

   586→function UjianNilaiTab({ ujianId, maxScore, onSuccess }: any) {
   587→  const [nilaiData, setNilaiData] = useState<any[]>([])
   588→  const [stats, setStats] = useState<any>(null)
   589→  const [loading, setLoading] = useState(true)
   590→  const [entries, setEntries] = useState<Record<string, string>>({})
   591→  const [saving, setSaving] = useState(false)
   592→  const { toast } = useToast()
   593→
   594→  const load = useCallback(async () => {
   595→    try {
   596→      setLoading(true)
   597→      const res = await apiGet(`/api/admin/ujian/${ujianId}/nilai`)
   598→      setNilaiData(res.data)
   599→      setStats(res.stats)
   600→      const init: Record<string, string> = {}
   601→      res.data.forEach((p: any) => {
   602→        if (p.nilai) init[p.santriId] = String(p.nilai.score)
   603→      })
   604→      setEntries(init)
   605→    } catch (e: any) {
   606→      toast({ title: "Gagal", description: e.message, variant: "destructive" })
   607→    } finally {
   608→      setLoading(false)
   609→    }
   610→  }, [ujianId, toast])
   611→
   612→  useEffect(() => { load() }, [load])
   613→
   614→  const handleSave = async () => {
   615→    const payload = Object.entries(entries)
   616→      .filter(([, v]) => v !== "" && v !== null)
   617→      .map(([santriId, v]) => ({ santriId, score: Number(v) }))
   618→    if (payload.length === 0) {
   619→      toast({ title: "Info", description: "Tidak ada nilai untuk disimpan" })
   620→      return
   621→    }
   622→    setSaving(true)
   623→    try {
   624→      const res = await apiPost(`/api/admin/ujian/${ujianId}/nilai`, { entries: payload })
   625→      toast({ title: "Berhasil", description: `${res.count} nilai tersimpan` })
   626→      load()
   627→      onSuccess()
   628→    } catch (e: any) {
   629→      toast({ title: "Gagal", description: e.message, variant: "destructive" })
   630→    } finally {
   631→      setSaving(false)
   632→    }
   633→  }
   634→
   635→  if (loading) return <LoadingState />
   636→
   637→  return (
   638→    <div>
   639→      {stats && (
   640→        <div className="grid grid-cols-2 sm:grid-cols-5 gap-2 mb-3">
   641→          <Card><CardContent className="p-2"><p className="text-[10px] text-muted-foreground">Total</p><p className="text-lg font-bold">{stats.total}</p></CardContent></Card>
   642→          <Card><CardContent className="p-2"><p className="text-[10px] text-muted-foreground">Dinilai</p><p className="text-lg font-bold text-green-600">{stats.graded}</p></CardContent></Card>
   643→          <Card><CardContent className="p-2"><p className="text-[10px] text-muted-foreground">Rata-rata</p><p className="text-lg font-bold text-blue-600">{stats.avgScore}</p></CardContent></Card>
   644→          <Card><CardContent className="p-2"><p className="text-[10px] text-muted-foreground">Lulus</p><p className="text-lg font-bold text-green-600">{stats.lulus}</p></CardContent></Card>
   645→          <Card><CardContent className="p-2"><p className="text-[10px] text-muted-foreground">Tidak Lulus</p><p className="text-lg font-bold text-red-600">{stats.tidakLulus}</p></CardContent></Card>
   646→        </div>
   647→      )}
   648→      <div className="space-y-1.5 max-h-[400px] overflow-y-auto mb-3">
   649→        {nilaiData.map((p) => (
   650→          <div key={p.id} className="flex items-center gap-2 p-2 border rounded">
   651→            <div className="flex h-8 w-8 items-center justify-center rounded-full bg-primary/10 text-primary text-xs font-semibold">
   652→              {p.santri?.name?.[0] || "?"}
   653→            </div>
   654→            <div className="flex-1 min-w-0">
   655→              <p className="text-sm font-medium truncate">{p.santri?.name}</p>
   656→              <p className="text-[10px] text-muted-foreground font-mono">{p.santri?.nis}</p>
   657→            </div>
   658→            <Input
   659→              type="number"
   660→              min={0}
   661→              max={maxScore}
   662→              value={entries[p.santriId] || ""}
   663→              onChange={(e) => setEntries({ ...entries, [p.santriId]: e.target.value })}
   664→              placeholder="-"
   665→              className="w-20 h-8"
   666→            />
   667→            <span className="text-xs text-muted-foreground shrink-0">/ {maxScore}</span>
   668→            {p.isGraded && <Badge variant="success" className="text-[10px]">Sudah dinilai</Badge>}
   669→          </div>
   670→        ))}
   671→      </div>
   672→      <div className="flex justify-end">
   673→        <Button onClick={handleSave} disabled={saving}>
   674→          <Save className="h-4 w-4 mr-1" /> {saving ? "Menyimpan..." : `Simpan ${Object.values(entries).filter(v => v).length} Nilai`}
   675→        </Button>
   676→      </div>
   677→    </div>
   678→  )
