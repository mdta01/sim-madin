    43→    43→async function apiGet(path: string) {
    44→    44→  const res = await fetch(`${API_BASE_URL}${path}`, { headers: authToken ? { Authorization: `Bearer ${authToken}` } : {} })

    49→    49→async function apiPost(path: string, body: any) {
    50→    50→  const res = await fetch(`${API_BASE_URL}${path}`, {
    51→    51→    method: "POST",
    52→    52→    headers: { "Content-Type": "application/json", ...(authToken ? { Authorization: `Bearer ${authToken}` } : {}) },
    53→    53→    body: JSON.stringify(body),
    54→    54→  })
    55→    55→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    56→    56→  return res.json()
    57→    57→}

  1248→  1248→            })
  1249→  1249→            if (cancelled) { try { unsub() } catch {} }
  1250→  1250→            else { unsubs.push(unsub) }

  1252→  1252→        }
