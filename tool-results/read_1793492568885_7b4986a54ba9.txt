   101→function LoginScreen({ onLogin, onForgotPin }: { onLogin: (token: string, wali: WaliData, santriList: SantriData[]) => void; onForgotPin?: () => void }) {
   102→  const [phone, setPhone] = useState("")
   103→  const [pin, setPin] = useState("")
   104→  const [loading, setLoading] = useState(false)
   105→  const [showPin, setShowPin] = useState(false)
   106→
   107→  const handleLogin = async () => {
   108→    if (!phone || !pin) {
   109→      Alert.alert("Validasi", "Nomor WhatsApp dan PIN wajib diisi")
   110→      return
   111→    }
   112→    if (pin.length !== 6) {
   113→      Alert.alert("Validasi", "PIN harus 6 digit")
   114→      return
   115→    }
   116→
   117→    setLoading(true)
   118→    try {
   119→      const data = await api.post("/api/wali/login", { phone, pin })
   120→      if (data.success) {
   121→        waliToken = data.token
   122→        await saveWaliSession(data.token, data.wali, data.santriList)
   123→        onLogin(data.token, data.wali, data.santriList)
   124→      } else {
   125→        Alert.alert("Login Gagal", data.error || "Terjadi kesalahan")
   126→      }
   127→    } catch (e: any) {
   128→      Alert.alert("Login Gagal", e.message)
   129→    } finally {
   130→      setLoading(false)
   131→    }
   132→  }
   133→
   134→  return (
   135→    <SafeAreaView style={styles.loginContainer}>
   136→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   137→      <ScrollView contentContainerStyle={styles.loginScroll}>
   138→        <View style={styles.loginLogo}>
   139→          <Ionicons name="people-circle" size={80} color={Colors.primary} />
   140→          <Text style={styles.loginTitle}>SIM MADIN</Text>
   141→          <Text style={styles.loginSubtitle}>Wali Santri</Text>
   142→        </View>
   143→
   144→        <View style={styles.loginCard}>
   145→          <Text style={styles.loginCardTitle}>Masuk ke Akun</Text>
   146→          <Text style={styles.loginCardDesc}>Masuk dengan Nomor WhatsApp & PIN</Text>
   147→
   148→          <Text style={styles.label}>Nomor WhatsApp</Text>
   149→          <View style={styles.inputContainer}>
   150→            <Ionicons name="call" size={20} color={Colors.textMuted} style={styles.inputIcon} />
   151→            <TextInput
   152→              style={styles.input}
   153→              placeholder="08xxx atau 628xxx"
   154→              value={phone}
   155→              onChangeText={setPhone}
   156→              keyboardType="phone-pad"
   157→              autoCapitalize="none"

   166→              placeholder="••••••"

   201→// --- Forgot PIN Screen (3-step OTP flow) ---
   202→function ForgotPinScreen({ onBack }: { onBack: () => void }) {
   203→  const [step, setStep] = useState<"phone" | "otp" | "setPin">("phone")
   204→  const [phone, setPhone] = useState("")
   205→  const [otp, setOtp] = useState("")
   206→  const [newPin, setNewPin] = useState("")
   207→  const [confirmPin, setConfirmPin] = useState("")
   208→  const [loading, setLoading] = useState(false)
   209→  const [verifyToken, setVerifyToken] = useState("")
   210→
   211→  const handleRequestOtp = async () => {
   212→    if (!phone) { Alert.alert("Validasi", "Nomor WhatsApp wajib diisi"); return }
   213→    setLoading(true)
   214→    try {
   215→      const res = await api.post("/api/wali/request-otp", { phone, type: "FORGOT_PIN" })
   216→      Alert.alert("OTP Terkirim", res.devOtp ? `Kode OTP (DEV): ${res.devOtp}` : res.message)
   217→      setStep("otp")
   218→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   219→    finally { setLoading(false) }
   220→  }
   221→
   222→  const handleVerifyOtp = async () => {
   223→    if (!otp || otp.length !== 6) { Alert.alert("Validasi", "OTP harus 6 digit"); return }
   224→    setLoading(true)
   225→    try {
   226→      const res = await api.post("/api/wali/verify-otp", { phone, otpCode: otp })
   227→      if (res.verifyToken) {
   228→        setVerifyToken(res.verifyToken)
   229→        setStep("setPin")
   230→        Alert.alert("Terverifikasi", "Silakan buat PIN baru")
   231→      } else { Alert.alert("Gagal", "Token verifikasi tidak diterima") }
   232→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   233→    finally { setLoading(false) }
   234→  }
   235→
   236→  const handleSetPin = async () => {
   237→    if (!newPin || newPin.length !== 6) { Alert.alert("Validasi", "PIN harus 6 digit"); return }
   238→    if (newPin !== confirmPin) { Alert.alert("Validasi", "PIN tidak cocok"); return }
   239→    setLoading(true)
   240→    try {
   241→      await api.post("/api/wali/set-pin", { verifyToken, pin: newPin })
   242→      Alert.alert("Berhasil", "PIN berhasil diubah. Silakan login dengan PIN baru.", [{ text: "OK", onPress: onBack }])
   243→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   244→    finally { setLoading(false) }
   245→  }
   246→
   247→  return (
   248→    <SafeAreaView style={styles.loginContainer}>
   249→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   250→      <ScrollView contentContainerStyle={styles.loginScroll}>
   251→        <View style={styles.loginLogo}>
   252→          <Ionicons name="key" size={80} color={Colors.white} />
   253→          <Text style={styles.loginTitle}>Lupa PIN</Text>
   254→          <Text style={styles.loginSubtitle}>Reset PIN via OTP</Text>
   255→        </View>
   256→
   257→        <View style={styles.loginCard}>
   258→          {step === "phone" && (
   259→            <>
   260→              <Text style={styles.loginCardTitle}>Masukkan Nomor WhatsApp</Text>
   261→              <Text style={styles.loginCardDesc}>Kami akan kirim kode OTP ke nomor Anda</Text>
   262→              <Text style={styles.label}>Nomor WhatsApp</Text>
   263→              <View style={styles.inputContainer}>
   264→                <Ionicons name="call" size={20} color={Colors.textMuted} style={styles.inputIcon} />
   265→                <TextInput style={styles.input} placeholder="08xxx atau 628xxx" value={phone} onChangeText={setPhone} keyboardType="phone-pad" />
   266→              </View>
   267→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleRequestOtp} disabled={loading}>
   268→                {loading ? <ActivityIndicator color={Colors.white} /> : <Text style={styles.loginButtonText}>Kirim OTP</Text>}
   269→              </TouchableOpacity>
   270→            </>
   271→          )}
   272→
   273→          {step === "otp" && (
   274→            <>
   275→              <Text style={styles.loginCardTitle}>Verifikasi OTP</Text>
   276→              <Text style={styles.loginCardDesc}>Masukkan 6 digit kode yang dikirim</Text>
   277→              <TextInput
   278→                style={[styles.input, { textAlign: "center", fontSize: 24, letterSpacing: 8, fontFamily: "monospace" }]}
   279→                placeholder="000000"
   280→                value={otp}
   281→                onChangeText={(t) => setOtp(t.replace(/\D/g, "").slice(0, 6))}
   282→                keyboardType="numeric"
   283→                maxLength={6}
   284→              />
   285→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleVerifyOtp} disabled={loading}>
   286→                {loading ? <ActivityIndicator color={Colors.white} /> : <Text style={styles.loginButtonText}>Verifikasi</Text>}
   287→              </TouchableOpacity>
   288→            </>
   289→          )}
   290→
   291→          {step === "setPin" && (
   292→            <>
   293→              <Text style={styles.loginCardTitle}>Buat PIN Baru</Text>
   294→              <Text style={styles.loginCardDesc}>PIN 6 digit untuk login</Text>
   295→              <Text style={styles.label}>PIN Baru</Text>
   296→              <TextInput
   297→                style={[styles.input, { textAlign: "center", fontSize: 20, letterSpacing: 6, fontFamily: "monospace" }]}
   298→                placeholder="••••••"
   299→                value={newPin}
   300→                onChangeText={(t) => setNewPin(t.replace(/\D/g, "").slice(0, 6))}
   301→                keyboardType="numeric"
   302→                secureTextEntry
   303→                maxLength={6}
   304→              />
   305→              <Text style={styles.label}>Konfirmasi PIN</Text>
   306→              <TextInput
   307→                style={[styles.input, { textAlign: "center", fontSize: 20, letterSpacing: 6, fontFamily: "monospace" }]}
   308→                placeholder="••••••"
   309→                value={confirmPin}
   310→                onChangeText={(t) => setConfirmPin(t.replace(/\D/g, "").slice(0, 6))}
   311→                keyboardType="numeric"
   312→                secureTextEntry
   313→                maxLength={6}
   314→              />
   315→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleSetPin} disabled={loading}>
   316→                {loading ? <ActivityIndicator color={Colors.white} /> : <Text style={styles.loginButtonText}>Simpan PIN</Text>}
   317→              </TouchableOpacity>
   318→            </>
   319→          )}
   320→
   321→          <TouchableOpacity onPress={onBack} style={[styles.forgotPinBtn, { marginTop: 16 }]}>
   322→            <Text style={styles.forgotPinText}>← Kembali ke Login</Text>
   323→          </TouchableOpacity>
   324→        </View>
   325→      </ScrollView>
   326→    </SafeAreaView>
   327→  )
   328→}

   410→function SantriDetailScreen({ santri, onBack, refreshKey }: { santri: SantriData, onBack: () => void, refreshKey?: number }) {
   411→  const [detail, setDetail] = useState<SantriDetail | null>(null)
   412→  const [perkembangan, setPerkembangan] = useState<PerkembanganV2 | null>(null)
   413→  const [loading, setLoading] = useState(true)
   414→  const [error, setError] = useState<string | null>(null)
   415→  const [activeTab, setActiveTab] = useState("profil")
   416→
   417→  const loadDetail = useCallback(async () => {
   418→    try {
   419→      setLoading(true)
   420→      setError(null)
   421→      // V2: fetch both detail (V1) and perkembangan (V2) in parallel
   422→      const [detailRes, perkRes] = await Promise.all([
   423→        api.get(`/api/wali/santri/${santri.id}`),
   424→        api.get(`/api/wali/perkembangan/${santri.id}`).catch(() => null), // best-effort
   425→      ])
   426→      setDetail(detailRes.data)
   427→      setPerkembangan(perkRes?.data || null)
   428→    } catch (e: any) {
   429→      setError(e.message)
   430→    } finally {
   431→      setLoading(false)
   432→    }
   433→  }, [santri.id])
   434→
   435→  useEffect(() => { loadDetail() }, [loadDetail])
   436→
   437→  // Auto-refresh when realtime push arrives (App-level refreshKey changes)
   438→  useEffect(() => {
   439→    if (refreshKey === undefined) return
   440→    loadDetail()
   441→    // eslint-disable-next-line react-hooks/exhaustive-deps
   442→  }, [refreshKey])
   443→
   444→  // Tab groups: 6 tabs that each render multiple sections (scrollable).
   445→  // Reduces visual clutter from 12 individual tabs. Each group lists the
   446→  // section ids (matching the original activeTab checks) that it shows.
   447→  const TAB_GROUPS: Array<{ id: string; label: string; icon: string; sections: string[] }> = [
   448→    { id: "profil", label: "Profil", icon: "person", sections: ["profile", "qr"] },
   449→    { id: "akademik", label: "Akademik", icon: "school", sections: ["absensi", "nilai", "perkembangan", "tugas", "rapor"] },
   450→    { id: "ujian", label: "Ujian", icon: "file-tray-full", sections: ["ujian"] },
   451→    { id: "keuangan", label: "Keuangan", icon: "wallet", sections: ["tagihan"] },
   452→    { id: "notif", label: "Notif", icon: "notifications", sections: ["notifikasi"] },
   453→    { id: "info", label: "Info", icon: "megaphone", sections: ["pengumuman", "chat"] },
   454→  ]
   455→  const activeSections = TAB_GROUPS.find(g => g.id === activeTab)?.sections || []
   456→  const showSection = (id: string) => activeSections.includes(id)
   457→  const tabs = TAB_GROUPS
   458→
   459→  return (
   460→    <SafeAreaView style={styles.screen}>
   461→      <View style={styles.detailHeader}>
   462→        <TouchableOpacity onPress={onBack} style={styles.backButton}>
   463→          <Ionicons name="arrow-back" size={24} color={Colors.white} />
   464→        </TouchableOpacity>
   465→        <Text style={styles.detailHeaderTitle}>{santri.name}</Text>
   466→        <View style={{ width: 24 }} />
   467→      </View>
   468→
   469→      {/* Santri Info Card */}
   470→      <View style={styles.detailCard}>
   471→        <View style={styles.detailAvatar}>
   472→          <Text style={styles.detailAvatarText}>{santri.name[0]}</Text>
   473→        </View>
   474→        <View style={styles.detailInfo}>
   475→          <Text style={styles.detailName}>{santri.name}</Text>
   476→          <Text style={styles.detailNis}>NIS: {santri.nis}</Text>
   477→          <Text style={styles.detailKelas}>{santri.kelas?.name || "Tanpa Kelas"}</Text>
   478→          {detail?.activePeriod && (
   479→            <Text style={styles.detailPeriod}>{detail.activePeriod.tahunAjaran} — Semester {detail.activePeriod.semester}</Text>
   480→          )}
   481→        </View>
   482→      </View>
   483→

   794→function WaliNotifikasiSection({ token }: { token: string | null }) {
   795→  const [list, setList] = useState<any[]>([])
   796→  const [loading, setLoading] = useState(true)
   797→  const [stats, setStats] = useState<any>(null)
   798→  const [error, setError] = useState<string | null>(null)
   799→
   800→  const load = useCallback(async () => {
   801→    if (!token) return
   802→    try {
   803→      setLoading(true)
   804→      setError(null)
   805→      const res = await fetch(`${API_BASE_URL}/api/wali/notifikasi`, {
   806→        headers: { Authorization: `Bearer ${token}` },
   807→      })
   808→      const data = await res.json()
   809→      setList(data.data || [])
   810→      setStats(data.stats)
   811→    } catch (e: any) {
   812→      setError(e.message || "Terjadi kesalahan")

   814→      setLoading(false)

   849→        <View style={[styles.statusBadge, { backgroundColor: Colors.dangerLight, alignSelf: "flex-start", marginBottom: 8 }]}>

   851→        </View>

   854→        <Text style={styles.emptyTextSmall}>Tidak ada notifikasi</Text>

   929→function WaliChatRoom({ room, token, onBack }: { room: any, token: string | null, onBack: () => void }) {
   930→  const [messages, setMessages] = useState<any[]>([])
   931→  const [input, setInput] = useState("")
   932→  const [loading, setLoading] = useState(true)
   933→
   934→  const loadMessages = useCallback(async () => {
   935→    if (!token) return
   936→    try {
   937→      const res = await fetch(`${API_BASE_URL}/api/wali/chat/${room.id}/messages`, {
   938→        headers: { Authorization: `Bearer ${token}` },
   939→      })
   940→      const data = await res.json()
   941→      setMessages(data.data || [])
   942→    } catch (e: any) {
   943→      console.error("msg load error:", e.message)
   944→    } finally {
   945→      setLoading(false)
   946→    }
   947→  }, [token, room.id])
   948→
   949→  useEffect(() => { loadMessages() }, [loadMessages])
   950→
   951→  const send = async () => {
   952→    if (!input.trim() || !token) return
   953→    try {
   954→      await fetch(`${API_BASE_URL}/api/wali/chat/${room.id}/messages`, {
   955→        method: "POST",
   956→        headers: { "Content-Type": "application/json", Authorization: `Bearer ${token}` },
   957→        body: JSON.stringify({ message: input }),
   958→      })
   959→      setInput("")
   960→      loadMessages()
   961→    } catch (e: any) {
   962→      Alert.alert("Gagal", e.message)
