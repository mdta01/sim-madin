    28→const SUPABASE_URL = process.env.EXPO_PUBLIC_SUPABASE_URL || ""

    31→let supabase: any = null
    32→function getSupabase() {
    33→  if (!SUPABASE_URL || !SUPABASE_ANON_KEY) {
    34→    console.warn("[Supabase] env vars not set. Configure EXPO_PUBLIC_SUPABASE_URL and EXPO_PUBLIC_SUPABASE_ANON_KEY in eas.json")
    35→    return null
    36→  }
    37→  if (supabase) return supabase
    38→  supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY, {
    39→    auth: {
    40→      storage: {
    41→        getItem: async (k: string) => { try { return await SecureStore.getItemAsync(k) } catch { return null } },
    42→        setItem: async (k: string, v: string) => { try { await SecureStore.setItemAsync(k, v) } catch {} },
    43→        removeItem: async (k: string) => { try { await SecureStore.deleteItemAsync(k) } catch {} },
    44→      },
    45→      autoRefreshToken: true, persistSession: true,
    46→    },
    47→  })
    48→  return supabase
    49→}

    51→let authToken: string | null = null

    53→async function apiGet(path: string) {
    54→  const res = await fetch(`${API_BASE_URL}${path}`, { headers: authToken ? { Authorization: `Bearer ${authToken}` } : {} })
    55→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    56→  return res.json()
    57→}
    58→
    59→async function apiPost(path: string, body: any) {
    60→  const res = await fetch(`${API_BASE_URL}${path}`, {
    61→    method: "POST",
    62→    headers: { "Content-Type": "application/json", ...(authToken ? { Authorization: `Bearer ${authToken}` } : {}) },
    63→    body: JSON.stringify(body),
    64→  })
    65→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    66→  return res.json()
    67→}

    89→function LoginScreen({ onLogin }: { onLogin: () => void }) {
    90→  const [email, setEmail] = useState("")
    91→  const [password, setPassword] = useState("")
    92→  const [loading, setLoading] = useState(false)
    93→
    94→  const handleLogin = async () => {
    95→    if (!email || !password) { Alert.alert("Validasi", "Email dan password wajib diisi"); return }
    96→    setLoading(true)
    97→    try {
    98→      const { data, error } = await getSupabase().auth.signInWithPassword({ email, password })
    99→      if (error) throw error
   100→      authToken = data.session?.access_token || null
   101→      await SecureStore.setItemAsync("authToken", authToken || "")
   102→      onLogin()
   103→    } catch (e: any) { Alert.alert("Login Gagal", e.message) }
   104→    finally { setLoading(false) }
   105→  }
   106→
   107→  return (
   108→    <SafeAreaView style={styles.loginContainer}>
   109→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   110→      <ScrollView contentContainerStyle={styles.loginScroll}>
   111→        <View style={styles.loginLogo}>
   112→          <Ionicons name="school" size={80} color={Colors.primary} />
   113→          <Text style={styles.loginTitle}>SIM MADIN</Text>
   114→          <Text style={styles.loginSubtitle}>Guru</Text>
   115→        </View>
   116→        <View style={styles.loginCard}>
   117→          <Text style={styles.loginCardTitle}>Masuk sebagai Guru</Text>
   118→          <Text style={styles.label}>Email</Text>
   119→          <TextInput style={styles.input} value={email} onChangeText={setEmail} keyboardType="email-address" autoCapitalize="none" placeholder="guru@madrasah.test" />
   120→          <Text style={styles.label}>Password</Text>

   288→  const [selectedDate, setSelectedDate] = useState<Date>(new Date())
   289→  const [showDatePicker, setShowDatePicker] = useState(false)
   290→  const [loading, setLoading] = useState(false)
   291→  const [saving, setSaving] = useState(false)
   292→  const [error, setError] = useState<string | null>(null)
   293→  const [refreshing, setRefreshing] = useState(false)

   297→      setError(null)
   298→      const res = await apiGet("/api/admin/kelas")
   299→      setKelasList(res.data || [])
   300→      if (!selectedKelas && res.data?.length > 0) {
   301→        setSelectedKelas(res.data[0].id)
   302→      }
   303→    } catch (e: any) {
   304→      setError(e.message || "Terjadi kesalahan")
