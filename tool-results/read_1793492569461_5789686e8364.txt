    26→// For EAS Build: configured in eas.json → build.preview.env / build.production.env
    27→const SUPABASE_URL = process.env.EXPO_PUBLIC_SUPABASE_URL || ""
    28→const SUPABASE_ANON_KEY = process.env.EXPO_PUBLIC_SUPABASE_ANON_KEY || ""

    30→let supabase: any = null
    31→function getSupabase() {
    32→  if (!SUPABASE_URL || !SUPABASE_ANON_KEY) {
    33→    console.warn("[Supabase] env vars not set. Configure EXPO_PUBLIC_SUPABASE_URL and EXPO_PUBLIC_SUPABASE_ANON_KEY in eas.json")
    34→    return null
    35→  }
    36→  if (supabase) return supabase
    37→  supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY, {
    38→    auth: {
    39→      storage: {
    40→        getItem: async (k: string) => { try { return await SecureStore.getItemAsync(k) } catch { return null } },
    41→        setItem: async (k: string, v: string) => { try { await SecureStore.setItemAsync(k, v) } catch {} },
    42→        removeItem: async (k: string) => { try { await SecureStore.deleteItemAsync(k) } catch {} },
    43→      },
    44→      autoRefreshToken: true, persistSession: true,
    45→    },
    46→  })
    47→  return supabase
    48→}

    52→async function apiGet(path: string) {
    53→  const res = await fetch(`${API_BASE_URL}${path}`, {
    54→    headers: authToken ? { Authorization: `Bearer ${authToken}` } : {},
    55→  })
    56→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    57→  return res.json()
    58→}

    78→function LoginScreen({ onLogin }: { onLogin: () => void }) {
    79→  const [email, setEmail] = useState("")
    80→  const [password, setPassword] = useState("")
    81→  const [loading, setLoading] = useState(false)
    82→
    83→  const handleLogin = async () => {
    84→    setLoading(true)
    85→    try {
    86→      const { data, error } = await getSupabase().auth.signInWithPassword({ email, password })
    87→      if (error) throw error
    88→      authToken = data.session?.access_token
    89→      await SecureStore.setItemAsync("authToken", authToken || "")
    90→      onLogin()
    91→    } catch (e: any) { Alert.alert("Login Gagal", e.message) }
    92→    finally { setLoading(false) }
    93→  }
    94→

    96→    <SafeAreaView style={styles.loginContainer}>

   651→export default function App() {
   652→  const [isLoggedIn, setIsLoggedIn] = useState(false)
   653→  const [checking, setChecking] = useState(true)
   654→  // Realtime refresh tick — bumped whenever a subscribed table changes.
   655→  // Drives dashboard stats refresh on Santri/Guru/Payment events.
   656→  const [refreshKey, setRefreshKey] = useState(0)
   657→
   658→  useEffect(() => {
   659→    initNetworkMonitor()
   660→    async function check() {
   661→      const token = await SecureStore.getItemAsync("authToken")
   662→      if (token) {
   663→        authToken = token
   664→        const { data } = await getSupabase().auth.getSession()
   665→        if (data.session) { authToken = data.session.access_token; setIsLoggedIn(true) }
   666→      }
   667→      setChecking(false)
   668→    }
   669→    check()
   670→  }, [])
   671→
   672→  // ============================================================================
   673→  // Supabase Realtime subscriptions (OPTIONAL — fails silently if unavailable)
   674→  // - Santri   INSERT/UPDATE : data santri berubah → refresh dashboard stats
   675→  // - Guru     INSERT/UPDATE : data guru berubah  → refresh dashboard stats
   676→  // - Payment  INSERT/UPDATE : pembayaran baru    → refresh keuangan stats
   677→  // ============================================================================
   678→  useEffect(() => {
   679→    if (!isLoggedIn) return
   680→    let cancelled = false
   681→    const unsubs: Array<() => void> = []
   682→    ;(async () => {
   683→      try {
   684→        for (const table of ["Santri", "Guru", "Payment"] as const) {
   685→          try {
   686→            const unsub = await subscribeToTable(table, () => {
   687→              setRefreshKey(k => k + 1)
   688→            })
   689→            if (cancelled) { try { unsub() } catch {} }
   690→            else { unsubs.push(unsub) }
   691→          } catch {}
   692→        }
   693→      } catch(e) { /* realtime optional */ }
   694→    })()
   695→    return () => {
   696→      cancelled = true
   697→      unsubs.forEach(u => { try { u() } catch {} })
   698→    }
   699→  }, [isLoggedIn])
   700→
   701→  if (checking) {
   702→    return <View style={{ flex: 1, justifyContent: "center", alignItems: "center", backgroundColor: Colors.primary }}><ActivityIndicator size="large" color={Colors.white} /></View>
   703→  }
   704→
   705→  return (
   706→    <View style={{ flex: 1 }}>
   707→      <NetworkStatusIndicator />
   708→      <NavigationContainer>
   709→        <Stack.Navigator screenOptions={{ headerShown: false }}>
   710→        {isLoggedIn ? (
   711→          <>
   712→            <Stack.Screen name="Dashboard">
   713→              {({ navigation }) => <DashboardScreen navigation={navigation} refreshKey={refreshKey} onLogout={async () => { await getSupabase().auth.signOut(); await SecureStore.deleteItemAsync("authToken"); authToken = null; setIsLoggedIn(false) }} />}
   714→            </Stack.Screen>
   715→            <Stack.Screen name="MonitorSantri">
   716→              {({ navigation }) => <MonitorSantriScreen navigation={navigation} refreshKey={refreshKey} />}
