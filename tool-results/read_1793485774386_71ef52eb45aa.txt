    29→    auth: {
    30→      storage: {

    43→async function apiGet(path: string) {
    44→  const res = await fetch(`${API_BASE_URL}${path}`, { headers: authToken ? { Authorization: `Bearer ${authToken}` } : {} })
    45→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    46→  return res.json()
    47→}
    48→
    49→async function apiPost(path: string, body: any) {
    50→  const res = await fetch(`${API_BASE_URL}${path}`, {
    51→    method: "POST",
    52→    headers: { "Content-Type": "application/json", ...(authToken ? { Authorization: `Bearer ${authToken}` } : {}) },
    53→    body: JSON.stringify(body),
    54→  })
    55→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    56→  return res.json()
    57→}

   219→function PembayaranInputScreen({ navigation }: { navigation: any }) {
   220→  const [santriList, setSantriList] = useState<any[]>([])
   221→  const [tagihanList, setTagihanList] = useState<any[]>([])
   222→  const [selectedSantri, setSelectedSantri] = useState("")
   223→  const [selectedTagihan, setSelectedTagihan] = useState("")
   224→  const [amount, setAmount] = useState("")
   225→  const [method, setMethod] = useState("TUNAI")
   226→  const [loading, setLoading] = useState(false)
   227→  const [saving, setSaving] = useState(false)
   228→  const [searchText, setSearchText] = useState("")
   229→
   230→  const loadSantri = useCallback(async () => {
   231→    try {
   232→      setLoading(true)
   233→      const res = await apiGet("/api/admin/santri?status=AKTIF")
   234→      setSantriList(res.data || [])
   235→    } catch (e: any) { Alert.alert("Error", e.message) }
   236→    finally { setLoading(false) }
   237→  }, [])
   238→
   239→  const loadTagihan = useCallback(async () => {
   240→    if (!selectedSantri) { setTagihanList([]); return }
   241→    try {
   242→      const res = await apiGet(`/api/admin/tagihan?santriId=${selectedSantri}`)
   243→      const unpaid = (res.data || []).filter((t: any) => t.status !== "LUNAS")
   244→      setTagihanList(unpaid)
   245→    } catch (e: any) { Alert.alert("Error", e.message) }
   246→  }, [selectedSantri])
   247→
   248→  useEffect(() => { loadSantri() }, [loadSantri])
   249→  useEffect(() => { loadTagihan() }, [loadTagihan])
   250→
   251→  const selectTagihan = (tagihanId: string, tagihanAmount: number) => {
   252→    setSelectedTagihan(tagihanId)
   253→    setAmount(String(tagihanAmount - (tagihanList.find(t => t.id === tagihanId)?.paidAmount || 0)))
   254→  }
   255→
   256→  const handleSave = async () => {
   257→    if (!selectedSantri || !amount || Number(amount) <= 0) {
   258→      Alert.alert("Validasi", "Pilih santri dan isi nominal")
   259→      return
   260→    }
   261→    setSaving(true)
   262→    try {
   263→      await apiPost("/api/admin/pembayaran", {
   264→        santriId: selectedSantri,
   265→        tagihanId: selectedTagihan || undefined,
   266→        amount: Number(amount),
   267→        paymentMethod: method,
   268→      })
   269→      Alert.alert("Berhasil", `Pembayaran ${formatRupiah(Number(amount))} tercatat`)
   270→      navigation.goBack()
   271→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   272→    finally { setSaving(false) }
   273→  }
   274→
   275→  const filteredSantri = santriList.filter(s =>
   276→    !searchText || s.name.toLowerCase().includes(searchText.toLowerCase()) || s.nis.includes(searchText)
   277→  )
   278→
   279→  return (
   280→    <SafeAreaView style={styles.screen}>
   281→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   282→      <View style={styles.header}>
   283→        <TouchableOpacity onPress={() => navigation.goBack()} style={styles.backButton}>
   284→          <Ionicons name="arrow-back" size={24} color={Colors.white} />
   285→        </TouchableOpacity>
   286→        <Text style={styles.headerTitle}>Input Pembayaran</Text>
   287→        <View style={{ width: 24 }} />
   288→      </View>
   289→
   290→      <ScrollView style={styles.content}>
   291→        {/* Search Santri */}
   292→        <Text style={styles.label}>Cari Santri</Text>
   293→        <View style={styles.searchContainer}>
   294→          <Ionicons name="search" size={20} color={Colors.textMuted} style={{ marginRight: 8 }} />
   295→          <TextInput
   296→            style={styles.searchInput}
   297→            placeholder="Nama atau NIS..."
   298→            value={searchText}
   299→            onChangeText={setSearchText}
   300→          />
   301→        </View>
   302→
   303→        {/* Santri List */}
   304→        {loading ? (
   305→          <ActivityIndicator size="large" color={Colors.primary} style={{ marginTop: 20 }} />
   306→        ) : (
   307→          <View>
   308→            {filteredSantri.slice(0, 5).map(s => (
   309→              <TouchableOpacity
   310→                key={s.id}
   311→                style={[styles.santriOption, selectedSantri === s.id && styles.santriOptionActive]}
   312→                onPress={() => { setSelectedSantri(s.id); setSelectedTagihan(""); setAmount("") }}
   313→              >
   314→                <View style={{ flex: 1 }}>
   315→                  <Text style={styles.santriOptionName}>{s.name}</Text>
   316→                  <Text style={styles.santriOptionNis}>{s.nis} • {s.kelas?.name}</Text>
   317→                </View>
   318→                {selectedSantri === s.id && <Ionicons name="checkmark-circle" size={24} color={Colors.primary} />}
   319→              </TouchableOpacity>
   320→            ))}
   321→            {filteredSantri.length > 5 && (
   322→              <Text style={styles.moreText}>+{filteredSantri.length - 5} santri lainnya (ketik untuk filter)</Text>
   323→            )}
   324→          </View>
   325→        )}
   326→
   327→        {/* Tagihan List (if santri selected) */}
   328→        {selectedSantri && tagihanList.length > 0 && (
   329→          <View>
   330→            <Text style={styles.label}>Tagihan Belum Lunas</Text>
   331→            {tagihanList.map(t => (
   332→              <TouchableOpacity
   333→                key={t.id}
   334→                style={[styles.tagihanOption, selectedTagihan === t.id && styles.tagihanOptionActive]}
   335→                onPress={() => selectTagihan(t.id, t.amount)}
   336→              >
   337→                <View style={{ flex: 1 }}>
   338→                  <Text style={styles.tagihanOptionName}>{t.jenisTagihan?.name}</Text>
   339→                  <Text style={styles.tagihanOptionDesc}>
   340→                    Total: {formatRupiah(t.amount)} • Sisa: {formatRupiah(t.amount - t.paidAmount)}
   341→                  </Text>
   342→                </View>
   343→                {selectedTagihan === t.id && <Ionicons name="checkmark-circle" size={24} color={Colors.primary} />}
   344→              </TouchableOpacity>
   345→            ))}
   346→          </View>
   347→        )}
   348→
   349→        {/* Amount */}
   350→        <Text style={styles.label}>Nominal Pembayaran</Text>
   351→        <View style={styles.amountContainer}>
   352→          <Text style={styles.amountPrefix}>Rp</Text>

   359→          />
   360→        </View>

   362→        {/* Method */}
   363→        <Text style={styles.label}>Metode Pembayaran</Text>

   375→
   376→        {/* Save */}

   378→          style={[styles.saveButton, saving && { opacity: 0.6 }]}

   384→              <Ionicons name="save" size={20} color={Colors.white} />

   386→            </>
   387→          )}
   388→        </TouchableOpacity>
   389→
   390→        <View style={{ height: 40 }} />
   391→      </ScrollView>
   392→    </SafeAreaView>
   393→  )

  1213→export default function App() {
  1214→  const [isLoggedIn, setIsLoggedIn] = useState(false)
  1215→  const [checking, setChecking] = useState(true)
  1216→  // Realtime refresh tick — bumped whenever a subscribed table changes.
  1217→  // Passed to DashboardScreen so it can re-fetch on Payment/Tagihan events.
  1218→  const [refreshKey, setRefreshKey] = useState(0)
  1219→
  1220→  useEffect(() => {
  1221→    async function check() {
  1222→      const token = await SecureStore.getItemAsync("authToken")
  1223→      if (token) {
  1224→        authToken = token
  1225→        const { data } = await getSupabase().auth.getSession()
  1226→        if (data.session) { authToken = data.session.access_token; setIsLoggedIn(true) }
  1227→      }
  1228→      setChecking(false)
  1229→    }
  1230→    check()
  1231→  }, [])
  1232→
  1233→  // ============================================================================
  1234→  // Supabase Realtime subscriptions (OPTIONAL — fails silently if unavailable)
  1235→  // - Payment INSERT  : new pembayaran recorded → refresh dashboard
  1236→  // - Tagihan  UPDATE : status BELUM_BAYAR → LUNAS → refresh tunggakan/tagihan
  1237→  // ============================================================================
  1238→  useEffect(() => {
  1239→    if (!isLoggedIn) return
  1240→    let cancelled = false
  1241→    const unsubs: Array<() => void> = []
  1242→    ;(async () => {
  1243→      try {
  1244→        for (const table of ["Payment", "Tagihan"] as const) {
  1245→          try {
  1246→            const unsub = await subscribeToTable(table, () => {
  1247→              setRefreshKey(k => k + 1)
  1248→            })
  1249→            if (cancelled) { try { unsub() } catch {} }
  1250→            else { unsubs.push(unsub) }
  1251→          } catch {}
  1252→        }
  1253→      } catch(e) { /* realtime optional */ }
  1254→    })()
  1255→    return () => {
  1256→      cancelled = true
  1257→      unsubs.forEach(u => { try { u() } catch {} })
  1258→    }
  1259→  }, [isLoggedIn])
  1260→
  1261→  if (checking) {
  1262→    return <View style={{ flex: 1, justifyContent: "center", alignItems: "center", backgroundColor: Colors.primary }}><ActivityIndicator size="large" color={Colors.white} /></View>
  1263→  }
  1264→
  1265→  return (
  1266→    <NavigationContainer>
  1267→      <Stack.Navigator screenOptions={{ headerShown: false }}>
  1268→        {isLoggedIn ? (
  1269→          <>
  1270→            <Stack.Screen name="Dashboard">
  1271→              {({ navigation }) => (
  1272→                <DashboardScreen
  1273→                  navigation={navigation}
  1274→                  onLogout={async () => { await getSupabase().auth.signOut(); await SecureStore.deleteItemAsync("authToken"); authToken = null; setIsLoggedIn(false) }}
  1275→                />
  1276→              )}
  1277→            </Stack.Screen>
  1278→            <Stack.Screen name="PembayaranInput" component={PembayaranInputScreen} />
  1279→            <Stack.Screen name="TunggakanList" component={TunggakanListScreen} />
  1280→            <Stack.Screen name="TagihanList" component={TagihanListScreen} />
