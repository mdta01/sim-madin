    26→function getSupabase() {
    27→  if (supabase) return supabase
    28→  supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY, {
    29→    auth: {
    30→      storage: {
    31→        getItem: async (k: string) => { try { return await SecureStore.getItemAsync(k) } catch { return null } },
    32→        setItem: async (k: string, v: string) => { try { await SecureStore.setItemAsync(k, v) } catch {} },
    33→        removeItem: async (k: string) => { try { await SecureStore.deleteItemAsync(k) } catch {} },
    34→      },
    35→      autoRefreshToken: true, persistSession: true,
    36→    },
    37→  })
    38→  return supabase

    43→async function apiGet(path: string) {
    44→  const res = await fetch(`${API_BASE_URL}${path}`, {
    45→    headers: authToken ? { Authorization: `Bearer ${authToken}` } : {},
    46→  })
    47→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    48→  return res.json()

   588→export default function App() {
   589→  const [isLoggedIn, setIsLoggedIn] = useState(false)
   590→  const [checking, setChecking] = useState(true)
   591→  // Realtime refresh tick — bumped whenever a subscribed table changes.
   592→  // Drives dashboard stats refresh on Santri/Guru/Payment events.
   593→  const [refreshKey, setRefreshKey] = useState(0)
   594→
   595→  useEffect(() => {
   596→    async function check() {
   597→      const token = await SecureStore.getItemAsync("authToken")
   598→      if (token) {
   599→        authToken = token
   600→        const { data } = await getSupabase().auth.getSession()
   601→        if (data.session) { authToken = data.session.access_token; setIsLoggedIn(true) }
   602→      }
   603→      setChecking(false)
   604→    }
   605→    check()
   606→  }, [])
   607→
   608→  // ============================================================================
   609→  // Supabase Realtime subscriptions (OPTIONAL — fails silently if unavailable)
   610→  // - Santri   INSERT/UPDATE : data santri berubah → refresh dashboard stats
   611→  // - Guru     INSERT/UPDATE : data guru berubah  → refresh dashboard stats
   612→  // - Payment  INSERT/UPDATE : pembayaran baru    → refresh keuangan stats
   613→  // ============================================================================
   614→  useEffect(() => {
   615→    if (!isLoggedIn) return
   616→    let cancelled = false
   617→    const unsubs: Array<() => void> = []
   618→    ;(async () => {
   619→      try {
   620→        for (const table of ["Santri", "Guru", "Payment"] as const) {
   621→          try {
   622→            const unsub = await subscribeToTable(table, () => {
   623→              setRefreshKey(k => k + 1)
   624→            })
   625→            if (cancelled) { try { unsub() } catch {} }
   626→            else { unsubs.push(unsub) }
   627→          } catch {}
   628→        }
   629→      } catch(e) { /* realtime optional */ }
   630→    })()
   631→    return () => {
   632→      cancelled = true
   633→      unsubs.forEach(u => { try { u() } catch {} })
   634→    }
   635→  }, [isLoggedIn])
   636→
   637→  if (checking) {
   638→    return <View style={{ flex: 1, justifyContent: "center", alignItems: "center", backgroundColor: Colors.primary }}><ActivityIndicator size="large" color={Colors.white} /></View>
   639→  }
   640→
   641→  return (
   642→    <NavigationContainer>
   643→      <Stack.Navigator screenOptions={{ headerShown: false }}>
   644→        {isLoggedIn ? (
   645→          <>
   646→            <Stack.Screen name="Dashboard">
   647→              {({ navigation }) => <DashboardScreen navigation={navigation} onLogout={async () => { await getSupabase().auth.signOut(); await SecureStore.deleteItemAsync("authToken"); authToken = null; setIsLoggedIn(false) }} />}
   648→            </Stack.Screen>
   649→            <Stack.Screen name="MonitorSantri" component={MonitorSantriScreen} />
   650→            <Stack.Screen name="MonitorGuru" component={MonitorGuruScreen} />
   651→            <Stack.Screen name="MonitorAbsensi" component={MonitorAbsensiScreen} />
   652→            <Stack.Screen name="MonitorKeuangan" component={MonitorKeuanganScreen} />
   653→            <Stack.Screen name="MonitorPengumuman" component={MonitorPengumumanScreen} />
   654→            <Stack.Screen name="MonitorNilai" component={MonitorNilaiScreen} />
   655→            <Stack.Screen name="DashboardKepala" component={DashboardKepalaScreen} />
   656→          </>
   657→        ) : (
   658→          <Stack.Screen name="Login">{() => <LoginScreen onLogin={() => setIsLoggedIn(true)} />}</Stack.Screen>
   659→        )}
   660→      </Stack.Navigator>
