   217→   217→function SantriForm({ open, onOpenChange, editing, kelasList, onSuccess }: {
   218→   218→  open: boolean
   219→   219→  onOpenChange: (v: boolean) => void
   220→   220→  editing: any | null
   221→   221→  kelasList: any[]
   222→   222→  onSuccess: () => void
   223→   223→}) {
   224→   224→  const [form, setForm] = useState<any>({})
   225→   225→  const [waliList, setWaliList] = useState<any[]>([])
   226→   226→  const [taList, setTaList] = useState<any[]>([])
   227→   227→  const [saving, setSaving] = useState(false)
   228→   228→  const { toast } = useToast()
   229→   229→
   230→   230→  useEffect(() => {
   231→   231→    if (open) {
   232→   232→      apiGet("/api/admin/wali").then((r) => setWaliList(r.data)).catch(() => {})
   233→   233→      apiGet("/api/admin/tahun-ajaran").then((r) => setTaList(r.data)).catch(() => {})
   234→   234→    }
   235→   235→  }, [open])
   236→   236→
   237→   237→  useEffect(() => {
   238→   238→    if (editing) {
   239→   239→      setForm({
   240→   240→        ...editing,
   241→   241→        birthDate: editing.birthDate ? new Date(editing.birthDate).toISOString().split("T")[0] : "",
   242→   242→        entryDate: editing.entryDate ? new Date(editing.entryDate).toISOString().split("T")[0] : "",
   243→   243→      })
   244→   244→    } else {
   245→   245→      setForm({ gender: "L", status: "AKTIF" })
   246→   246→    }
   247→   247→  }, [editing, open])
   248→   248→
   249→   249→  const handleSave = async () => {
   250→   250→    // Saat create, NIS opsional (auto-generate). Saat edit, NIS wajib.
   251→   251→    if (editing && !form.nis) {
   252→   252→      toast({ title: "Validasi gagal", description: "NIS wajib diisi saat edit", variant: "destructive" })
   253→   253→      return
   254→   254→    }
   255→   255→    if (!form.name || !form.gender) {
   256→   256→      toast({ title: "Validasi gagal", description: "Nama dan Jenis Kelamin wajib diisi", variant: "destructive" })
   257→   257→      return
   258→   258→    }
   259→   259→    // Jika createNewWali aktif, wajib isi nama wali
   260→   260→    if (form.createNewWali && !form.waliName) {
   261→   261→      toast({ title: "Validasi gagal", description: "Nama wali wajib diisi jika membuat wali baru", variant: "destructive" })
   262→   262→      return
   263→   263→    }
   264→   264→    setSaving(true)
   265→   265→    try {
   266→   266→      if (editing) {
   267→   267→        await apiPut(`/api/admin/santri/${editing.id}`, form)
   268→   268→        toast({ title: "Berhasil", description: "Data santri diperbarui" })
   269→   269→      } else {
   270→   270→        const result = await apiPost<any>("/api/admin/santri", form)
   271→   271→        // Tampilkan info NIS + aktivasi wali
   272→   272→        const nisInfo = result.nis ? `NIS: ${result.nis}` : ""
   273→   273→        let waliInfo = ""
   274→   274→        if (result.waliActivation) {
   275→   275→          const wa = result.waliActivation
   276→   276→          waliInfo = ` | Wali: ${wa.waliName}`
   277→   277→          if (wa.otpCode) {
   278→   278→            waliInfo += ` | OTP: ${wa.otpCode} (dev mode)`
   279→   279→          }
   280→   280→          waliInfo += ` | Aktivasi: ${wa.activationUrl}`
   281→   281→        }
   282→   282→        toast({
   283→   283→          title: "✅ Santri ditambahkan",
   284→   284→          description: `${nisInfo}${waliInfo}`.trim(),
   285→   285→          duration: 10000, // tampilkan lebih lama agar bisa baca OTP
   286→   286→        })
   287→   287→      }
   288→   288→      onOpenChange(false)
   289→   289→      onSuccess()
   290→   290→    } catch (e: any) {
   291→   291→      toast({ title: "Gagal", description: e.message, variant: "destructive" })
   292→   292→    } finally {
   293→   293→      setSaving(false)
   294→   294→    }
   295→   295→  }
   296→   296→
   297→   297→  return (
   298→   298→    <Dialog open={open} onOpenChange={onOpenChange}>
   299→   299→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   300→   300→        <DialogHeader>
   301→   301→          <DialogTitle>{editing ? "Edit Santri" : "Tambah Santri Baru"}</DialogTitle>
   302→   302→        </DialogHeader>
   303→   303→        <div className="mb-3 pb-3 border-b">
   304→   304→          <Label className="text-xs mb-2 block">Foto Profil</Label>
   305→   305→          <PhotoUpload
   306→   306→            value={form.photoUrl}
   307→   307→            onChange={(url) => setForm({ ...form, photoUrl: url })}
   308→   308→            name={form.name}
   309→   309→            prefix="santri"
   310→   310→          />
   311→   311→        </div>
   312→   312→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 py-2">
   313→   313→          <div className="space-y-1">
   314→   314→            <Label className="text-xs">NIS {editing ? "*" : "(kosongkan untuk auto-generate)"}</Label>
   315→   315→            <Input
   316→   316→              value={form.nis || ""}
   317→   317→              onChange={(e) => setForm({ ...form, nis: e.target.value })}
   318→   318→              placeholder={editing ? "Wajib diisi" : "Auto: YYNNNNNN (mis. 26000001)"}
   319→   319→              disabled={!!editing}
   320→   320→            />
   321→   321→            {!editing && (
   322→   322→              <p className="text-[10px] text-muted-foreground">
   323→   323→                💡 NIS = identitas permanen santri. Format: 2 digit tahun + 6 digit urut.
   324→   324→              </p>
   325→   325→            )}
   326→   326→          </div>
   327→   327→          <div className="space-y-1">
   328→   328→            <Label className="text-xs">NISN</Label>
   329→   329→            <Input value={form.nisn || ""} onChange={(e) => setForm({ ...form, nisn: e.target.value })} />
   330→   330→          </div>
   331→   331→          <div className="space-y-1 sm:col-span-2">
   332→   332→            <Label className="text-xs">Nama Lengkap *</Label>
   333→   333→            <Input value={form.name || ""} onChange={(e) => setForm({ ...form, name: e.target.value })} />
   334→   334→          </div>
   335→   335→          <div className="space-y-1">
   336→   336→            <Label className="text-xs">Jenis Kelamin *</Label>

   338→   338→              <SelectTrigger><SelectValue /></SelectTrigger>
   339→   339→              <SelectContent>
