    33→function SantriList() {
    34→  const [data, setData] = useState<any[]>([])
    35→  const [kelasList, setKelasList] = useState<any[]>([])
    36→  const [loading, setLoading] = useState(true)
    37→  const [error, setError] = useState<string | null>(null)
    38→  const [search, setSearch] = useState("")
    39→  const [kelasFilter, setKelasFilter] = useState("")
    40→  const [statusFilter, setStatusFilter] = useState("")
    41→  const [dialogOpen, setDialogOpen] = useState(false)
    42→  const [editing, setEditing] = useState<any | null>(null)
    43→  const { toast } = useToast()
    44→  const selectEntity = useAdminStore((s) => s.selectEntity)
    45→
    46→  const load = useCallback(async () => {
    47→    try {
    48→      setLoading(true)
    49→      setError(null)
    50→      const params: any = {}
    51→      if (search) params.search = search
    52→      if (kelasFilter) params.kelasId = kelasFilter
    53→      if (statusFilter) params.status = statusFilter
    54→      const [res, kelasRes] = await Promise.all([
    55→        apiGet("/api/admin/santri", params),
    56→        apiGet("/api/admin/kelas"),
    57→      ])
    58→      setData(res.data)
    59→      setKelasList(kelasRes.data)
    60→    } catch (e: any) {
    61→      setError(e.message)
    62→    } finally {
    63→      setLoading(false)
    64→    }
    65→  }, [search, kelasFilter, statusFilter])
    66→
    67→  useEffect(() => { load() }, [load])
    68→
    69→  const handleDelete = async (id: string) => {
    70→    if (!confirm("Yakin ingin menghapus data santri ini? (soft delete)")) return
    71→    try {
    72→      await apiDelete(`/api/admin/santri/${id}`)
    73→      toast({ title: "Berhasil", description: "Santri telah dihapus" })
    74→      load()
    75→    } catch (e: any) {
    76→      toast({ title: "Gagal", description: e.message, variant: "destructive" })
    77→    }
    78→  }
    79→
    80→  return (
    81→    <div>
    82→      <PageHeader
    83→        title="Data Santri"
    84→        description="Kelola seluruh data santri madrasah"
    85→        icon={<GraduationCap className="h-5 w-5" />}
    86→        actions={
    87→          <Button onClick={() => { setEditing(null); setDialogOpen(true) }}>
    88→            <Plus className="h-4 w-4 mr-1" /> Tambah Santri
    89→          </Button>
    90→        }
    91→      />
    92→      <DataTableToolbar
    93→        searchValue={search}
    94→        onSearchChange={setSearch}
    95→        searchPlaceholder="Cari nama, NIS, NISN..."
    96→        filters={
    97→          <>
    98→            <Select value={kelasFilter} onValueChange={(v) => setKelasFilter(v === "all" ? "" : v)}>
    99→              <SelectTrigger className="w-[180px] h-9"><SelectValue placeholder="Semua Kelas" /></SelectTrigger>
   100→              <SelectContent>
   101→                <SelectItem value="all">Semua Kelas</SelectItem>
   102→                {kelasList.map((k) => <SelectItem key={k.id} value={k.id}>{k.name}</SelectItem>)}
   103→              </SelectContent>
   104→            </Select>
   105→            <Select value={statusFilter} onValueChange={(v) => setStatusFilter(v === "all" ? "" : v)}>
   106→              <SelectTrigger className="w-[140px] h-9"><SelectValue placeholder="Status" /></SelectTrigger>
   107→              <SelectContent>
   108→                <SelectItem value="all">Semua Status</SelectItem>
   109→                <SelectItem value="AKTIF">Aktif</SelectItem>
   110→                <SelectItem value="LULUS">Lulus</SelectItem>
   111→                <SelectItem value="PINDAH">Pindah</SelectItem>
   112→                <SelectItem value="KELUAR">Keluar</SelectItem>
   113→              </SelectContent>
   114→            </Select>
   115→          </>
   116→        }
   117→      />
   118→
   119→      <Card>
   120→        <CardContent className="p-0">
   121→          {loading ? <LoadingState /> :
   122→            error ? <ErrorState message={error} onRetry={load} /> :
   123→            data.length === 0 ? <EmptyState icon={<GraduationCap className="h-5 w-5" />} title="Belum ada data santri" description="Tambahkan data santri pertama Anda" /> :
   124→            <div className="overflow-x-auto">
   125→              <Table>
   126→                <TableHeader>
   127→                  <TableRow>
   128→                    <TableHead className="min-w-[200px]">Santri</TableHead>
   129→                    <TableHead>NIS</TableHead>
   130→                    <TableHead>Kelas</TableHead>
   131→                    <TableHead>L/P</TableHead>
   132→                    <TableHead>Wali</TableHead>
   133→                    <TableHead>Status Wali</TableHead>
   134→                    <TableHead>Status</TableHead>
   135→                    <TableHead className="text-right">Aksi</TableHead>
   136→                  </TableRow>
   137→                </TableHeader>
   138→                <TableBody>
   139→                  {data.map((s) => (
   140→                    <TableRow key={s.id} className="cursor-pointer hover:bg-muted/50" onClick={() => selectEntity(s.id)}>
   141→                      <TableCell>
   142→                        <div className="flex items-center gap-2">
   143→                          <Avatar className="h-8 w-8">
   144→                            {s.photoUrl ? (
   145→                              <img src={s.photoUrl} alt={s.name} className="h-full w-full object-cover" />
   146→                            ) : (
   147→                              <AvatarFallback className="bg-primary/10 text-primary text-xs">{s.name[0]}</AvatarFallback>
   148→                            )}
   149→                          </Avatar>
   150→                          <div>
   151→                            <p className="font-medium text-sm">{s.name}</p>
   152→                            <p className="text-[11px] text-muted-foreground">{s.birthPlace}</p>
   153→                          </div>
   154→                        </div>
   155→                      </TableCell>
   156→                      <TableCell className="font-mono text-xs">{s.nis}</TableCell>
   157→                      <TableCell>{s.kelas?.name || "-"}</TableCell>
   158→                      <TableCell><Badge variant="outline">{s.gender}</Badge></TableCell>
   159→                      <TableCell className="text-xs">
   160→                        <p>{s.wali?.name || "-"}</p>
   161→                        <p className="text-muted-foreground">{s.wali?.phone}</p>
   162→                      </TableCell>
   163→                      <TableCell>
   164→                        {s.wali ? (
   165→                          <Badge
   166→                            variant="outline"
   167→                            className={
   168→                              s.wali.activationStatus === "ACTIVE" ? "bg-emerald-100 text-emerald-700 border-emerald-200" :
   169→                              s.wali.activationStatus === "PENDING" ? "bg-amber-100 text-amber-700 border-amber-200" :
   170→                              s.wali.activationStatus === "PENDING_NO_PHONE" ? "bg-orange-100 text-orange-700 border-orange-200" :
   171→                              s.wali.activationStatus === "SUSPENDED" ? "bg-red-100 text-red-700 border-red-200" :
   172→                              ""
   173→                            }
   174→                            title={s.wali.activationStatus}
   175→                          >
   176→                            {s.wali.activationStatus === "ACTIVE" ? "✓ Aktif" :
   177→                              s.wali.activationStatus === "PENDING" ? "⏳ Pending" :
   178→                              s.wali.activationStatus === "PENDING_NO_PHONE" ? "⚠️ No WA" :
   179→                              s.wali.activationStatus === "SUSPENDED" ? "🚫 Suspended" :
   180→                              s.wali.activationStatus}
   181→                          </Badge>
   182→                        ) : (
   183→                          <span className="text-xs text-muted-foreground">-</span>
   184→                        )}
   185→                      </TableCell>
   186→                      <TableCell><StatusBadge status={s.status} /></TableCell>
   187→                      <TableCell className="text-right" onClick={(e) => e.stopPropagation()}>
   188→                        <div className="flex justify-end gap-1">
   189→                          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => { setEditing(s); setDialogOpen(true) }}>
   190→                            <Pencil className="h-3.5 w-3.5" />
   191→                          </Button>
   192→                          <Button variant="ghost" size="icon" className="h-8 w-8 text-destructive" onClick={() => handleDelete(s.id)}>
   193→                            <Trash2 className="h-3.5 w-3.5" />
   194→                          </Button>
   195→                        </div>
   196→                      </TableCell>
   197→                    </TableRow>
   198→                  ))}
   199→                </TableBody>
   200→              </Table>
   201→            </div>
   202→          }
   203→        </CardContent>
   204→      </Card>
   205→
   206→      <SantriForm
   207→        open={dialogOpen}
   208→        onOpenChange={setDialogOpen}
   209→        editing={editing}
   210→        kelasList={kelasList}
   211→        onSuccess={load}
   212→      />
   213→    </div>
   214→  )
   215→}

   217→function SantriForm({ open, onOpenChange, editing, kelasList, onSuccess }: {
   218→  open: boolean
   219→  onOpenChange: (v: boolean) => void
   220→  editing: any | null
   221→  kelasList: any[]
   222→  onSuccess: () => void
   223→}) {
   224→  const [form, setForm] = useState<any>({})
   225→  const [waliList, setWaliList] = useState<any[]>([])
   226→  const [taList, setTaList] = useState<any[]>([])
   227→  const [saving, setSaving] = useState(false)
   228→  const { toast } = useToast()
   229→
   230→  useEffect(() => {
   231→    if (open) {
   232→      apiGet("/api/admin/wali").then((r) => setWaliList(r.data)).catch(() => {})
   233→      apiGet("/api/admin/tahun-ajaran").then((r) => setTaList(r.data)).catch(() => {})
   234→    }
   235→  }, [open])
   236→
   237→  useEffect(() => {
   238→    if (editing) {
   239→      setForm({
   240→        ...editing,
   241→        birthDate: editing.birthDate ? new Date(editing.birthDate).toISOString().split("T")[0] : "",
   242→        entryDate: editing.entryDate ? new Date(editing.entryDate).toISOString().split("T")[0] : "",
   243→      })
   244→    } else {
   245→      setForm({ gender: "L", status: "AKTIF" })
   246→    }
   247→  }, [editing, open])
   248→
   249→  const handleSave = async () => {
   250→    // Saat create, NIS opsional (auto-generate). Saat edit, NIS wajib.
   251→    if (editing && !form.nis) {
   252→      toast({ title: "Validasi gagal", description: "NIS wajib diisi saat edit", variant: "destructive" })
   253→      return
   254→    }
   255→    if (!form.name || !form.gender) {
   256→      toast({ title: "Validasi gagal", description: "Nama dan Jenis Kelamin wajib diisi", variant: "destructive" })
   257→      return
   258→    }
   259→    // Jika createNewWali aktif, wajib isi nama wali
   260→    if (form.createNewWali && !form.waliName) {
   261→      toast({ title: "Validasi gagal", description: "Nama wali wajib diisi jika membuat wali baru", variant: "destructive" })
   262→      return
   263→    }
   264→    setSaving(true)
   265→    try {
   266→      if (editing) {
   267→        await apiPut(`/api/admin/santri/${editing.id}`, form)
   268→        toast({ title: "Berhasil", description: "Data santri diperbarui" })
   269→      } else {
   270→        const result = await apiPost<any>("/api/admin/santri", form)
   271→        // Tampilkan info NIS + aktivasi wali
   272→        const nisInfo = result.nis ? `NIS: ${result.nis}` : ""
   273→        let waliInfo = ""
   274→        if (result.waliActivation) {
   275→          const wa = result.waliActivation
   276→          waliInfo = ` | Wali: ${wa.waliName}`
   277→          if (wa.otpCode) {
   278→            waliInfo += ` | OTP: ${wa.otpCode} (dev mode)`
   279→          }
   280→          waliInfo += ` | Aktivasi: ${wa.activationUrl}`
   281→        }
   282→        toast({
   283→          title: "✅ Santri ditambahkan",
   284→          description: `${nisInfo}${waliInfo}`.trim(),
   285→          duration: 10000, // tampilkan lebih lama agar bisa baca OTP
   286→        })
   287→      }
   288→      onOpenChange(false)
   289→      onSuccess()
   290→    } catch (e: any) {
   291→      toast({ title: "Gagal", description: e.message, variant: "destructive" })
   292→    } finally {
   293→      setSaving(false)
   294→    }
   295→  }
   296→
   297→  return (
   298→    <Dialog open={open} onOpenChange={onOpenChange}>
   299→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   300→        <DialogHeader>
   301→          <DialogTitle>{editing ? "Edit Santri" : "Tambah Santri Baru"}</DialogTitle>
   302→        </DialogHeader>
   303→        <div className="mb-3 pb-3 border-b">
   304→          <Label className="text-xs mb-2 block">Foto Profil</Label>
   305→          <PhotoUpload
   306→            value={form.photoUrl}
   307→            onChange={(url) => setForm({ ...form, photoUrl: url })}
   308→            name={form.name}
   309→            prefix="santri"
   310→          />
   311→        </div>
   312→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 py-2">
   313→          <div className="space-y-1">
   314→            <Label className="text-xs">NIS {editing ? "*" : "(kosongkan untuk auto-generate)"}</Label>
   315→            <Input
   316→              value={form.nis || ""}
   317→              onChange={(e) => setForm({ ...form, nis: e.target.value })}
   318→              placeholder={editing ? "Wajib diisi" : "Auto: YYNNNNNN (mis. 26000001)"}
   319→              disabled={!!editing}
   320→            />
   321→            {!editing && (
   322→              <p className="text-[10px] text-muted-foreground">
   323→                💡 NIS = identitas permanen santri. Format: 2 digit tahun + 6 digit urut.
   324→              </p>
   325→            )}
   326→          </div>
   327→          <div className="space-y-1">
   328→            <Label className="text-xs">NISN</Label>
   329→            <Input value={form.nisn || ""} onChange={(e) => setForm({ ...form, nisn: e.target.value })} />
   330→          </div>
   331→          <div className="space-y-1 sm:col-span-2">
   332→            <Label className="text-xs">Nama Lengkap *</Label>
   333→            <Input value={form.name || ""} onChange={(e) => setForm({ ...form, name: e.target.value })} />
   334→          </div>
   335→          <div className="space-y-1">
   336→            <Label className="text-xs">Jenis Kelamin *</Label>
   337→            <Select value={form.gender} onValueChange={(v) => setForm({ ...form, gender: v })}>
   338→              <SelectTrigger><SelectValue /></SelectTrigger>
   339→              <SelectContent>
   340→                <SelectItem value="L">Laki-laki</SelectItem>
   341→                <SelectItem value="P">Perempuan</SelectItem>
   342→              </SelectContent>
   343→            </Select>
   344→          </div>
   345→          <div className="space-y-1">
   346→            <Label className="text-xs">Tempat Lahir</Label>
   347→            <Input value={form.birthPlace || ""} onChange={(e) => setForm({ ...form, birthPlace: e.target.value })} />
   348→          </div>
   349→          <div className="space-y-1">
   350→            <Label className="text-xs">Tanggal Lahir</Label>
   351→            <Input type="date" value={form.birthDate || ""} onChange={(e) => setForm({ ...form, birthDate: e.target.value })} />
   352→          </div>
   353→          <div className="space-y-1">
   354→            <Label className="text-xs">No. HP Santri (opsional)</Label>
   355→            <Input value={form.phone || ""} onChange={(e) => setForm({ ...form, phone: e.target.value })} />
   356→          </div>
   357→          <div className="space-y-1">
   358→            <Label className="text-xs">Kelas</Label>
   359→            <Select value={form.kelasId} onValueChange={(v) => setForm({ ...form, kelasId: v })}>
   360→              <SelectTrigger><SelectValue placeholder="Pilih Kelas" /></SelectTrigger>
   361→              <SelectContent>
   362→                {kelasList.map((k) => <SelectItem key={k.id} value={k.id}>{k.name}</SelectItem>)}
   363→              </SelectContent>
   364→            </Select>
   365→          </div>
   366→
   367→          {/* WALI SECTION — pilih existing ATAU buat baru */}
   368→          <div className="sm:col-span-2 mt-2 pt-3 border-t">
   369→            <div className="flex items-center justify-between mb-2">
   370→              <Label className="text-xs font-medium">Wali Santri</Label>
   371→              <Button
   372→                type="button"
   373→                variant="outline"
   374→                size="sm"
   375→                onClick={() => setForm({ ...form, createNewWali: !form.createNewWali, waliId: form.createNewWali ? form.waliId : null })}
   376→              >
   377→                {form.createNewWali ? "← Pilih Wali Existing" : "+ Buat Wali Baru"}
   378→              </Button>
   379→            </div>
   380→
   381→            {!form.createNewWali ? (
   382→              <Select value={form.waliId} onValueChange={(v) => setForm({ ...form, waliId: v })}>
   383→                <SelectTrigger><SelectValue placeholder="Pilih Wali Existing" /></SelectTrigger>
   384→                <SelectContent>
   385→                  {waliList.map((w) => (
   386→                    <SelectItem key={w.id} value={w.id}>
   387→                      {w.name} {w.phone ? `(${w.phone})` : "(no WA)"} {w.activationStatus === "ACTIVE" ? "✓" : "⏳"}
   388→                    </SelectItem>
   389→                  ))}
   390→                </SelectContent>
   391→              </Select>
   392→            ) : (
   393→              <div className="grid grid-cols-1 sm:grid-cols-3 gap-2 p-3 rounded border bg-muted/20">
   394→                <div className="space-y-1">
   395→                  <Label className="text-xs">Nama Wali *</Label>
   396→                  <Input
   397→                    value={form.waliName || ""}
   398→                    onChange={(e) => setForm({ ...form, waliName: e.target.value })}
   399→                    placeholder="Mis. Bapak Ahmad"
   400→                  />
   401→                </div>
   402→                <div className="space-y-1">
   403→                  <Label className="text-xs">No. WhatsApp Wali</Label>
   404→                  <Input
   405→                    value={form.waliPhone || ""}
   406→                    onChange={(e) => setForm({ ...form, waliPhone: e.target.value })}
   407→                    placeholder="08xxx atau 628xxx"
   408→                  />
   409→                  <p className="text-[10px] text-muted-foreground">
   410→                    Kosongkan jika wali belum punya WA. Akun akan berstatus PENDING_NO_PHONE.
   411→                  </p>
   412→                </div>
   413→                <div className="space-y-1">
   414→                  <Label className="text-xs">Hubungan</Label>
   415→                  <Select
   416→                    value={form.waliRelationship || "WALI"}
   417→                    onValueChange={(v) => setForm({ ...form, waliRelationship: v })}
   418→                  >
   419→                    <SelectTrigger><SelectValue /></SelectTrigger>
   420→                    <SelectContent>
   421→                      <SelectItem value="AYAH">Ayah</SelectItem>
   422→                      <SelectItem value="IBU">Ibu</SelectItem>
   423→                      <SelectItem value="WALI">Wali</SelectItem>
   424→                      <SelectItem value="KAKEK">Kakek</SelectItem>
   425→                      <SelectItem value="NENEK">Nenek</SelectItem>
   426→                      <SelectItem value="PAMAN">Paman</SelectItem>
   427→                      <SelectItem value="BIBI">Bibi</SelectItem>
   428→                    </SelectContent>
   429→                  </Select>
   430→                </div>
   431→                {form.waliPhone && (
   432→                  <p className="text-[10px] text-emerald-600 sm:col-span-3">
   433→                    ✨ Wali akan otomatis dibuat dengan status PENDING. OTP & link aktivasi akan ditampilkan setelah santri disimpan.
   434→                  </p>
   435→                )}
   436→                {!form.waliPhone && form.waliName && (
   437→                  <p className="text-[10px] text-amber-600 sm:col-span-3">
   438→                    ⚠️ Wali tanpa nomor WhatsApp akan dibuat dengan status PENDING_NO_PHONE. Admin bisa tambahkan nomor WA nanti.
   439→                  </p>
   440→                )}
   441→              </div>
   442→            )}
   443→          </div>
   444→
   445→          <div className="space-y-1 sm:col-span-2">
   446→            <Label className="text-xs">Alamat</Label>
   447→            <Input value={form.address || ""} onChange={(e) => setForm({ ...form, address: e.target.value })} />
   448→          </div>
   449→          <div className="space-y-1">
   450→            <Label className="text-xs">Status</Label>
   451→            <Select value={form.status} onValueChange={(v) => setForm({ ...form, status: v })}>
   452→              <SelectTrigger><SelectValue /></SelectTrigger>
   453→              <SelectContent>
   454→                <SelectItem value="AKTIF">Aktif</SelectItem>
   455→                <SelectItem value="LULUS">Lulus</SelectItem>
   456→                <SelectItem value="PINDAH">Pindah</SelectItem>
   457→                <SelectItem value="KELUAR">Keluar</SelectItem>
   458→              </SelectContent>
   459→            </Select>
   460→          </div>
   461→        </div>
   462→        <DialogFooter>
   463→          <Button variant="outline" onClick={() => onOpenChange(false)}>Batal</Button>
   464→          <Button onClick={handleSave} disabled={saving}>{saving ? "Menyimpan..." : "Simpan"}</Button>
