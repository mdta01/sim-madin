    42→async function apiGet(path: string) {
    43→  const res = await fetch(`${API_BASE_URL}${path}`, { headers: authToken ? { Authorization: `Bearer ${authToken}` } : {} })
    44→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    45→  return res.json()
    46→}

    48→async function apiPost(path: string, body: any) {
    49→  const res = await fetch(`${API_BASE_URL}${path}`, {
    50→    method: "POST",
    51→    headers: { "Content-Type": "application/json", ...(authToken ? { Authorization: `Bearer ${authToken}` } : {}) },
    52→    body: JSON.stringify(body),
    53→  })
    54→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    55→  return res.json()
    56→}

   247→function AbsensiInputScreen({ route, navigation }: { route: any; navigation: any }) {
   248→  const kelasIdFromRoute = route?.params?.kelasId
   249→  const kelasNameFromRoute = route?.params?.kelasName
   250→
   251→  const [kelasList, setKelasList] = useState<any[]>([])
   252→  const [selectedKelas, setSelectedKelas] = useState(kelasIdFromRoute || "")
   253→  const [santriList, setSantriList] = useState<any[]>([])
   254→  const [absensiData, setAbsensiData] = useState<Record<string, string>>({})
   255→  const [selectedDate, setSelectedDate] = useState(new Date().toISOString().split("T")[0])
   256→  const [loading, setLoading] = useState(false)
   257→  const [saving, setSaving] = useState(false)
   258→
   259→  const loadKelas = useCallback(async () => {
   260→    try {
   261→      const res = await apiGet("/api/admin/kelas")
   262→      setKelasList(res.data || [])
   263→      if (!selectedKelas && res.data?.length > 0) {
   264→        setSelectedKelas(res.data[0].id)
   265→      }
   266→    } catch (e: any) { Alert.alert("Error", e.message) }
   267→  }, [selectedKelas])
   268→
   269→  const loadSantri = useCallback(async () => {
   270→    if (!selectedKelas) return
   271→    try {
   272→      setLoading(true)
   273→      const res = await apiGet(`/api/admin/santri?kelasId=${selectedKelas}&status=AKTIF`)
   274→      setSantriList(res.data || [])
   275→      // Default semua HADIR
   276→      const defaults: Record<string, string> = {}
   277→      res.data?.forEach((s: any) => { defaults[s.id] = "HADIR" })
   278→      setAbsensiData(defaults)
   279→    } catch (e: any) { Alert.alert("Error", e.message) }
   280→    finally { setLoading(false) }
   281→  }, [selectedKelas])
   282→
   283→  useEffect(() => { loadKelas() }, [loadKelas])
   284→  useEffect(() => { if (selectedKelas) loadSantri() }, [selectedKelas, loadSantri])
   285→
   286→  const setStatus = (santriId: string, status: string) => {
   287→    setAbsensiData(prev => ({ ...prev, [santriId]: status }))
   288→  }
   289→
   290→  const setAllStatus = (status: string) => {
   291→    const newData: Record<string, string> = {}
   292→    santriList.forEach(s => { newData[s.id] = status })
   293→    setAbsensiData(newData)
   294→  }
   295→
   296→  const handleSave = async () => {
   297→    if (santriList.length === 0) { Alert.alert("Error", "Belum ada santri"); return }
   298→    setSaving(true)
   299→    try {
   300→      const entries = Object.entries(absensiData).map(([santriId, status]) => ({ santriId, status }))
   301→      const res = await apiPost("/api/admin/absensi", { entries, kelasId: selectedKelas, date: selectedDate })
   302→      Alert.alert("Berhasil", `${res.count} absensi tersimpan`)
   303→      navigation.goBack()
   304→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   305→    finally { setSaving(false) }
   306→  }
   307→
   308→  return (
   309→    <SafeAreaView style={styles.screen}>
   310→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   311→      <View style={styles.header}>
   312→        <TouchableOpacity onPress={() => navigation.goBack()} style={styles.backButton}>
   313→          <Ionicons name="arrow-back" size={24} color={Colors.white} />
   314→        </TouchableOpacity>
   315→        <Text style={styles.headerTitle}>Input Absensi</Text>
   316→        <View style={{ width: 24 }} />
   317→      </View>
   318→
   319→      <View style={styles.absensiControls}>
   320→        {/* Date picker */}
   321→        <Text style={styles.label}>Tanggal</Text>
   322→        <TextInput
   323→          style={styles.dateInput}
   324→          value={selectedDate}
   325→          onChangeText={setSelectedDate}
   326→          placeholder="YYYY-MM-DD"
   327→        />
   328→
   329→        {/* Kelas selector */}
   330→        <Text style={styles.label}>Kelas</Text>
   331→        <View style={styles.pickerContainer}>
   332→          {kelasList.map((k) => (
   333→            <TouchableOpacity
   334→              key={k.id}
   335→              style={[styles.kelasPill, selectedKelas === k.id && styles.kelasPillActive]}
   336→              onPress={() => setSelectedKelas(k.id)}
   337→            >
   338→              <Text style={[styles.kelasPillText, selectedKelas === k.id && styles.kelasPillTextActive]}>
   339→                {k.name}
   340→              </Text>
   341→            </TouchableOpacity>
   342→          ))}
   343→        </View>
   344→
   345→        {/* Quick set all */}
   346→        <View style={styles.quickSetRow}>
   347→          <Text style={styles.quickSetLabel}>Set Semua:</Text>
   348→          {ABS_STATUS.map(s => (
   349→            <TouchableOpacity
   350→              key={s.value}
   351→              style={[styles.quickSetBtn, { backgroundColor: s.bg, borderColor: s.color }]}
   352→              onPress={() => setAllStatus(s.value)}
   353→            >
   354→              <Text style={[styles.quickSetText, { color: s.color }]}>{s.label}</Text>
   355→            </TouchableOpacity>
   356→          ))}
   357→        </View>
   358→      </View>
   359→
   360→      {loading ? (
   361→        <ActivityIndicator size="large" color={Colors.primary} style={{ marginTop: 20 }} />
   362→      ) : (
   363→        <FlatList
   364→          data={santriList}
   365→          keyExtractor={(item) => item.id}
   366→          renderItem={({ item }) => (
   367→            <View style={styles.absensiRow}>
   368→              <View style={styles.absensiInfo}>
   369→                <Text style={styles.absensiName}>{item.name}</Text>
   370→                <Text style={styles.absensiNis}>{item.nis}</Text>
   371→              </View>
   372→              <View style={styles.absensiButtons}>
   373→                {ABS_STATUS.map(s => (
   374→                  <TouchableOpacity
   375→                    key={s.value}
   376→                    style={[
   377→                      styles.absensiBtn,

   395→              <Text style={styles.emptyText}>Pilih kelas untuk melihat daftar santri</Text>
   396→            </View>
   397→          )}
   398→          contentContainerStyle={{ paddingBottom: 80 }}
   399→        />
   400→      )}
   401→
   402→      {/* Save button */}
   403→      {santriList.length > 0 && (
   404→        <TouchableOpacity
   405→          style={[styles.saveButton, saving && { opacity: 0.6 }]}
   406→          onPress={handleSave}
   407→          disabled={saving}
   408→        >
   409→          {saving ? <ActivityIndicator color={Colors.white} /> : (
   410→            <>
   411→              <Ionicons name="save" size={20} color={Colors.white} />
   412→              <Text style={styles.saveButtonText}>Simpan Absensi ({santriList.length} santri)</Text>
   413→            </>
   414→          )}
   415→        </TouchableOpacity>
   416→      )}
   417→    </SafeAreaView>
   418→  )
   419→}
