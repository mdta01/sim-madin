    43→async function apiGet(path: string) {
    44→  const res = await fetch(`${API_BASE_URL}${path}`, { headers: authToken ? { Authorization: `Bearer ${authToken}` } : {} })
    45→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    46→  return res.json()
    47→}

    49→async function apiPost(path: string, body: any) {
    50→  const res = await fetch(`${API_BASE_URL}${path}`, {
    51→    method: "POST",
    52→    headers: { "Content-Type": "application/json", ...(authToken ? { Authorization: `Bearer ${authToken}` } : {}) },
    53→    body: JSON.stringify(body),
    54→  })
    55→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    56→  return res.json()
    57→}

   254→function AbsensiInputScreen({ route, navigation }: { route: any; navigation: any }) {
   255→  const kelasIdFromRoute = route?.params?.kelasId
   256→  const kelasNameFromRoute = route?.params?.kelasName
   257→
   258→  const [kelasList, setKelasList] = useState<any[]>([])
   259→  const [selectedKelas, setSelectedKelas] = useState(kelasIdFromRoute || "")
   260→  const [santriList, setSantriList] = useState<any[]>([])
   261→  const [absensiData, setAbsensiData] = useState<Record<string, string>>({})
   262→  const [selectedDate, setSelectedDate] = useState(new Date().toISOString().split("T")[0])
   263→  const [loading, setLoading] = useState(false)
   264→  const [saving, setSaving] = useState(false)
   265→
   266→  const loadKelas = useCallback(async () => {
   267→    try {
   268→      const res = await apiGet("/api/admin/kelas")
   269→      setKelasList(res.data || [])
   270→      if (!selectedKelas && res.data?.length > 0) {
   271→        setSelectedKelas(res.data[0].id)
   272→      }
   273→    } catch (e: any) { Alert.alert("Error", e.message) }
   274→  }, [selectedKelas])
   275→
   276→  const loadSantri = useCallback(async () => {
   277→    if (!selectedKelas) return
   278→    try {
   279→      setLoading(true)
   280→      const res = await apiGet(`/api/admin/santri?kelasId=${selectedKelas}&status=AKTIF`)
   281→      setSantriList(res.data || [])
   282→      // Default semua HADIR
   283→      const defaults: Record<string, string> = {}
   284→      res.data?.forEach((s: any) => { defaults[s.id] = "HADIR" })
   285→      setAbsensiData(defaults)
   286→    } catch (e: any) { Alert.alert("Error", e.message) }
   287→    finally { setLoading(false) }
   288→  }, [selectedKelas])
   289→
   290→  useEffect(() => { loadKelas() }, [loadKelas])
   291→  useEffect(() => { if (selectedKelas) loadSantri() }, [selectedKelas, loadSantri])
   292→
   293→  const setStatus = (santriId: string, status: string) => {
   294→    setAbsensiData(prev => ({ ...prev, [santriId]: status }))
   295→  }
   296→
   297→  const setAllStatus = (status: string) => {
   298→    const newData: Record<string, string> = {}
   299→    santriList.forEach(s => { newData[s.id] = status })
   300→    setAbsensiData(newData)
   301→  }
   302→
   303→  const handleSave = async () => {
   304→    if (santriList.length === 0) { Alert.alert("Error", "Belum ada santri"); return }
   305→    setSaving(true)
   306→    try {
   307→      const entries = Object.entries(absensiData).map(([santriId, status]) => ({ santriId, status }))
   308→      const res = await apiPost("/api/admin/absensi", { entries, kelasId: selectedKelas, date: selectedDate })
   309→      Alert.alert("Berhasil", `${res.count} absensi tersimpan`)
   310→      navigation.goBack()
   311→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   312→    finally { setSaving(false) }
   313→  }
   314→
   315→  return (
   316→    <SafeAreaView style={styles.screen}>
   317→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   318→      <View style={styles.header}>
   319→        <TouchableOpacity onPress={() => navigation.goBack()} style={styles.backButton}>
   320→          <Ionicons name="arrow-back" size={24} color={Colors.white} />
   321→        </TouchableOpacity>
   322→        <Text style={styles.headerTitle}>Input Absensi</Text>
   323→        <View style={{ width: 24 }} />
   324→      </View>
   325→
   326→      <View style={styles.absensiControls}>
   327→        {/* Date picker */}
   328→        <Text style={styles.label}>Tanggal</Text>
   329→        <TextInput
   330→          style={styles.dateInput}
   331→          value={selectedDate}
   332→          onChangeText={setSelectedDate}
   333→          placeholder="YYYY-MM-DD"
   334→        />
   335→
   336→        {/* Kelas selector */}
   337→        <Text style={styles.label}>Kelas</Text>
   338→        <View style={styles.pickerContainer}>
   339→          {kelasList.map((k) => (
   340→            <TouchableOpacity
   341→              key={k.id}
   342→              style={[styles.kelasPill, selectedKelas === k.id && styles.kelasPillActive]}
   343→              onPress={() => setSelectedKelas(k.id)}
   344→            >
   345→              <Text style={[styles.kelasPillText, selectedKelas === k.id && styles.kelasPillTextActive]}>
   346→                {k.name}
   347→              </Text>
   348→            </TouchableOpacity>
   349→          ))}
   350→        </View>
   351→
   352→        {/* Quick set all */}
   353→        <View style={styles.quickSetRow}>
   354→          <Text style={styles.quickSetLabel}>Set Semua:</Text>
   355→          {ABS_STATUS.map(s => (
   356→            <TouchableOpacity
   357→              key={s.value}
   358→              style={[styles.quickSetBtn, { backgroundColor: s.bg, borderColor: s.color }]}
   359→              onPress={() => setAllStatus(s.value)}
   360→            >
   361→              <Text style={[styles.quickSetText, { color: s.color }]}>{s.label}</Text>
   362→            </TouchableOpacity>
   363→          ))}
   364→        </View>
   365→      </View>
   366→
   367→      {loading ? (
   368→        <ActivityIndicator size="large" color={Colors.primary} style={{ marginTop: 20 }} />
   369→      ) : (
   370→        <FlatList
   371→          data={santriList}
   372→          keyExtractor={(item) => item.id}
   373→          renderItem={({ item }) => (
   374→            <View style={styles.absensiRow}>
   375→              <View style={styles.absensiInfo}>
   376→                <Text style={styles.absensiName}>{item.name}</Text>
   377→                <Text style={styles.absensiNis}>{item.nis}</Text>
   378→              </View>
   379→              <View style={styles.absensiButtons}>
   380→                {ABS_STATUS.map(s => (
   381→                  <TouchableOpacity
   382→                    key={s.value}
   383→                    style={[
   384→                      styles.absensiBtn,
   385→                      absensiData[item.id] === s.value && { backgroundColor: s.bg, borderColor: s.color },
   386→                    ]}
   387→                    onPress={() => setStatus(item.id, s.value)}
   388→                  >
   389→                    <Text style={[
   390→                      styles.absensiBtnText,
   391→                      absensiData[item.id] === s.value ? { color: s.color, fontWeight: "bold" } : { color: Colors.textMuted },
   392→                    ]}>
   393→                      {s.label}
   394→                    </Text>
   395→                  </TouchableOpacity>
   396→                ))}
   397→              </View>
   398→            </View>
   399→          )}
   400→          ListEmptyComponent={() => (
   401→            <View style={styles.emptyCard}>
   402→              <Text style={styles.emptyText}>Pilih kelas untuk melihat daftar santri</Text>
   403→            </View>
   404→          )}
   405→          contentContainerStyle={{ paddingBottom: 80 }}
   406→        />
   407→      )}
   408→
   409→      {/* Save button */}
   410→      {santriList.length > 0 && (
   411→        <TouchableOpacity
   412→          style={[styles.saveButton, saving && { opacity: 0.6 }]}
   413→          onPress={handleSave}
   414→          disabled={saving}
   415→        >
   416→          {saving ? <ActivityIndicator color={Colors.white} /> : (
   417→            <>
   418→              <Ionicons name="save" size={20} color={Colors.white} />
   419→              <Text style={styles.saveButtonText}>Simpan Absensi ({santriList.length} santri)</Text>
   420→            </>
   421→          )}
   422→        </TouchableOpacity>
   423→      )}
   424→    </SafeAreaView>
   425→  )
   426→}
