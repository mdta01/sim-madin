    24→let supabase: any = null
    25→function getSupabase() {
    26→  if (supabase) return supabase
    27→  supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY, {
    28→    auth: {
    29→      storage: {
    30→        getItem: async (k: string) => { try { return await SecureStore.getItemAsync(k) } catch { return null } },
    31→        setItem: async (k: string, v: string) => { try { await SecureStore.setItemAsync(k, v) } catch {} },
    32→        removeItem: async (k: string) => { try { await SecureStore.deleteItemAsync(k) } catch {} },
    33→      },
    34→      autoRefreshToken: true, persistSession: true,
    35→    },
    36→  })
    37→  return supabase
    38→}

    42→async function apiGet(path: string) {
    43→  const res = await fetch(`${API_BASE_URL}${path}`, { headers: authToken ? { Authorization: `Bearer ${authToken}` } : {} })
    44→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    45→  return res.json()
    46→}
    47→
    48→async function apiPost(path: string, body: any) {
    49→  const res = await fetch(`${API_BASE_URL}${path}`, {
    50→    method: "POST",
    51→    headers: { "Content-Type": "application/json", ...(authToken ? { Authorization: `Bearer ${authToken}` } : {}) },
    52→    body: JSON.stringify(body),
    53→  })
    54→  if (!res.ok) throw new Error((await res.json())?.error || `Failed: ${res.status}`)
    55→  return res.json()
    56→}

   213→function PembayaranInputScreen({ navigation }: { navigation: any }) {
   214→  const [santriList, setSantriList] = useState<any[]>([])
   215→  const [tagihanList, setTagihanList] = useState<any[]>([])
   216→  const [selectedSantri, setSelectedSantri] = useState("")
   217→  const [selectedTagihan, setSelectedTagihan] = useState("")
   218→  const [amount, setAmount] = useState("")
   219→  const [method, setMethod] = useState("TUNAI")
   220→  const [loading, setLoading] = useState(false)
   221→  const [saving, setSaving] = useState(false)
   222→  const [searchText, setSearchText] = useState("")
   223→
   224→  const loadSantri = useCallback(async () => {
   225→    try {
   226→      setLoading(true)
   227→      const res = await apiGet("/api/admin/santri?status=AKTIF")
   228→      setSantriList(res.data || [])
   229→    } catch (e: any) { Alert.alert("Error", e.message) }
   230→    finally { setLoading(false) }
   231→  }, [])
   232→
   233→  const loadTagihan = useCallback(async () => {
   234→    if (!selectedSantri) { setTagihanList([]); return }
   235→    try {
   236→      const res = await apiGet(`/api/admin/tagihan?santriId=${selectedSantri}`)
   237→      const unpaid = (res.data || []).filter((t: any) => t.status !== "LUNAS")
   238→      setTagihanList(unpaid)
   239→    } catch (e: any) { Alert.alert("Error", e.message) }
   240→  }, [selectedSantri])
   241→
   242→  useEffect(() => { loadSantri() }, [loadSantri])
   243→  useEffect(() => { loadTagihan() }, [loadTagihan])
   244→
   245→  const selectTagihan = (tagihanId: string, tagihanAmount: number) => {
   246→    setSelectedTagihan(tagihanId)
   247→    setAmount(String(tagihanAmount - (tagihanList.find(t => t.id === tagihanId)?.paidAmount || 0)))
   248→  }
   249→
   250→  const handleSave = async () => {
   251→    if (!selectedSantri || !amount || Number(amount) <= 0) {
   252→      Alert.alert("Validasi", "Pilih santri dan isi nominal")
   253→      return
   254→    }
   255→    setSaving(true)
   256→    try {
   257→      await apiPost("/api/admin/pembayaran", {
   258→        santriId: selectedSantri,
   259→        tagihanId: selectedTagihan || undefined,
   260→        amount: Number(amount),
   261→        paymentMethod: method,
   262→      })
   263→      Alert.alert("Berhasil", `Pembayaran ${formatRupiah(Number(amount))} tercatat`)
   264→      navigation.goBack()
   265→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   266→    finally { setSaving(false) }
   267→  }
   268→
   269→  const filteredSantri = santriList.filter(s =>
   270→    !searchText || s.name.toLowerCase().includes(searchText.toLowerCase()) || s.nis.includes(searchText)
   271→  )
   272→
   273→  return (
   274→    <SafeAreaView style={styles.screen}>
   275→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   276→      <View style={styles.header}>
   277→        <TouchableOpacity onPress={() => navigation.goBack()} style={styles.backButton}>
   278→          <Ionicons name="arrow-back" size={24} color={Colors.white} />
   279→        </TouchableOpacity>
   280→        <Text style={styles.headerTitle}>Input Pembayaran</Text>
   281→        <View style={{ width: 24 }} />
   282→      </View>
   283→
   284→      <ScrollView style={styles.content}>
   285→        {/* Search Santri */}
   286→        <Text style={styles.label}>Cari Santri</Text>
   287→        <View style={styles.searchContainer}>
   288→          <Ionicons name="search" size={20} color={Colors.textMuted} style={{ marginRight: 8 }} />
   289→          <TextInput
   290→            style={styles.searchInput}
   291→            placeholder="Nama atau NIS..."
   292→            value={searchText}
   293→            onChangeText={setSearchText}
   294→          />
   295→        </View>
   296→
   297→        {/* Santri List */}
   298→        {loading ? (
   299→          <ActivityIndicator size="large" color={Colors.primary} style={{ marginTop: 20 }} />
   300→        ) : (
   301→          <View>
   302→            {filteredSantri.slice(0, 5).map(s => (
   303→              <TouchableOpacity
   304→                key={s.id}
   305→                style={[styles.santriOption, selectedSantri === s.id && styles.santriOptionActive]}
   306→                onPress={() => { setSelectedSantri(s.id); setSelectedTagihan(""); setAmount("") }}
   307→              >
   308→                <View style={{ flex: 1 }}>
   309→                  <Text style={styles.santriOptionName}>{s.name}</Text>
   310→                  <Text style={styles.santriOptionNis}>{s.nis} • {s.kelas?.name}</Text>
   311→                </View>
   312→                {selectedSantri === s.id && <Ionicons name="checkmark-circle" size={24} color={Colors.primary} />}
   313→              </TouchableOpacity>
   314→            ))}
   315→            {filteredSantri.length > 5 && (
   316→              <Text style={styles.moreText}>+{filteredSantri.length - 5} santri lainnya (ketik untuk filter)</Text>
   317→            )}
   318→          </View>
   319→        )}
   320→
   321→        {/* Tagihan List (if santri selected) */}
   322→        {selectedSantri && tagihanList.length > 0 && (
   323→          <View>
   324→            <Text style={styles.label}>Tagihan Belum Lunas</Text>
   325→            {tagihanList.map(t => (
   326→              <TouchableOpacity
   327→                key={t.id}
   328→                style={[styles.tagihanOption, selectedTagihan === t.id && styles.tagihanOptionActive]}
   329→                onPress={() => selectTagihan(t.id, t.amount)}
   330→              >
   331→                <View style={{ flex: 1 }}>
   332→                  <Text style={styles.tagihanOptionName}>{t.jenisTagihan?.name}</Text>
   333→                  <Text style={styles.tagihanOptionDesc}>
   334→                    Total: {formatRupiah(t.amount)} • Sisa: {formatRupiah(t.amount - t.paidAmount)}
   335→                  </Text>
   336→                </View>
   337→                {selectedTagihan === t.id && <Ionicons name="checkmark-circle" size={24} color={Colors.primary} />}
   338→              </TouchableOpacity>
   339→            ))}
   340→          </View>
   341→        )}
   342→
   343→        {/* Amount */}
   344→        <Text style={styles.label}>Nominal Pembayaran</Text>
   345→        <View style={styles.amountContainer}>
   346→          <Text style={styles.amountPrefix}>Rp</Text>
   347→          <TextInput
   348→            style={styles.amountInput}
   349→            value={amount}
   350→            onChangeText={(v) => setAmount(v.replace(/\D/g, ""))}
   351→            keyboardType="numeric"
   352→            placeholder="0"
   353→          />
   354→        </View>
   355→
   356→        {/* Method */}
   357→        <Text style={styles.label}>Metode Pembayaran</Text>
   358→        <View style={styles.methodRow}>
   359→          {PAYMENT_METHODS.map(m => (
   360→            <TouchableOpacity
   361→              key={m.value}
   362→              style={[styles.methodBtn, method === m.value && styles.methodBtnActive]}

   368→        </View>
   369→
   370→        {/* Save */}

   372→          style={[styles.saveButton, saving && { opacity: 0.6 }]}
   373→          onPress={handleSave}
   374→          disabled={saving}
   375→        >
   376→          {saving ? <ActivityIndicator color={Colors.white} /> : (

   384→        <View style={{ height: 40 }} />
   385→      </ScrollView>

   644→      <Stack.Navigator screenOptions={{ headerShown: false }}>
   645→        {isLoggedIn ? (
   646→          <>
   647→            <Stack.Screen name="Dashboard">

   649→                <DashboardScreen
   650→                  navigation={navigation}
   651→                  onLogout={async () => { await getSupabase().auth.signOut(); await SecureStore.deleteItemAsync("authToken"); authToken = null; setIsLoggedIn(false) }}
   652→                />
   653→              )}

   662→        )}
   663→      </Stack.Navigator>
   664→    </NavigationContainer>
   665→  )
   666→}
