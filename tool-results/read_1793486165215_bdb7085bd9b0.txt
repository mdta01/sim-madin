    99→    99→function LoginScreen({ onLogin, onForgotPin }: { onLogin: (token: string, wali: WaliData, santriList: SantriData[]) => void; onForgotPin?: () => void }) {
   100→   100→  const [phone, setPhone] = useState("")

   102→   102→  const [loading, setLoading] = useState(false)
   103→   103→  const [showPin, setShowPin] = useState(false)
   104→   104→
   105→   105→  const handleLogin = async () => {
   106→   106→    if (!phone || !pin) {
   107→   107→      Alert.alert("Validasi", "Nomor WhatsApp dan PIN wajib diisi")
   108→   108→      return
   109→   109→    }
   110→   110→    if (pin.length !== 6) {
   111→   111→      Alert.alert("Validasi", "PIN harus 6 digit")
   112→   112→      return
   113→   113→    }
   114→   114→
   115→   115→    setLoading(true)
   116→   116→    try {
   117→   117→      const data = await api.post("/api/wali/login", { phone, pin })
   118→   118→      if (data.success) {
   119→   119→        waliToken = data.token
   120→   120→        await saveWaliSession(data.token, data.wali, data.santriList)
   121→   121→        onLogin(data.token, data.wali, data.santriList)
   122→   122→      } else {
   123→   123→        Alert.alert("Login Gagal", data.error || "Terjadi kesalahan")
   124→   124→      }
   125→   125→    } catch (e: any) {
   126→   126→      Alert.alert("Login Gagal", e.message)
   127→   127→    } finally {
   128→   128→      setLoading(false)
   129→   129→    }
   130→   130→  }
   131→   131→
   132→   132→  return (

   199→   199→// --- Forgot PIN Screen (3-step OTP flow) ---
   200→   200→function ForgotPinScreen({ onBack }: { onBack: () => void }) {
   201→   201→  const [step, setStep] = useState<"phone" | "otp" | "setPin">("phone")
   202→   202→  const [phone, setPhone] = useState("")
   203→   203→  const [otp, setOtp] = useState("")
   204→   204→  const [newPin, setNewPin] = useState("")
   205→   205→  const [confirmPin, setConfirmPin] = useState("")
   206→   206→  const [loading, setLoading] = useState(false)
   207→   207→  const [verifyToken, setVerifyToken] = useState("")
   208→   208→
   209→   209→  const handleRequestOtp = async () => {
   210→   210→    if (!phone) { Alert.alert("Validasi", "Nomor WhatsApp wajib diisi"); return }
   211→   211→    setLoading(true)
   212→   212→    try {
   213→   213→      const res = await api.post("/api/wali/request-otp", { phone, type: "FORGOT_PIN" })
   214→   214→      Alert.alert("OTP Terkirim", res.devOtp ? `Kode OTP (DEV): ${res.devOtp}` : res.message)
   215→   215→      setStep("otp")
   216→   216→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   217→   217→    finally { setLoading(false) }
   218→   218→  }
   219→   219→
   220→   220→  const handleVerifyOtp = async () => {
   221→   221→    if (!otp || otp.length !== 6) { Alert.alert("Validasi", "OTP harus 6 digit"); return }
   222→   222→    setLoading(true)
   223→   223→    try {
   224→   224→      const res = await api.post("/api/wali/verify-otp", { phone, otpCode: otp })
   225→   225→      if (res.verifyToken) {
   226→   226→        setVerifyToken(res.verifyToken)
   227→   227→        setStep("setPin")
   228→   228→        Alert.alert("Terverifikasi", "Silakan buat PIN baru")
   229→   229→      } else { Alert.alert("Gagal", "Token verifikasi tidak diterima") }
   230→   230→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   231→   231→    finally { setLoading(false) }
   232→   232→  }
   233→   233→
   234→   234→  const handleSetPin = async () => {
   235→   235→    if (!newPin || newPin.length !== 6) { Alert.alert("Validasi", "PIN harus 6 digit"); return }
   236→   236→    if (newPin !== confirmPin) { Alert.alert("Validasi", "PIN tidak cocok"); return }
   237→   237→    setLoading(true)
   238→   238→    try {
   239→   239→      await api.post("/api/wali/set-pin", { verifyToken, pin: newPin })
   240→   240→      Alert.alert("Berhasil", "PIN berhasil diubah. Silakan login dengan PIN baru.", [{ text: "OK", onPress: onBack }])
   241→   241→    } catch (e: any) { Alert.alert("Gagal", e.message) }
   242→   242→    finally { setLoading(false) }
   243→   243→  }
   244→   244→
   245→   245→  return (
   246→   246→    <SafeAreaView style={styles.loginContainer}>
   247→   247→      <StatusBar barStyle="light-content" backgroundColor={Colors.primary} />
   248→   248→      <ScrollView contentContainerStyle={styles.loginScroll}>
   249→   249→        <View style={styles.loginLogo}>
   250→   250→          <Ionicons name="key" size={80} color={Colors.white} />
   251→   251→          <Text style={styles.loginTitle}>Lupa PIN</Text>
   252→   252→          <Text style={styles.loginSubtitle}>Reset PIN via OTP</Text>
   253→   253→        </View>
   254→   254→
   255→   255→        <View style={styles.loginCard}>
   256→   256→          {step === "phone" && (
   257→   257→            <>
   258→   258→              <Text style={styles.loginCardTitle}>Masukkan Nomor WhatsApp</Text>
   259→   259→              <Text style={styles.loginCardDesc}>Kami akan kirim kode OTP ke nomor Anda</Text>
   260→   260→              <Text style={styles.label}>Nomor WhatsApp</Text>
   261→   261→              <View style={styles.inputContainer}>
   262→   262→                <Ionicons name="call" size={20} color={Colors.textMuted} style={styles.inputIcon} />
   263→   263→                <TextInput style={styles.input} placeholder="08xxx atau 628xxx" value={phone} onChangeText={setPhone} keyboardType="phone-pad" />
   264→   264→              </View>
   265→   265→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleRequestOtp} disabled={loading}>
   266→   266→                {loading ? <ActivityIndicator color={Colors.white} /> : <Text style={styles.loginButtonText}>Kirim OTP</Text>}
   267→   267→              </TouchableOpacity>
   268→   268→            </>
   269→   269→          )}
   270→   270→
   271→   271→          {step === "otp" && (
   272→   272→            <>
   273→   273→              <Text style={styles.loginCardTitle}>Verifikasi OTP</Text>
   274→   274→              <Text style={styles.loginCardDesc}>Masukkan 6 digit kode yang dikirim</Text>
   275→   275→              <TextInput
   276→   276→                style={[styles.input, { textAlign: "center", fontSize: 24, letterSpacing: 8, fontFamily: "monospace" }]}
   277→   277→                placeholder="000000"
   278→   278→                value={otp}
   279→   279→                onChangeText={(t) => setOtp(t.replace(/\D/g, "").slice(0, 6))}
   280→   280→                keyboardType="numeric"
   281→   281→                maxLength={6}
   282→   282→              />
   283→   283→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleVerifyOtp} disabled={loading}>
   284→   284→                {loading ? <ActivityIndicator color={Colors.white} /> : <Text style={styles.loginButtonText}>Verifikasi</Text>}
   285→   285→              </TouchableOpacity>
   286→   286→            </>
   287→   287→          )}
   288→   288→
   289→   289→          {step === "setPin" && (
   290→   290→            <>
   291→   291→              <Text style={styles.loginCardTitle}>Buat PIN Baru</Text>
   292→   292→              <Text style={styles.loginCardDesc}>PIN 6 digit untuk login</Text>
   293→   293→              <Text style={styles.label}>PIN Baru</Text>
   294→   294→              <TextInput
   295→   295→                style={[styles.input, { textAlign: "center", fontSize: 20, letterSpacing: 6, fontFamily: "monospace" }]}
   296→   296→                placeholder="••••••"
   297→   297→                value={newPin}
   298→   298→                onChangeText={(t) => setNewPin(t.replace(/\D/g, "").slice(0, 6))}
   299→   299→                keyboardType="numeric"
   300→   300→                secureTextEntry
   301→   301→                maxLength={6}
   302→   302→              />
   303→   303→              <Text style={styles.label}>Konfirmasi PIN</Text>
   304→   304→              <TextInput
   305→   305→                style={[styles.input, { textAlign: "center", fontSize: 20, letterSpacing: 6, fontFamily: "monospace" }]}
   306→   306→                placeholder="••••••"
   307→   307→                value={confirmPin}

   311→   311→                maxLength={6}
   312→   312→              />
   313→   313→              <TouchableOpacity style={[styles.loginButton, loading && { opacity: 0.6 }]} onPress={handleSetPin} disabled={loading}>

   391→   391→function SantriDetailScreen({ santri, onBack }: { santri: SantriData, onBack: () => void }) {
   392→   392→  const [detail, setDetail] = useState<SantriDetail | null>(null)
   393→   393→  const [perkembangan, setPerkembangan] = useState<PerkembanganV2 | null>(null)
   394→   394→  const [loading, setLoading] = useState(true)
   395→   395→  const [error, setError] = useState<string | null>(null)
   396→   396→  const [activeTab, setActiveTab] = useState("profile")
   397→   397→
   398→   398→  const loadDetail = useCallback(async () => {
   399→   399→    try {
   400→   400→      setLoading(true)
   401→   401→      setError(null)
   402→   402→      // V2: fetch both detail (V1) and perkembangan (V2) in parallel
   403→   403→      const [detailRes, perkRes] = await Promise.all([
   404→   404→        api.get(`/api/wali/santri/${santri.id}`),
   405→   405→        api.get(`/api/wali/perkembangan/${santri.id}`).catch(() => null), // best-effort
   406→   406→      ])
   407→   407→      setDetail(detailRes.data)
   408→   408→      setPerkembangan(perkRes?.data || null)
   409→   409→    } catch (e: any) {
   410→   410→      setError(e.message)
   411→   411→    } finally {
   412→   412→      setLoading(false)
   413→   413→    }
   414→   414→  }, [santri.id])
   415→   415→
   416→   416→  useEffect(() => { loadDetail() }, [loadDetail])

   766→   766→function WaliNotifikasiSection({ token }: { token: string | null }) {
   767→   767→  const [list, setList] = useState<any[]>([])
   768→   768→  const [loading, setLoading] = useState(true)
   769→   769→  const [stats, setStats] = useState<any>(null)
   770→   770→
   771→   771→  const load = useCallback(async () => {
   772→   772→    if (!token) return
   773→   773→    try {
   774→   774→      setLoading(true)
   775→   775→      const res = await fetch(`${API_BASE_URL}/api/wali/notifikasi`, {
   776→   776→        headers: { Authorization: `Bearer ${token}` },
   777→   777→      })
   778→   778→      const data = await res.json()
   779→   779→      setList(data.data || [])
   780→   780→      setStats(data.stats)
   781→   781→    } catch (e: any) {
   782→   782→      console.error("notif load error:", e.message)
   783→   783→    } finally {
   784→   784→      setLoading(false)
   785→   785→    }
   786→   786→  }, [token])
   787→   787→
   788→   788→  useEffect(() => { load() }, [load])
   789→   789→
   790→   790→  const markRead = async (id: string) => {
   791→   791→    try {
   792→   792→      await fetch(`${API_BASE_URL}/api/wali/notifikasi/${id}`, {
   793→   793→        method: "PUT",
   794→   794→        headers: { Authorization: `Bearer ${token}` },
   795→   795→      })
   796→   796→      load()
   797→   797→    } catch (e: any) {
   798→   798→      Alert.alert("Gagal", e.message)
   799→   799→    }
   800→   800→  }
   801→   801→
   802→   802→  if (loading) return <View style={styles.contentCard}><ActivityIndicator size="large" color={Colors.primary} /></View>
   803→   803→
   804→   804→  return (
   805→   805→    <View style={styles.contentCard}>
   806→   806→      <Text style={styles.contentTitle}>Notifikasi</Text>
   807→   807→      {stats && stats.totalUnread > 0 && (
   808→   808→        <View style={[styles.statusBadge, { backgroundColor: Colors.dangerLight, alignSelf: "flex-start", marginBottom: 8 }]}>
   809→   809→          <Text style={[styles.statusBadgeText, { color: Colors.danger }]}>{stats.totalUnread} belum dibaca</Text>
   810→   810→        </View>
   811→   811→      )}
   812→   812→      {list.length === 0 ? (
   813→   813→        <Text style={styles.emptyTextSmall}>Tidak ada notifikasi</Text>
   814→   814→      ) : (
