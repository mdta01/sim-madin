    81→// Helper: auto-generate isi surat dari data santri/guru berdasarkan tipe.
    82→// Dipakai saat santri dipilih & content textarea kosong (dapat di-edit manual).
    83→// ----------------------------------------------------------------------------
    84→function generateSuratContent(type: string, santri: any, guru: any, periodeMulai?: string, periodeSelesai?: string): string {
    85→  if (!santri && !guru) return ""
    86→  const today = new Date()
    87→  const dateStr = formatDate(today)
    88→  const namaMadrasah = "Madrasah Diniyah Al-Hidayah"
    89→
    90→  const santriBlock = santri
    91→    ? [
    92→        `Nama: ${santri.name}`,
    93→        `NIS: ${santri.nis || "-"}`,
    94→        santri.kelas?.name ? `Kelas: ${santri.kelas.name}` : null,
    95→        santri.birthPlace ? `Tempat/Tgl Lahir: ${santri.birthPlace}${santri.birthDate ? `, ${formatDate(santri.birthDate)}` : ""}` : null,
    96→      ].filter(Boolean).join("\n")
    97→    : ""
    98→
    99→  const guruBlock = guru
   100→    ? [`Nama: ${guru.name}`, guru.nip ? `NIP: ${guru.nip}` : null].filter(Boolean).join("\n")
   101→    : ""
   102→
   103→  switch (type) {
   104→    case "AKTIF":
   105→      return `Yang bertanda tangan di bawah ini, Kepala ${namaMadrasah}, menerangkan dengan sesungguhnya bahwa:\n\n${santriBlock}\n\nAdalah benar-benar santri yang masih aktif mengikuti pembelajaran di ${namaMadrasah} pada tahun ajaran berjalan.\n\nDemikian surat keterangan aktif ini dibuat untuk dapat dipergunakan sebagaimana mestinya.\n\n${dateStr}`
   106→    case "IZIN":
   107→      return `Yang bertanda tangan di bawah ini, Kepala ${namaMadrasah}, memberikan izin kepada:\n\n${santriBlock || guruBlock}\n\nuntuk tidak mengikuti kegiatan pembelajaran${periodeMulai ? ` mulai ${formatDate(periodeMulai)}` : ""}${periodeSelesai ? ` sampai ${formatDate(periodeSelesai)}` : ""}.\n\nDemikian surat izin ini dibuat untuk dapat dipergunakan sebagaimana mestinya.\n\n${dateStr}`
   108→    case "TUGAS":
   109→      return `Yang bertanda tangan di bawah ini, Kepala ${namaMadrasah}, dengan ini menugaskan kepada:\n\n${guruBlock || santriBlock}\n\nuntuk melaksanakan tugas sebagaimana mestinya${periodeMulai ? ` mulai ${formatDate(periodeMulai)}` : ""}${periodeSelesai ? ` sampai ${formatDate(periodeSelesai)}` : ""}.\n\nDemikian surat tugas ini dibuat untuk dapat dilaksanakan dengan penuh tanggung jawab.\n\n${dateStr}`
   110→    case "UNDANGAN":
   111→      return `Dengan hormat,\n\nKami mengundang wali dari santri:\n\n${santriBlock}\n\nuntuk hadir pada kegiatan yang akan diselenggarakan oleh ${namaMadrasah}${periodeMulai ? ` pada ${formatDate(periodeMulai)}` : ""}.\n\nDemikian undangan ini kami sampaikan. Atas perhatian dan kehadirannya kami ucapkan terima kasih.\n\n${dateStr}`
   112→    case "KETERANGAN":
   113→      return `Yang bertanda tangan di bawah ini, Kepala ${namaMadrasah}, menerangkan bahwa:\n\n${santriBlock || guruBlock}\n\nadalah benar warga ${namaMadrasah}.\n\nDemikian surat keterangan ini dibuat untuk dapat dipergunakan sebagaimana mestinya.\n\n${dateStr}`
   114→    case "KEPUTUSAN":
   115→      return `KEPUTUSAN KEPALA ${namaMadrasah.toUpperCase()}\n\nNomor: .../KEP/${today.getFullYear()}\n\nKepala ${namaMadrasah}, menetapkan:\n\n${santriBlock || guruBlock}\n\nDemikian keputusan ini dibuat untuk dapat dilaksanakan sebagaimana mestinya.\n\n${dateStr}`
   116→    case "KELULUSAN":
   117→      return `Yang bertanda tangan di bawah ini, Kepala ${namaMadrasah}, menerangkan bahwa:\n\n${santriBlock}\n\ndinyatakan LULUS dari ${namaMadrasah} pada tahun ${today.getFullYear()}.\n\nDemikian surat keterangan kelulusan ini dibuat untuk dapat dipergunakan sebagaimana mestinya.\n\n${dateStr}`
   118→    default:
   119→      return ""
   120→  }
   121→}

   158→function SuratTab() {
   159→  const [data, setData] = useState<any[]>([])
   160→  const [santriList, setSantriList] = useState<any[]>([])
   161→  const [guruList, setGuruList] = useState<any[]>([])
   162→  const [loading, setLoading] = useState(true)
   163→  const [error, setError] = useState<string | null>(null)
   164→  const [search, setSearch] = useState("")
   165→  const [typeFilter, setTypeFilter] = useState("")
   166→  const [statusFilter, setStatusFilter] = useState("")
   167→  const [dialogOpen, setDialogOpen] = useState(false)
   168→  const [editing, setEditing] = useState<any | null>(null)
   169→  const { toast } = useToast()
   170→
   171→  const load = useCallback(async () => {
   172→    try {
   173→      setLoading(true)
   174→      setError(null)
   175→      const params: any = {}
   176→      if (search) params.search = search
   177→      if (typeFilter) params.type = typeFilter
   178→      if (statusFilter) params.status = statusFilter
   179→      const [res, santriRes, guruRes] = await Promise.all([
   180→        apiGet("/api/admin/surat", params),
   181→        apiGet("/api/admin/santri"),
   182→        apiGet("/api/admin/guru"),
   183→      ])
   184→      setData(res.data || [])
   185→      setSantriList(santriRes.data || [])
   186→      setGuruList(guruRes.data || [])
   187→    } catch (e: any) {
   188→      setError(e.message)
   189→    } finally {
   190→      setLoading(false)
   191→    }
   192→  }, [search, typeFilter, statusFilter])
   193→
   194→  useEffect(() => {
   195→    load()
   196→  }, [load])
   197→
   198→  const handleDelete = async (id: string, nomor: string) => {
   199→    if (!confirm(`Yakin ingin menghapus surat "${nomor}"?`)) return
   200→    try {
   201→      await apiDelete(`/api/admin/surat/${id}`)
   202→      toast({ title: "Berhasil", description: "Surat dihapus" })
   203→      load()
   204→    } catch (e: any) {
   205→      toast({ title: "Gagal", description: e.message, variant: "destructive" })
   206→    }
   207→  }
   208→
   209→  const handlePrintPdf = (id: string) => {
   210→    window.open(`/api/admin/surat/${id}/pdf`, "_blank")
   211→  }
   212→
   213→  // Resolve nama penerima (santri atau guru)
   214→  const resolvePenerima = (s: any): string => {
   215→    if (s.santri) return s.santri.name
   216→    if (s.guru) return s.guru.name
   217→    if (s.penerimaNama) return s.penerimaNama
   218→    return "-"
   219→  }
   220→
   221→  return (
   222→    <div>
   223→      <DataTableToolbar
   224→        searchValue={search}
   225→        onSearchChange={setSearch}
   226→        searchPlaceholder="Cari nomor surat / judul..."
   227→        filters={
   228→          <>
   229→            <Select
   230→              value={typeFilter}
   231→              onValueChange={(v) => setTypeFilter(v === "all" ? "" : v)}
   232→            >
   233→              <SelectTrigger className="w-[180px] h-9">
   234→                <SelectValue placeholder="Semua Tipe" />
   235→              </SelectTrigger>
   236→              <SelectContent>
   237→                <SelectItem value="all">Semua Tipe</SelectItem>
   238→                {SURAT_TYPES.map((t) => (
   239→                  <SelectItem key={t.value} value={t.value}>
   240→                    {t.label}
   241→                  </SelectItem>
   242→                ))}
   243→              </SelectContent>
   244→            </Select>
   245→            <Select
   246→              value={statusFilter}
   247→              onValueChange={(v) => setStatusFilter(v === "all" ? "" : v)}
   248→            >
   249→              <SelectTrigger className="w-[160px] h-9">
   250→                <SelectValue placeholder="Semua Status" />
   251→              </SelectTrigger>
   252→              <SelectContent>
   253→                <SelectItem value="all">Semua Status</SelectItem>
   254→                {Object.entries(SURAT_STATUS).map(([k, v]) => (
   255→                  <SelectItem key={k} value={k}>
   256→                    {v.label}
   257→                  </SelectItem>
   258→                ))}
   259→              </SelectContent>
   260→            </Select>
   261→          </>
   262→        }
   263→        actions={
   264→          <Button
   265→            onClick={() => {
   266→              setEditing(null)
   267→              setDialogOpen(true)
   268→            }}
   269→          >
   270→            <Plus className="h-4 w-4 mr-1" /> Buat Surat
   271→          </Button>
   272→        }
   273→      />
   274→      <Card>
   275→        <CardContent className="p-0">
   276→          {loading ? (
   277→            <LoadingState />
   278→          ) : error ? (
   279→            <ErrorState message={error} onRetry={load} />
   280→          ) : data.length === 0 ? (
   281→            <EmptyState
   282→              icon={<FileText className="h-5 w-5" />}
   283→              title="Belum ada surat"
   284→              description="Klik 'Buat Surat' untuk membuat surat pertama."
   285→            />
   286→          ) : (
   287→            <div className="overflow-x-auto">
   288→              <Table>
   289→                <TableHeader>
   290→                  <TableRow>
   291→                    <TableHead className="min-w-[140px]">Nomor Surat</TableHead>
   292→                    <TableHead className="min-w-[200px]">Judul</TableHead>
   293→                    <TableHead>Tipe</TableHead>
   294→                    <TableHead>Penerima</TableHead>
   295→                    <TableHead>Status</TableHead>
   296→                    <TableHead>Tanggal</TableHead>
   297→                    <TableHead className="text-right">Aksi</TableHead>
   298→                  </TableRow>
   299→                </TableHeader>
   300→                <TableBody>
   301→                  {data.map((s) => {
   302→                    const tipe = SURAT_TYPES.find((t) => t.value === s.type)
   303→                    const status = SURAT_STATUS[s.status]
   304→                    return (
   305→                      <TableRow key={s.id}>
   306→                        <TableCell>
   307→                          <span className="font-mono text-xs">{s.nomorSurat || "-"}</span>
   308→                        </TableCell>
   309→                        <TableCell>
   310→                          <p className="font-medium text-sm">{s.title}</p>
   311→                          {s.tempat && (
   312→                            <p className="text-xs text-muted-foreground">{s.tempat}</p>
   313→                          )}
   314→                        </TableCell>
   315→                        <TableCell>
   316→                          <Badge variant="outline" className="text-xs">
   317→                            {tipe?.label || s.type}
   318→                          </Badge>
   319→                        </TableCell>
   320→                        <TableCell className="text-sm">{resolvePenerima(s)}</TableCell>
   321→                        <TableCell>
   322→                          {status ? (
   323→                            <Badge variant={badgeVariant(status.variant)}>
   324→                              {status.label}
   325→                            </Badge>
   326→                          ) : (
   327→                            <Badge variant="outline">{s.status}</Badge>
   328→                          )}
   329→                        </TableCell>
   330→                        <TableCell className="text-xs">
   331→                          {s.tanggalSurat ? formatDate(s.tanggalSurat) : "-"}
   332→                        </TableCell>
   333→                        <TableCell className="text-right">
   334→                          <div className="flex justify-end gap-1">
   335→                            <Button
   336→                              variant="ghost"
   337→                              size="icon"
   338→                              className="h-8 w-8"
   339→                              title="Cetak PDF"
   340→                              onClick={() => handlePrintPdf(s.id)}
   341→                            >
   342→                              <Printer className="h-3.5 w-3.5" />
   343→                            </Button>
   344→                            <Button
   345→                              variant="ghost"
   346→                              size="icon"
   347→                              className="h-8 w-8"
   348→                              title="Edit"
   349→                              onClick={() => {
   350→                                setEditing(s)
   351→                                setDialogOpen(true)
   352→                              }}
   353→                            >
   354→                              <Pencil className="h-3.5 w-3.5" />
   355→                            </Button>
   356→                            <Button
   357→                              variant="ghost"
   358→                              size="icon"
   359→                              className="h-8 w-8 text-destructive"
   360→                              title="Hapus"
   361→                              onClick={() => handleDelete(s.id, s.nomorSurat || s.title)}
   362→                            >
   363→                              <Trash2 className="h-3.5 w-3.5" />
   364→                            </Button>
   365→                          </div>
   366→                        </TableCell>
   367→                      </TableRow>
   368→                    )
   369→                  })}
   370→                </TableBody>
   371→              </Table>
   372→            </div>
   373→          )}
   374→        </CardContent>
   375→      </Card>
   376→
   377→      <SuratForm
   378→        open={dialogOpen}
   379→        onOpenChange={setDialogOpen}
   380→        editing={editing}
   381→        santriList={santriList}
   382→        guruList={guruList}
   383→        onSuccess={load}
   384→      />
   385→    </div>
   386→  )
   387→}

   389→// ----------------------------------------------------------------------------
   390→// Surat Form Dialog
   391→// ----------------------------------------------------------------------------
   392→function SuratForm({
   393→  open,
   394→  onOpenChange,
   395→  editing,
   396→  santriList,
   397→  guruList,
   398→  onSuccess,
   399→}: {
   400→  open: boolean
   401→  onOpenChange: (v: boolean) => void
   402→  editing: any | null
   403→  santriList: any[]
   404→  guruList: any[]
   405→  onSuccess: () => void
   406→}) {
   407→  const [form, setForm] = useState<any>({})
   408→  const [saving, setSaving] = useState(false)
   409→  const { toast } = useToast()
   410→
   411→  // Inisialisasi form ketika dialog dibuka
   412→  useEffect(() => {
   413→    if (!open) return
   414→    if (editing) {
   415→      setForm({
   416→        ...editing,
   417→        tanggalSurat: editing.tanggalSurat
   418→          ? new Date(editing.tanggalSurat).toISOString().split("T")[0]
   419→          : "",
   420→        periodeMulai: editing.periodeMulai
   421→          ? new Date(editing.periodeMulai).toISOString().split("T")[0]
   422→          : "",
   423→        periodeSelesai: editing.periodeSelesai
   424→          ? new Date(editing.periodeSelesai).toISOString().split("T")[0]
   425→          : "",
   426→      })
   427→    } else {
   428→      setForm({
   429→        type: "AKTIF",
   430→        status: "DRAFT",
   431→        tanggalSurat: new Date().toISOString().split("T")[0],
   432→      })
   433→    }
   434→  }, [editing, open])
   435→
   436→  // Auto-fill content ketika santri/guru dipilih (hanya jika content kosong)
   437→  const handleSantriChange = (santriId: string) => {
   438→    const santri = santriList.find((s) => s.id === santriId)
   439→    const next = { ...form, santriId }
   440→    if (santri && (!form.content || form.content.trim() === "")) {
   441→      next.content = generateSuratContent(
   442→        form.type,
   443→        santri,
   444→        guruList.find((g) => g.id === form.guruId),
   445→        form.periodeMulai,
   446→        form.periodeSelesai,
   447→      )
   448→    }
   449→    setForm(next)
   450→  }
   451→
   452→  const handleGuruChange = (guruId: string) => {
   453→    const guru = guruList.find((g) => g.id === guruId)
   454→    const next = { ...form, guruId }
   455→    if (guru && (!form.content || form.content.trim() === "")) {
   456→      next.content = generateSuratContent(
   457→        form.type,
   458→        santriList.find((s) => s.id === form.santriId),
   459→        guru,
   460→        form.periodeMulai,
   461→        form.periodeSelesai,
   462→      )
   463→    }
   464→    setForm(next)
   465→  }
   466→
   467→  // Regenerate content secara manual (tombol "Auto-generate")
   468→  const handleRegenerate = () => {
   469→    const santri = santriList.find((s) => s.id === form.santriId)
   470→    const guru = guruList.find((g) => g.id === form.guruId)
   471→    if (!santri && !guru) {
   472→      toast({
   473→        title: "Pilih penerima dulu",
   474→        description: "Pilih santri atau guru terlebih dahulu.",
   475→        variant: "destructive",
   476→      })
   477→      return
   478→    }
   479→    const content = generateSuratContent(
   480→      form.type,
   481→      santri,
   482→      guru,
   483→      form.periodeMulai,
   484→      form.periodeSelesai,
   485→    )
   486→    setForm({ ...form, content })
   487→    toast({ title: "Isi surat dihasilkan", description: "Masih dapat diedit manual." })
   488→  }
   489→
   490→  const handleSave = async () => {
   491→    if (!form.type || !form.title) {
   492→      toast({
   493→        title: "Validasi gagal",
   494→        description: "Tipe dan judul surat wajib diisi.",
   495→        variant: "destructive",
   496→      })
   497→      return
   498→    }
   499→    if (form.type === "TUGAS" && !form.guruId) {
   500→      toast({
   501→        title: "Validasi gagal",
   502→        description: "Surat Tugas wajib memilih guru penerima.",
   503→        variant: "destructive",
   504→      })
   505→      return
   506→    }
   507→    setSaving(true)
   508→    try {
   509→      if (editing) {
   510→        await apiPut(`/api/admin/surat/${editing.id}`, form)
   511→        toast({ title: "Berhasil", description: "Surat diperbarui" })
   512→      } else {
   513→        await apiPost("/api/admin/surat", form)
   514→        toast({ title: "Berhasil", description: "Surat baru dibuat" })
   515→      }
   516→      onOpenChange(false)
   517→      onSuccess()
   518→    } catch (e: any) {
   519→      toast({ title: "Gagal", description: e.message, variant: "destructive" })
   520→    } finally {
   521→      setSaving(false)
   522→    }
   523→  }
   524→
   525→  const showGuru = form.type === "TUGAS"
   526→

   747→// ----------------------------------------------------------------------------
   748→// Dokumen Select Dialog — untuk kartu santri / berita acara / kartu ujian
   749→// Komponen di-remount (via key) setiap kali `type` berubah, sehingga state
   750→// internal (selectedSantri/selectedUjian) otomatis ter-reset.
   751→// ----------------------------------------------------------------------------
   752→function DokumenSelectDialog({
   753→  type,
   754→  onClose,
   755→}: {
   756→  type: string
   757→  onClose: () => void
   758→}) {
   759→  const [santriList, setSantriList] = useState<any[]>([])
   760→  const [ujianList, setUjianList] = useState<any[]>([])
   761→  const [loading, setLoading] = useState(false)
   762→  const [selectedSantri, setSelectedSantri] = useState<string>("")
   763→  const [selectedUjian, setSelectedUjian] = useState<string>("")
   764→  const { toast } = useToast()
   765→
   766→  // Load data pendukung sesuai tipe dokumen
   767→  const load = useCallback(async () => {
   768→    setLoading(true)
   769→    try {
   770→      if (type === "KARTU_SANTRI") {
   771→        const res = await apiGet("/api/admin/santri")
   772→        setSantriList(res.data || [])
   773→      } else if (type === "BERITA_ACARA" || type === "KARTU_UJIAN") {
   774→        const res = await apiGet("/api/admin/ujian")
   775→        setUjianList(res.data || [])
   776→      }
   777→    } catch (e: any) {
   778→      toast({ title: "Gagal memuat data", description: e.message, variant: "destructive" })
   779→    } finally {
   780→      setLoading(false)
   781→    }
   782→  }, [type, toast])
   783→
   784→  useEffect(() => {
   785→    load()
   786→  }, [load])
   787→
   788→  const titleMap: Record<string, string> = {
   789→    KARTU_SANTRI: "Cetak Kartu Santri",
   790→    BERITA_ACARA: "Cetak Berita Acara Ujian",
   791→    KARTU_UJIAN: "Cetak Kartu Ujian",
   792→  }
   793→
   794→  const handlePreview = () => {
   795→    if (type === "KARTU_SANTRI") {
   796→      if (!selectedSantri) {
   797→        toast({ title: "Pilih santri", description: "Pilih santri terlebih dahulu.", variant: "destructive" })
   798→        return
   799→      }
   800→      window.open(`/api/admin/dokumen/kartu-santri/${selectedSantri}`, "_blank")
   801→    } else if (type === "BERITA_ACARA") {
   802→      if (!selectedUjian) {
   803→        toast({ title: "Pilih ujian", description: "Pilih ujian terlebih dahulu.", variant: "destructive" })
   804→        return
   805→      }
   806→      window.open(`/api/admin/dokumen/berita-acara/${selectedUjian}`, "_blank")
   807→    } else if (type === "KARTU_UJIAN") {
   808→      if (!selectedUjian) {
   809→        toast({ title: "Pilih ujian", description: "Pilih ujian terlebih dahulu.", variant: "destructive" })
   810→        return
   811→      }
   812→      // Redirect ke endpoint kartu-pdf ujian (per-ujian, semua peserta)
   813→      window.open(`/api/admin/ujian/${selectedUjian}/kartu-pdf`, "_blank")
   814→    }
   815→    onClose()
   816→  }
   817→
   818→  return (
   819→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   820→      <DialogContent className="max-w-lg">
   821→        <DialogHeader>
   822→          <DialogTitle>{titleMap[type] || "Pilih Dokumen"}</DialogTitle>
   823→        </DialogHeader>
   824→        {loading ? (
   825→          <LoadingState />
   826→        ) : (
   827→          <div className="space-y-3 py-2">
   828→            {type === "KARTU_SANTRI" && (
   829→              <div className="space-y-1">
   830→                <Label className="text-xs">Pilih Santri</Label>
   831→                <Select value={selectedSantri} onValueChange={setSelectedSantri}>
   832→                  <SelectTrigger>
   833→                    <SelectValue placeholder="Cari & pilih santri..." />
   834→                  </SelectTrigger>
   835→                  <SelectContent>
   836→                    {santriList.map((s) => (
   837→                      <SelectItem key={s.id} value={s.id}>
   838→                        {s.name} {s.nis ? `(${s.nis})` : ""}
   839→                      </SelectItem>
   840→                    ))}
   841→                  </SelectContent>
   842→                </Select>
   843→                <p className="text-[10px] text-muted-foreground">
   844→                  Kartu identitas santri digital akan dibuka di tab baru.
   845→                </p>
   846→              </div>
   847→            )}
   848→
   849→            {(type === "BERITA_ACARA" || type === "KARTU_UJIAN") && (
   850→              <div className="space-y-1">
   851→                <Label className="text-xs">Pilih Ujian</Label>
   852→                <Select value={selectedUjian} onValueChange={setSelectedUjian}>
   853→                  <SelectTrigger>
   854→                    <SelectValue placeholder="Pilih ujian..." />
   855→                  </SelectTrigger>
   856→                  <SelectContent>
   857→                    {ujianList.map((u) => (
   858→                      <SelectItem key={u.id} value={u.id}>
   859→                        {u.title}
   860→                      </SelectItem>
   861→                    ))}
   862→                  </SelectContent>
   863→                </Select>
   864→                <p className="text-[10px] text-muted-foreground">
   865→                  {type === "BERITA_ACARA"
   866→                    ? "Berita acara ujian akan dibuka di tab baru."
   867→                    : "Kartu ujian untuk semua peserta akan dibuka di tab baru."}
   868→                </p>
   869→              </div>
   870→            )}
   871→
   872→            {type === "KARTU_SANTRI" && santriList.length === 0 && (
   873→              <EmptyState
   874→                icon={<IdCard className="h-5 w-5" />}
   875→                title="Belum ada data santri"
   876→                description="Tambahkan santri terlebih dahulu."
   877→              />
   878→            )}
   879→            {(type === "BERITA_ACARA" || type === "KARTU_UJIAN") && ujianList.length === 0 && (
   880→              <EmptyState
   881→                icon={<CalendarClock className="h-5 w-5" />}
   882→                title="Belum ada ujian"
   883→                description="Buat ujian terlebih dahulu di menu Ujian."
   884→              />
   885→            )}
   886→          </div>
   887→        )}
   888→        <DialogFooter>
   889→          <Button variant="outline" onClick={onClose}>
   890→            Batal
   891→          </Button>
   892→          <Button onClick={handlePreview} disabled={loading}>
   893→            <ExternalLink className="h-4 w-4 mr-1" /> Buka Dokumen
   894→          </Button>
   895→        </DialogFooter>
   896→      </DialogContent>
   897→    </Dialog>
   898→  )
   899→}
